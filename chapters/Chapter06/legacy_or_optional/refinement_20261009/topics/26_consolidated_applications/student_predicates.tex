\subsection{Student টেবিলে শর্তের অতিরিক্ত প্রয়োগ}
এখানে অংশ~\ref{sec:ch06-student-master}-এর একই Student টেবিল ব্যবহার করা হয়েছে। মূল পাঠে প্রতিটি operator-এর নিয়ম একবার আলোচনা করা হয়েছে; নিচে নাম, পরিচিতি, ভিন্ন সীমা ও pattern-এর ফল দেখানো হলো।
\begin{minipage}{\linewidth}
\chSixExample{৭.৪}{Rahim নামের শিক্ষার্থীর সব তথ্য দেখাও।}

\begin{chSixSideBySide}[.32]
\begin{verbatim}
SELECT *
FROM Student
WHERE Name='Rahim';
\end{verbatim}
\chSixNextColumn
\textbf{ফল}\par\smallskip
{\raggedright\footnotesize\setlength{\tabcolsep}{2pt}
\begin{tabular}{|l|l|l|l|l|l|}\hline\chSixHeader
Roll & Name & Department & GPA & Fee & Email\\\hline
102 & Rahim & Science & 5.00 & 600 & rahim@test.com\\
\hline\end{tabular}\par}
\end{chSixSideBySide}
\end{minipage}
\begin{minipage}{\linewidth}
