\begin{boardbox}{ORDER BY-এর গুরুত্বপূর্ণ তথ্য}

\begin{itemize}

\item ASC = Ascending।

\item DESC = Descending।

\item ASC/DESC না লিখলে সাধারণত ঊর্ধ্বক্রম ব্যবহৃত হয়।

\item সাজানোর জন্য একাধিক ফিল্ড থাকলে বামদিকের ফিল্ড আগে বিবেচিত হয়।

\item শুধু \texttt{ORDER BY GPA DESC} দিলে সমান GPA রেকর্ডের
নিজেদের মধ্যকার নির্দিষ্ট ক্রম নিশ্চিত নয়।

\end{itemize}

\end{boardbox}
