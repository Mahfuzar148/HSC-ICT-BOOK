\section{বহুনির্বাচনি প্রশ্ন (MCQ)}

\subsection*{\textbf{\underline{ডেটাবেস বেসিক ও মডেল}}}

\begin{multicols}{2}
\begin{enumerate}

\item ডেটাবেস (Database) এর প্রধান কাজ কোনটি? \textbf{[ঢাকা-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item তথ্য সংগ্রহ ও ব্যবস্থাপনা \checkmark
\item শুধু ডেটা ডিলিট করা
\item হার্ডওয়্যার নিয়ন্ত্রণ
\item কোনোটিই নয়
\end{enumerate}

\item DBMS এর পূর্ণরূপ কী? \textbf{[চট্টগ্রাম-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item Database Management System \checkmark
\item Data Base Maintenance System
\item Digital Base Management System
\item Data Block Management System
\end{enumerate}

\item ডেটাবেস টেবিলের প্রতিটি সারিকে (Row) কী বলে? \textbf{[রাজশাহী-২০২২]}
\begin{enumerate}[label=\alph*)]
\item রেকর্ড (Record) \checkmark
\item ফিল্ড (Field)
\item অ্যাট্রিবিউট
\item রিলেশন
\end{enumerate}

\item ডেটাবেস টেবিলের প্রতিটি কলামকে (Column) কী বলে? \textbf{[সিলেট-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item ফিল্ড (Field) \checkmark
\item রেকর্ড
\item টাপল
\item রিলেশন
\end{enumerate}

\item প্রাইমারি কি (Primary Key) এর প্রধান বৈশিষ্ট্য কোনটি? \textbf{[বরিশাল-২০২২]}
\begin{enumerate}[label=\alph*)]
\item ইউনিক এবং নাল হতে পারবে না \checkmark
\item ডুপ্লিকেট হতে পারবে
\item নাল হতে পারবে
\item কোনোটিই নয়
\end{enumerate}

\end{enumerate}
\end{multicols}

\subsection*{\textbf{\underline{বহুপদী সমাপ্তিসূচক ও অভিন্ন তথ্যভিত্তিক প্রশ্ন}}}

\begin{multicols}{2}
\begin{enumerate}[resume]

\item \textbf{নিচের উদ্দীপকটি পড়ো এবং প্রশ্নের উত্তর দাও:}

\textit{একটি স্কুলের ছাত্র এবং লাইব্রেরির বইয়ের মধ্যে সম্পর্ক স্থাপনের জন্য একটি ডেটাবেস ডিজাইন করা হলো। যেখানে একজন ছাত্র অনেকগুলো বই নিতে পারে এবং একটি বই অনেক ছাত্র নিতে পারে।}

\item উদ্দীপকের রিলেশনটি হলো—
\begin{enumerate}[label=\alph*)]
\item One to One
\item One to Many
\item Many to Many \checkmark
\item Many to One
\end{enumerate}

\item রিলেশনাল ডেটাবেসের বৈশিষ্ট্য হলো—
\begin{enumerate}[label=\Roman*.]
\item ডেটা টেবিলে সংরক্ষিত থাকে
\item টেবিলগুলোর মধ্যে সম্পর্ক থাকে
\item ডেটা ডুপ্লিকেট হয় না
\end{enumerate}

নিচের কোনটি সঠিক?
\begin{enumerate}[label=\alph*)]
\item i ও ii \checkmark
\item i ও iii
\item ii ও iii
\item i, ii ও iii
\end{enumerate}

\end{enumerate}
\end{multicols}
