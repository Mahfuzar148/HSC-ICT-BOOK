\section{HTML ট্যাগ, এলিমেন্ট, অ্যাট্রিবিউট ও সিনট্যাক্স}

% Composition pass 2026-10-04
একই লেখায় একটি শব্দ ক্লিকযোগ্য আর অন্যটি শুধু অনুচ্ছেদের অংশ হতে পারে।

\chFourMotivation{উপাদান ও বৈশিষ্ট্য আলাদা করা $\rightarrow$ ট্যাগ, এলিমেন্ট ও অ্যাট্রিবিউট $\rightarrow$ সঠিক নিয়মে গঠন ও গন্তব্য লেখা $\rightarrow$ ব্রাউজার প্রত্যাশিত ফল দেখায়।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item HTML ট্যাগ, এলিমেন্ট, অ্যাট্রিবিউট, বিষয়বস্তু ও অ্যাট্রিবিউটের মান আলাদা করতে।
\item শুরুর ট্যাগ, শেষের ট্যাগ, জোড়া ট্যাগ ও এম্পটি/ভয়েড ট্যাগ শনাক্ত করতে।
\item HTML-এর সঠিক সিনট্যাক্স ও সাধারণ সিনট্যাক্সের ভুল বোঝাতে।
\item ছোট HTML কোড বিশ্লেষণ করে প্রতিটি অংশের কাজ বলতে।
\end{itemize}
\end{learningbox}


\subsection{HTML ট্যাগ কী}
\begin{definitionbox}{ট্যাগ (Tag)}
HTML-এ এলিমেন্টের শুরু বা শেষ চিহ্নিত করার জন্য কৌণিক বন্ধনী \texttt{< >}-সহ লেখা নির্দেশনাকে ট্যাগ বলে। যেমন, \texttt{<p>} অনুচ্ছেদ শুরু করে এবং \texttt{</p>} শেষ করে। বন্ধনীর ভেতরের \texttt{p} হলো ট্যাগের নাম।
\end{definitionbox}

\begin{definitionbox}{বিষয়বস্তু (Content)}
শুরুর ট্যাগ ও শেষের ট্যাগের মাঝখানে যে লেখা, ছবি, লিংক বা অন্য এলিমেন্ট থাকে তাকে বিষয়বস্তু বলা হয়। যেমন \texttt{<h1>Welcome to ICT</h1>}-এ \texttt{Welcome to ICT} বিষয়বস্তু।
\end{definitionbox}

\begin{definitionbox}{এলিমেন্ট (Element)}
HTML ডকুমেন্টের একটি গঠনগত উপাদানকে \textbf{এলিমেন্ট} বলে। সাধারণ এলিমেন্ট শুরুর ট্যাগ, ভেতরের বিষয়বস্তু ও শেষের ট্যাগ দিয়ে লেখা হয়; যেমন \texttt{<h1>Title</h1>}। ভয়েড এলিমেন্টের শেষের ট্যাগ ও ভেতরের বিষয়বস্তু থাকে না।
\end{definitionbox}

\begin{definitionbox}{অ্যাট্রিবিউট (Attribute)}
HTML এলিমেন্টের বৈশিষ্ট্য বা অতিরিক্ত তথ্য প্রকাশের জন্য ব্যবহৃত নামকে অ্যাট্রিবিউট বলে। অ্যাট্রিবিউট শুরুর ট্যাগে লেখা হয়। যেমন, \texttt{<a href="page.html">Link</a>}-এ \texttt{href} গন্তব্য নির্দেশকারী অ্যাট্রিবিউট।
\end{definitionbox}

\begin{definitionbox}{অ্যাট্রিবিউটের মান (Attribute Value)}
কোনো অ্যাট্রিবিউটে বৈশিষ্ট্যের যে নির্দিষ্ট তথ্য দেওয়া হয়, তাকে অ্যাট্রিবিউটের মান বলে। যেমন \texttt{<p align="center">ICT</p>}-এ \texttt{center} হলো \texttt{align} অ্যাট্রিবিউটের মান।
\end{definitionbox}

\begin{keypointbox}{এক নজরে}
\begin{itemize}
\item ট্যাগ = এলিমেন্টের শুরু বা শেষ বোঝানোর চিহ্ন
\item সাধারণ এলিমেন্ট = শুরুর ট্যাগ + বিষয়বস্তু + শেষের ট্যাগ
\item অ্যাট্রিবিউট = অতিরিক্ত তথ্য
\item মান = অ্যাট্রিবিউটের মান
\end{itemize}
\end{keypointbox}

\subsection{ট্যাগের ধরন}
জোড়া ট্যাগ ও এম্পটি ট্যাগের পার্থক্য হলো শেষের ট্যাগ ও ভেতরের বিষয়বস্তুর উপস্থিতি। অ্যাট্রিবিউট দুই ধরনের ট্যাগেই থাকতে পারে; তাই নিচের তৃতীয় সারিটি আলাদা, পরস্পর বিচ্ছিন্ন শ্রেণি নয়।
\begin{center}
\begin{tabular}{|p{0.22\textwidth}|p{0.30\textwidth}|p{0.34\textwidth}|}
\hline
\textbf{ধরন} & \textbf{উদাহরণ} & \textbf{কাজ} \\
\hline
জোড়া ট্যাগ / কন্টেইনার ট্যাগ & \texttt{<p>...</p>} & বিষয়বস্তু ঘিরে রাখে \\
\hline
এম্পটি ট্যাগ / ভয়েড ট্যাগ & \texttt{<br>}, \texttt{<img>} & শেষের ট্যাগ থাকে না; নিজেই কাজ সম্পন্ন করে \\
\hline
অ্যাট্রিবিউটসহ ট্যাগ & \texttt{<img src="a.jpg" alt="A">} & অতিরিক্ত তথ্য দেয় \\
\hline
\end{tabular}
\end{center}

\begin{definitionbox}{জোড়া ট্যাগ (Paired Tag)}
যে HTML ট্যাগের শুরুর ট্যাগ ও শেষের ট্যাগ দু'টি অংশ থাকে এবং মাঝখানে বিষয়বস্তু থাকে, তাকে জোড়া ট্যাগ বলে। যেমন \texttt{<p>Bangladesh</p>}।
\end{definitionbox}

\begin{definitionbox}{এম্পটি ট্যাগ ও ভয়েড এলিমেন্ট (Empty Tag / Void Element)}
যে HTML এলিমেন্টের ভেতরে বিষয়বস্তু রাখা যায় না এবং যার শেষের ট্যাগ লেখা হয় না, তাকে \textbf{ভয়েড এলিমেন্ট} বলে। পাঠ্যবইয়ে এগুলোর শুরুর ট্যাগকে এম্পটি ট্যাগ বলা হয়। যেমন \texttt{<br>}, \texttt{<hr>}, \texttt{<img>}।
\end{definitionbox}

\begin{keypointbox}{গুরুত্বপূর্ণ এম্পটি ট্যাগ}
নিচের ভয়েড এলিমেন্টগুলোর শেষের ট্যাগ লেখা হয় না।
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{<br>} — লাইনবিরতি
\item \texttt{<hr>} — বিষয়গত বিভাজক, সাধারণত অনুভূমিক রেখা
\item \texttt{<img>} — ছবি
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{<meta>} — পেজের তথ্য বা এনকোডিং ঘোষণা
\item \texttt{<link>} - বাহ্যিক সম্পদের লিংক, যেমন CSS ফাইল
\item \texttt{<base>} - আপেক্ষিক URL-এর ভিত্তি ঠিকানা
\end{itemize}
\end{minipage}\par

\end{keypointbox}

\subsection{HTML সিনট্যাক্স}
সিনট্যাক্স মেনে লিখলে কোডের গঠন স্পষ্ট থাকে। ভুল কোড ব্রাউজার কখনো নিজে সামলে দেখালেও সেই ফল কাঙ্ক্ষিত গঠনের সঙ্গে নাও মিলতে পারে।

\begin{definitionbox}{সিনট্যাক্স (Syntax)}
HTML ডকুমেন্টে ট্যাগ, এলিমেন্ট, অ্যাট্রিবিউট ও বিষয়বস্তু সঠিকভাবে লেখার নিয়মকে সিনট্যাক্স বলে।
\end{definitionbox}

\begin{center}
\fbox{\parbox{0.92\textwidth}{
\textbf{লেখার সাধারণ ছক}\\[4pt]
\texttt{<tag>content</tag>}\\
\texttt{<tag attribute="value">content</tag>}\\
\texttt{<empty-tag>}\\[2pt]
ট্যাগ সাধারণত কৌণিক বন্ধনীর মধ্যে থাকে, অ্যাট্রিবিউট শুরুর ট্যাগের মধ্যে থাকে, আর শেষের ট্যাগে স্ল্যাশ \texttt{/} ব্যবহার করা হয়।
}}
\end{center}

\begin{keypointbox}{HTML সিনট্যাক্সের প্রধান নিয়ম}
\begin{itemize}
\item ট্যাগ অবশ্যই \texttt{< >} কৌণিক বন্ধনীর মধ্যে লিখতে হবে।
\item শেষের ট্যাগে মূলশব্দের আগে \texttt{/} স্ল্যাশ থাকবে।
\item অ্যাট্রিবিউট সাধারণত শুরুর ট্যাগের মধ্যে লিখতে হবে।
\item মানসহ অ্যাট্রিবিউটে নাম ও মানের মাঝে \texttt{=} চিহ্ন থাকবে। \texttt{required}-এর মতো সত্য/মিথ্যা অবস্থা বোঝানো Boolean অ্যাট্রিবিউট শুধু নাম দিয়েও লেখা যায়।
\item অ্যাট্রিবিউট মান উদ্ধৃতির মধ্যে লেখা উত্তম।
\item ট্যাগ নেস্টিং সঠিক ক্রমে থাকতে হবে।
\item HTML ট্যাগের ইংরেজি বড় ও ছোট হাতের অক্ষর একইভাবে বোঝা হয়; তবে ছোট হাতের অক্ষরে লেখা প্রচলিত ভালো অভ্যাস।
\item ভয়েড এলিমেন্টের শেষের ট্যাগ লিখবে না।
\end{itemize}
\end{keypointbox}

\subsection{নেস্টেড HTML এলিমেন্ট}

\begin{definitionbox}{নেস্টেড এলিমেন্ট (Nested Element)}
একটি HTML এলিমেন্টের বিষয়বস্তু হিসেবে আরেকটি HTML এলিমেন্ট ব্যবহার করা হলে তাকে নেস্টেড HTML এলিমেন্ট বলা হয়।
\end{definitionbox}

নেস্টেড এলিমেন্টে ভেতরের এলিমেন্ট আগে বন্ধ হয়, তারপর বাইরের এলিমেন্ট বন্ধ হয়। এই ক্রমে বন্ধ করলে এলিমেন্টগুলোর সম্পর্ক ঠিক থাকে।

\par\noindent

  This is <b><i>important</i></b> text.
</p>

\par\smallskip
\end{minipage}\hfill

\textbf{\color{chFourTeal}ব্যাখ্যা}\par
এখানে \texttt{<i>} ট্যাগ \texttt{<b>} ট্যাগের ভিতরে আছে। তাই আগে \texttt{</i>} এবং পরে \texttt{</b>} লেখা হয়েছে। এটি সঠিক নেস্টিং।
\par\smallskip
\end{minipage}\par


\begin{mistakebox}{ভুল নেস্টিং}
\texttt{<b><i>ICT</b></i>} ভুল, কারণ এখানে \texttt{<i>} ট্যাগ আগে বন্ধ হওয়ার কথা ছিল, কিন্তু \texttt{</b>} আগে লেখা হয়েছে। তাই নেস্টিং ক্রম ভুল।
\end{mistakebox}

\subsection{সঠিক ও ভুল সিনট্যাক্স}
\begin{examplebox}{সিনট্যাক্সের উদাহরণ}
\textbf{১. অনুচ্ছেদ ট্যাগ}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<p>ICT is an important subject.</p>
\end{verbatim}
\end{minipage}\par
সঠিক, কারণ শুরুর ট্যাগ, বিষয়বস্তু এবং শেষের ট্যাগ তিনটিই আছে।

\vspace{3pt}
\textbf{২. অ্যাট্রিবিউটসহ ট্যাগ}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<p align="center">ICT</p>
\end{verbatim}
\end{minipage}\par
এখানে \texttt{align} হলো অ্যাট্রিবিউট এবং \texttt{center} তার মান। পরীক্ষায় এমন পুরোনো রূপ দেখা যায়। বর্তমান HTML মানে \texttt{<p>}-এর \texttt{align} পুরোনো বৈশিষ্ট্য; নতুন কাজে CSS-এর \texttt{text-align} ব্যবহার করা হয়।

\vspace{3pt}
\textbf{৩. হাইপারলিংক ট্যাগ}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<a href="next.html" target="_blank">Next Page</a>
\end{verbatim}
\end{minipage}\par
এখানে \texttt{href} গন্তব্য পেজ নির্দেশ করে, আর \texttt{target} লিংক কোথায় খুলবে তা বোঝায়।

\vspace{3pt}
\textbf{৪. ছবি ট্যাগ}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<img src="images/photo.jpg" alt="Photo description" width="300">
\end{verbatim}
\end{minipage}\par
\texttt{img} একটি এম্পটি ট্যাগ; তাই এর শেষের ট্যাগ নেই। \texttt{src} ছবির অবস্থান দেখায় এবং \texttt{alt} বিকল্প লেখা দেয়।

\vspace{3pt}
\textbf{৫. ভুল নেস্টিং}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<b><i>ICT</b></i>
\end{verbatim}
\end{minipage}\par
এটি ভুল, কারণ ট্যাগ বন্ধ করার ক্রম ঠিক নেই।
\end{examplebox}

\begin{mistakebox}{সাধারণ ভুল}
\begin{itemize}
\item ট্যাগ ও এলিমেন্ট এক মনে করা।
\item অ্যাট্রিবিউটকে বিষয়বস্তু মনে করা।
\item ফাঁক বা বিশেষ চিহ্নযুক্ত অ্যাট্রিবিউটের মান প্রয়োজনীয় উদ্ধৃতিচিহ্ন ছাড়া লেখা। কিছু সাধারণ মান উদ্ধৃতিচিহ্ন ছাড়াও বৈধ, তবে উদ্ধৃতিতে লিখলে ভুল কমে।
\item শেষের ট্যাগে \texttt{/} না দেওয়া।
\item এম্পটি ট্যাগের শেষের ট্যাগ লিখতে যাওয়া।
\item ট্যাগ নেস্টিং উল্টে দেওয়া।
\item \texttt{center} বা \texttt{photo.jpg}-এর মতো মানকে অ্যাট্রিবিউট বলে ভুল করা।
\end{itemize}
\end{mistakebox}

\subsection{কোডের ভুল খুঁজে বের করা}

\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\textbf{\color{chFourRose}ভুল কোড}\par
\begin{verbatim}
<table border="1">
  <td>Name</td>
  <td>Roll</td>
</table>

<image src="photo.jpg">
<a src="index.html">Home</a>
\end{verbatim}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\textbf{\color{chFourTeal}সঠিক কোড}\par
\begin{verbatim}
<table border="1">
  <tr>
    <td>Name</td>
    <td>Roll</td>
  </tr>
</table>

<img src="photo.jpg" alt="Photo">
<a href="index.html">Home</a>
\end{verbatim}
\end{minipage}
\par\smallskip
\textbf{ভুলের কারণ:}
\begin{itemize}
\item \texttt{<td>} সরাসরি \texttt{<table>} ট্যাগের ভেতরে লেখা সঠিক গঠন নয়; ঘরগুলোকে \texttt{<tr>} সারিতে রাখতে হয়।
\item ছবি দেখানোর সঠিক ট্যাগ হলো \texttt{<img>}; \texttt{<image>} লেখা সঠিক HTML নয়।
\item হাইপারলিংকের গন্তব্য জানাতে \texttt{href} অ্যাট্রিবিউট ব্যবহার করতে হয়। এখানে \texttt{src} ব্যবহার ভুল।
\end{itemize}
