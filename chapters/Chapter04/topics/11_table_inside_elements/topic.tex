\section{সারণির ঘরে তালিকা, ছবি, হাইপারলিংক ও নেস্টেড সারণি}

% Composition pass 2026-10-04
লাইব্রেরির তালিকায় একই বইয়ের নাম, প্রচ্ছদ, বিষয় ও বিস্তারিত লিংক একসঙ্গে দরকার।

\chFourMotivation{সম্পর্কিত তথ্য একই ঘরে রাখা $\rightarrow$ ঘরের ভেতর HTML এলিমেন্ট $\rightarrow$ td/th-এর মধ্যে তালিকা, ছবি ও লিংক বসানো $\rightarrow$ বই বা পণ্যের সমন্বিত সারণি।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item HTML সারণির ঘরের ভেতরে তালিকা, ছবি, হাইপারলিংক ও ফরম্যাটিং ট্যাগ ব্যবহার করতে।
\item সারণির ঘরের ভেতরে ক্রমিক তালিকা ও অক্রমিক তালিকা বসাতে।
\item ছবির হাইপারলিংক তৈরি করতে।
\item নেস্টেড সারণি কী তা ব্যাখ্যা করতে এবং নেস্টেড সারণির সহজ কোড লিখতে।
\item সৃজনশীল ও ব্যবহারিক প্রশ্নে সারণির ভেতরে একাধিক HTML উপাদান ব্যবহার করে কোড লিখতে।
\end{itemize}
\end{learningbox}



একটি বইয়ের তালিকায় এক কলামে বইয়ের নাম এবং পাশের কলামে তার ছবি, বিষয়গুলোর তালিকা ও বিস্তারিত জানার লিংক দেখাতে চাই। HTML সারণির ঘরে কেবল লেখা নয়; তালিকা, ছবি, হাইপারলিংক ও আরেকটি সারণিও রাখা যায়। এবার আগের ট্যাগগুলো একসঙ্গে ব্যবহার করে এমন চিত্রের কোড লিখব। ঘর ও তার ভেতরের উপাদানের শুরু--শেষ সঠিকভাবে মিলিয়ে রাখাই এখানে প্রধান কাজ।

\subsection{সারণির ঘরে রাখা যায় যেসব উপাদান}

HTML সারণির \texttt{<td>} বা \texttt{<th>} ঘরের ভেতরে অনেক ধরনের HTML উপাদান রাখা যায়।

\begin{center}
\footnotesize
\begin{tabular}{|p{0.30\textwidth}|p{0.52\textwidth}|}
\hline
\textbf{ঘরের ভেতরে রাখা যায়} & \textbf{উদাহরণ} \\
\hline
সাধারণ লেখা & \texttt{<td>ICT</td>} \\
\hline
বোল্ড/আন্ডারলাইন লেখা & \texttt{<td><u>ICT</u></td>} \\
\hline
হাইপারলিংক & \texttt{<td><a href="board.html">Board</a></td>} \\
\hline
ছবি & \texttt{<td><img src="book.jpg" alt="Book" width="100"></td>} \\
\hline
ক্রমিক তালিকা & \texttt{<td><ol><li>Book</li></ol></td>} \\
\hline
অক্রমিক তালিকা & \texttt{<td><ul><li>Pen</li></ul></td>} \\
\hline
নেস্টেড সারণি & \texttt{<td><table>...</table></td>} \\
\hline
\end{tabular}
\end{center}

\begin{keypointbox}{মূল ধারণা}
সারণির ঘরের ভেতরে বিষয়বস্তু রাখার সময় বিষয়বস্তু অবশ্যই \texttt{<td>} বা \texttt{<th>} ট্যাগের ভেতরে রাখতে হবে। \texttt{<table>} ট্যাগের ভেতরে সরাসরি \texttt{<ol>}, \texttt{<ul>}, \texttt{<img>} বা \texttt{<a>} লেখা ঠিক নয়।
\end{keypointbox}

\subsection{সারণির ঘরে তালিকা}

কোনো সারণির ঘরের ভেতরে ক্রমিক তালিকা বা অক্রমিক তালিকা বসাতে হলে \texttt{<td>} ট্যাগের ভেতরে \texttt{<ol>} বা \texttt{<ul>} লিখতে হয়।

\subsubsection{লেখার সাধারণ রূপ}

\begin{examplebox}{কোড}
\begin{verbatim}
<td>
  <ol>
    <li>Item 1</li>
    <li>Item 2</li>
  </ol>
</td>
\end{verbatim}
\end{examplebox}

\Needspace{10\baselineskip}
অথবা--

\begin{examplebox}{কোড}
\begin{verbatim}
<td>
  <ul>
    <li>Item 1</li>
    <li>Item 2</li>
  </ul>
</td>
\end{verbatim}
\end{examplebox}

\subsection{উদাহরণ: ক্রমিক ও অক্রমিক তালিকা}

\begin{examplebox}{ফলের ধারণা}
\begin{center}
\begin{tabular}{|p{0.36\textwidth}|p{0.36\textwidth}|}
\hline
\multicolumn{2}{|c|}{\textbf{A \& H Company Ltd.}} \\
\hline
\textbf{Order List} & \textbf{Unorder List} \\
\hline
i. Book \newline ii. Paper \newline iii. Note Book & \rule{.6ex}{.6ex} Marker \newline \rule{.6ex}{.6ex} Ink \newline \rule{.6ex}{.6ex} Pencil \\
\hline
\end{tabular}
\end{center}
\end{examplebox}

\Needspace{6\baselineskip}
\par\noindent{\color{chFourTeal}\bfseries\small কোড: বাম কলাম শেষে ডান কলাম পড়বে}\par
\begin{multicols}{2}
\begin{verbatim}
<!DOCTYPE html>
<html>
<head>
  <title>HTML List in Table</title>
</head>
<body>
  <table border="2" cellspacing="0" cellpadding="5">
    <tr>
      <th colspan="2" align="center">A & H Company Ltd.</th>
    </tr>
    <tr>
      <th>Order List</th><th>Unorder List</th>
    </tr>
    <tr>
      <td>
        <ol type="i">
          <li>Book</li>
          <li>Paper</li>
          <li>Note Book</li>
        </ol>
      </td>
      <td>
        <ul type="square">
          <li>Marker</li>
          <li>Ink</li>
          <li>Pencil</li>
        </ul>
      </td>
    </tr>
  </table>
</body>
</html>
\end{verbatim}
\end{multicols}


\begin{boardbox}{পরীক্ষার জন্য মনে রাখবে}
ওপরের উদাহরণের Order List ও Unorder List দুটি দৃশ্যমান শিরোনাম। তালিকার সঠিক পরিভাষা Ordered List ও Unordered List। দুই ধরনের তালিকাই ঘরের ভেতরে শুরু ও শেষ হয়েছে।
\end{boardbox}

\subsection{সারণির ঘরে হাইপারলিংক}

সারণির ঘরের ভেতরে লিংক বসাতে \texttt{<td>} ট্যাগের ভেতরে \texttt{<a href="">...</a>} লিখতে হয়।

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.54\linewidth}\vspace{0pt}\small
\subsubsection{লেখার রূপ}

\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<td>
  <a href="url">Link Text</a>
</td>
\end{verbatim}
\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.42\linewidth}\vspace{0pt}\small
\subsubsection{উদাহরণ}

\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<td>
  <a href="https://www.educationboard.gov.bd">Board</a>
</td>
\end{verbatim}
\par\smallskip


\texttt{target="\_blank"} দিলে নতুন ব্রাউজিং প্রসঙ্গে লিংক খোলার অনুরোধ করা হয়। ব্রাউজার ও ব্যবহারকারীর সেটিং অনুযায়ী এটি নতুন ট্যাব বা উইন্ডো হতে পারে। বাহ্যিক লিংকে এটি দেওয়া বাধ্যতামূলক নয়; অ্যাট্রিবিউট না দিলে সাধারণত বর্তমান ট্যাবেই পেজ খোলে।
\end{minipage}\par
\subsection{উদাহরণ: বিষয়ের পাশে লিংক}

\begin{examplebox}{ফলের ধারণা}
\begin{center}
\begin{tabular}{|c|c|}
\hline
\textbf{Subject} & \textbf{Board} \\
\hline
ICT & \underline{Board} \\
\hline
\end{tabular}
\end{center}
Board লেখাটি লিংক; এতে ক্লিক করলে \texttt{href}-এ দেওয়া ঠিকানায় যাওয়া যায়।
\end{examplebox}

\begin{examplebox}{কোড}
\begin{verbatim}
<table border="1" cellspacing="0" cellpadding="5">
  <tr>
    <th>Subject</th>
    <th>Board</th>
  </tr>
  <tr>
    <td>ICT</td>
    <td>
      <a href="https://www.educationboard.gov.bd" target="_blank">
        Board
      </a>
    </td>
  </tr>
</table>
\end{verbatim}
\end{examplebox}

\subsection{সারণির ঘরে ছবি}

সারণির ঘরের ভেতরে ছবি বসাতে \texttt{<td>} ট্যাগের ভেতরে \texttt{<img>} ট্যাগ লিখতে হয়।

\subsubsection{লেখার রূপ}

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.54\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<td>
  <img src="image.jpg" alt="Image" width="100" height="80">
</td>
\end{verbatim}
\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.42\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}ব্যাখ্যা}\par
এই উদাহরণে \texttt{src} ছবির ঠিকানা এবং \texttt{alt} বিকল্প লেখা জানায়। \texttt{width} ও \texttt{height} ছবির প্রদর্শিত প্রস্থ ও উচ্চতা নির্ধারণ করে।
\end{minipage}\par
\subsection{উদাহরণ: বইয়ের ছবি}

\begin{examplebox}{ফলের ধারণা}
\begin{center}
\begin{tabular}{|c|c|}
\hline
\textbf{Book} & \textbf{Image} \\
\hline
ICT Book & \fbox{\parbox[c][15mm][c]{24mm}{\centering ছবির স্থান\\\texttt{book.jpg}}} \\
\hline
\end{tabular}
\end{center}
\end{examplebox}

\begin{examplebox}{কোড}
\begin{verbatim}
<table border="1" cellspacing="0" cellpadding="5">
  <tr>
    <th>Book</th>
    <th>Image</th>
  </tr>
  <tr>
    <td>ICT Book</td>
    <td>
      <img src="book.jpg" alt="ICT Book" width="100" height="80">
    </td>
  </tr>
</table>
\end{verbatim}
\end{examplebox}

এখানে ছবির জায়গাটি বোঝানো হয়েছে। একই ফোল্ডারে \texttt{book.jpg} থাকলে ব্রাউজারে সেই ছবি দেখা যাবে; ফাইলের নাম দেখানোর নির্দেশ এটি নয়।

\subsection{সারণির ঘরে ছবির হাইপারলিংক}

সারণির ঘরের ছবিতে ক্লিক করে কোনো পেজ বা ওয়েবসাইটে যেতে চাইলে \texttt{<a>} ট্যাগের ভেতরে \texttt{<img>} ট্যাগ বসাতে হবে।

\subsubsection{লেখার রূপ}

\begin{examplebox}{কোড}
\begin{verbatim}
<td>
  <a href="url">
    <img src="image.jpg" alt="Image">
  </a>
</td>
\end{verbatim}
\end{examplebox}

\subsection{উদাহরণ: বইয়ের ছবিতে ক্লিক করে গন্তব্যে যাওয়া}

\begin{examplebox}{কোড}
\begin{verbatim}
<table border="1" cellspacing="0" cellpadding="5">
  <tr>
    <th>Book</th>
    <th>Image Link</th>
  </tr>
  <tr>
    <td>ICT Book</td>
    <td>
      <a href="https://www.book.com" target="_blank">
        <img src="book.jpg" alt="ICT Book" width="100" height="80">
      </a>
    </td>
  </tr>
</table>
\end{verbatim}
\end{examplebox}

\begin{keypointbox}{মনে রাখবে}
ক্লিক করা যায় এমন লেখা তৈরি করতে \texttt{<a>} ট্যাগের ভেতরে লেখা লিখতে হয়। ক্লিক করা যায় এমন ছবি তৈরি করতে \texttt{<a>} ট্যাগের ভেতরে \texttt{<img>} ট্যাগ লিখতে হয়।
\end{keypointbox}

\subsection{তালিকার ভেতরে লিংক: ঘরেও প্রয়োগ করা যায়}

তালিকার উপাদানের ভেতরেও লিংক বসানো যায়। নিচের স্বতন্ত্র তালিকায় Fruits একটি লিংক এবং তার অধীনে Ripe ও Dried উপতালিকা আছে। এখানে \texttt{start="4"} দেওয়ায় উপতালিকার রোমান ক্রম iv ও v থেকে শুরু হয়। এই উদাহরণে সারণি নেই; সারণির ঘরে চাইলে পুরো \texttt{<ul>...</ul>} অংশটি \texttt{<td>...</td>}-এর ভেতরে বসাবে।

\begin{examplebox}{কোড}
\begin{verbatim}
<ul type="circle">
  <li>Vegetables</li>
  <li>
    <a href="https://www.fruitbox.com" target="_blank">Fruits</a>
    <ol type="i" start="4">
      <li>Ripe</li>
      <li>Dried</li>
    </ol>
  </li>
</ul>
\end{verbatim}
\end{examplebox}

\subsection{নেস্টেড সারণির প্রয়োগ}

একটি সারণির ঘরের ভেতরে থাকা আরেকটি সারণিকে নেস্টেড সারণি (Nested Table) বলে। নিচে একজন শিক্ষার্থীর Marks ঘরে বিষয় ও নম্বরের একটি উপসারণি রাখা হয়েছে।

নেস্টেড সারণিতে বাইরের ঘর বন্ধ করার আগে ভেতরের পুরো সারণি বন্ধ করতে হয়। অপ্রয়োজনে বহু স্তরের সারণি করলে কোড পড়া কঠিন হয়। ছোট পর্দায় বিন্যাস ও সহায়ক প্রযুক্তিতে তথ্য বোঝার সমস্যাও হতে পারে। তথ্যের উপসারণি প্রয়োজন হলেই নেস্টেড সারণি ব্যবহার করা উচিত।

\subsection{নেস্টেড সারণি লেখার রূপ}

\begin{examplebox}{কোড}
\begin{verbatim}
<table border="1">
  <tr>
    <td>
      Outer Table Cell
    </td>
    <td>
      <table border="1">
        <tr>
          <td>Inner Table Cell</td>
        </tr>
      </table>
    </td>
  </tr>
</table>
\end{verbatim}
\end{examplebox}

\subsection{নেস্টেড সারণিতে যে নিয়মগুলো দেখবে}

\begin{mistakebox}{সতর্কতা}
\begin{itemize}
\item অনেক স্তরের নেস্টেড সারণির কোড পড়া কঠিন হয়।
\item ভুল সমাপ্তি ট্যাগ দিলে বিন্যাস নষ্ট হয়।
\item সহায়ক প্রযুক্তিতে ব্যবহারের সমস্যা হতে পারে।
\item ছোট পর্দায় বিন্যাসে সমস্যা হতে পারে।
\item শুধুমাত্র সারণিভিত্তিক তথ্যের ভেতরে উপসারণি দরকার হলে ব্যবহার করা উচিত।
\end{itemize}
\end{mistakebox}

\begin{boardbox}{ব্যবহারের ক্ষেত্র}
নেস্টেড সারণি দিয়ে সারণির ভেতরে তথ্যের উপসারণি দেখানো যায়। পুরো ওয়েবপেজের বিন্যাস তৈরির জন্য আধুনিক ওয়েব ডিজাইনে CSS ব্যবহার করবে।
\end{boardbox}

\subsection{প্রয়োগ: ঘরের লেখা, লিংক, ছবি ও তালিকা}

টেবিলের ঘরের ভিতরে লেখা ছাড়াও অন্য HTML এলিমেন্ট রাখা যায়। এখানে ছবি, তালিকা, ফর্মের নিয়ন্ত্রণ, আড়াআড়ি রেখা, হাইপারলিংক এবং আরেকটি টেবিল রাখার উদাহরণ আছে। ফলে ফলাফলের সঙ্গে শিক্ষার্থীর ছবি, পণ্যের তালিকার সঙ্গে কেনার লিংক এবং কোর্স পরিকল্পনার সঙ্গে ক্রমযুক্ত তালিকা দেখানো যায়। বিস্তারিত তথ্যের জন্য কোনো ঘরে আরেকটি টেবিলও রাখা যায়।

\begin{keypointbox}{ঘরের ভিতরে বিষয়বস্তু রাখার নিয়ম}

\begin{itemize}
\item \texttt{<td>} ট্যাগের বাইরে তালিকা লিখে ফেলা।
\item \texttt{<td>} শেষ করার আগে \texttt{<ol>} বা \texttt{<ul>} ঠিকভাবে শেষ না করা।
\item ছবির লিংক করতে \texttt{<a>} ট্যাগের ভেতরে \texttt{<img>} না রাখা।
\item \texttt{href} ও \texttt{src} অ্যাট্রিবিউট গুলিয়ে ফেলা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{<table>} ট্যাগের ভেতরে সরাসরি \texttt{<ol>} লিখে ফেলা; ঘর দরকার।
\item নেস্টেড সারণির সমাপ্তি ট্যাগ ভুল করা।
\item সারণির ঘরের ভেতরে বিষয়বস্তু বেশি দিলে বিন্যাস অগোছালো করা।
\item সব বাহ্যিক লিংকে \texttt{target="\_blank"} বাধ্যতামূলক ভাবা; এটি নতুন ট্যাব/উইন্ডোতে খোলার প্রয়োজন অনুযায়ী দেওয়া হয়।
\end{itemize}
\end{minipage}\par

\end{mistakebox}

পরের অংশে \texttt{<div>}, \texttt{<span>} ও \texttt{style} অ্যাট্রিবিউট দিয়ে বিষয়বস্তু সাজানো এবং ইনলাইন CSS-এর মৌলিক প্রয়োগ দেখব।
\end{boardbox}
