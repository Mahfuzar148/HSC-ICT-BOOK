\section{বিভাগ, ইনলাইন অংশ ও রীতি: div, span ও style}

% Composition pass 2026-10-04
পুরো বিজ্ঞপ্তির পটভূমি বদলাতে হবে, আবার একটি বাক্যের শুধু শেষ তারিখ আলাদা রঙে দেখাতে হবে।

\chFourMotivation{বড় অংশ ও ছোট লেখাংশ আলাদা করা $\rightarrow$ div ও span $\rightarrow$ নির্বাচিত ধারকে CSS প্রয়োগ $\rightarrow$ কলেজের বিজ্ঞপ্তি সুস্পষ্ট হয়।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item \texttt{<div>} ট্যাগের কাজ ও ব্যবহার ব্যাখ্যা করতে।
\item \texttt{<span>} ট্যাগের কাজ ও ব্যবহার ব্যাখ্যা করতে।
\item \texttt{<div>} ও \texttt{<span>}-এর পার্থক্য লিখতে।
\item \texttt{style} অ্যাট্রিবিউট ব্যবহার করে ইনলাইন CSS লিখতে।
\item \texttt{color}, \texttt{background-color}, \texttt{font-size}, \texttt{text-align}, \texttt{margin}, \texttt{padding}, \texttt{border} ইত্যাদি মৌলিক CSS প্রোপার্টি ব্যবহার করতে।
\end{itemize}
\end{learningbox}



আগের পাঠগুলোতে আমরা HTML দিয়ে শিরোনাম, অনুচ্ছেদ, তালিকা, হাইপারলিংক, ছবি ও সারণি তৈরি করেছি। এখন ওয়েবপেজের বিভিন্ন অংশ একসঙ্গে রাখা এবং নির্দিষ্ট অংশে রীতি প্রয়োগ করা শিখব। এই পাঠে তিনটি বিষয় সবচেয়ে গুরুত্বপূর্ণ: \texttt{<div>}, \texttt{<span>} এবং \texttt{style} অ্যাট্রিবিউট।

\subsection{\texttt{<div>} ট্যাগ কী?}

\begin{definitionbox}{বিভাগের ধারক (div Element)}
একাধিক HTML এলিমেন্টকে একটি সাধারণ ধারকে দলবদ্ধ করতে \texttt{<div>} ট্যাগ ব্যবহার করা হয়।
\end{definitionbox}

\texttt{<div>}-এ পেজের একটি বড় অংশের শিরোনাম, অনুচ্ছেদ ও অন্যান্য এলিমেন্ট রাখা যায়। ধারকটি নিজে বিশেষ নকশা তৈরি করে না। CSS প্রয়োগ করে পুরো অংশের পটভূমি, সীমানা ও অবস্থান একসঙ্গে নির্ধারণ করা যায়।

\end{minipage}\par
\subsection{\texttt{<div>} ট্যাগের ব্যবহার}

\texttt{<div>} সাধারণত ওয়েবপেজের বড় অংশ ভাগ করতে ব্যবহৃত হয়।

\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item হেডার অংশ তৈরি করতে
\item চলাচল/মেনু এলাকা একসঙ্গে রাখতে
\item প্রধান বিষয়বস্তু অংশ তৈরি করতে
\item পাশের অংশ তৈরি করতে
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item ফুটার অংশ তৈরি করতে
\item একাধিক এলিমেন্ট একসঙ্গে রীতি প্রয়োগ করতে
\item ওয়েবপেজ বিন্যাস তৈরি করতে
\end{itemize}
\end{minipage}\par

\subsection{\texttt{<span>} ট্যাগ কী?}

\begin{definitionbox}{ইনলাইন অংশের ধারক (span Element)}
লেখা বা ইনলাইন বিষয়বস্তুর নির্দিষ্ট অংশ দলবদ্ধ করতে \texttt{<span>} ট্যাগ ব্যবহার করা হয়। ওই অংশে আলাদা CSS বা স্ক্রিপ্ট প্রয়োগ করা যায়।
\end{definitionbox}

\texttt{<span>} সাধারণত নতুন লাইন শুরু করে না। তাই অনুচ্ছেদের একটি শব্দ বা বাক্যাংশ আলাদা রঙে দেখাতে এটি সুবিধাজনক। ব্লক ও ইনলাইন এখানে স্বাভাবিক প্রদর্শনের ধরন বোঝায়। CSS-এর \texttt{display} দিয়ে প্রদর্শন বদলালেও কোন এলিমেন্টের ভেতরে কী রাখা বৈধ, সেই HTML নিয়ম বদলায় না।

\subsection{\texttt{<div>} ও \texttt{<span>}-এর পার্থক্য}

\begin{center}
\begin{tabular}{|p{0.24\textwidth}|p{0.30\textwidth}|p{0.30\textwidth}|}
\hline
\textbf{বিষয়} & \textbf{\texttt{<div>}} & \textbf{\texttt{<span>}} \\
\hline

কাজ & বড় অংশ বা ব্লক একসঙ্গে রাখে & লেখা/ইনলাইন অংশ একসঙ্গে রাখে \\

\hline

\hline
উদাহরণ & হেডার, ফুটার, বিজ্ঞপ্তির ঘর & লাল শব্দ, বিশেষভাবে দেখানো লেখা \\

শেষের ট্যাগ & আছে & আছে \\
\hline

\texttt{div} = বিভাগ = বড় ভাগ বা অংশ। \texttt{span} = ছোট ইনলাইন অংশ।
\end{keypointbox}

\subsection{style অ্যাট্রিবিউট কী?}

\begin{definitionbox}{রীতির অ্যাট্রিবিউট (style Attribute)}
HTML এলিমেন্টে সরাসরি CSS ঘোষণা প্রয়োগের জন্য ব্যবহৃত অ্যাট্রিবিউটকে \texttt{style} অ্যাট্রিবিউট বলে। এই অ্যাট্রিবিউটের মান হিসেবে CSS বৈশিষ্ট্য ও তার মান লেখা হয়; যেমন \texttt{style="color:red;"}।
\end{definitionbox}

\texttt{style} হলো HTML-এর একটি সাধারণভাবে প্রযোজ্য অ্যাট্রিবিউট। এর মাধ্যমে কোনো এলিমেন্টে সরাসরি CSS রীতি দেওয়া যায়।

\subsubsection{লেখার সাধারণ রূপ}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<tag style="property:value;">Content</tag>
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}

\end{minipage}\par
\end{examplebox}

\subsection{style অ্যাট্রিবিউটের সঠিক নিয়ম}

\texttt{style} অ্যাট্রিবিউটের ভিতরে CSS প্রোপার্টি ও মান লিখতে হয়।

\subsubsection{সঠিক কাঠামো}

\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
style="property:value; property:value;"
\end{verbatim}
\end{minipage}\par

\subsubsection{উদাহরণ}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<div style="color:red; background-color:yellow;">
  HSC ICT
</div>
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\begin{boardbox}{গুরুত্বপূর্ণ নিয়ম}
\begin{itemize}
\item প্রোপার্টি ও মানের মাঝে কোলন \texttt{:} দিতে হয়।
\item একাধিক ঘোষণার মাঝে সেমিকোলন \texttt{;} দিতে হয়।
\item অ্যাট্রিবিউট মান উদ্ধৃতি চিহ্নের মধ্যে লেখা ভালো।
\item CSS বৈশিষ্ট্য ও তার মান লেখার সময় \texttt{=} ব্যবহার করা যাবে না।
\item কমা দিয়ে CSS ঘোষণা আলাদা করা যাবে না।
\end{itemize}
\end{boardbox}

\subsection{ইনলাইন CSS কী?}

\begin{definitionbox}{ইনলাইন CSS}
কোনো HTML এলিমেন্টের শুরুর ট্যাগের মধ্যে \texttt{style} অ্যাট্রিবিউট ব্যবহার করে সরাসরি CSS বৈশিষ্ট্য ও তার মান লেখা হলে তাকে ইনলাইন CSS বলে।
\end{definitionbox}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<h1 style="color:blue;">HSC ICT</h1>
<p style="font-size:20px;">Web Design and HTML</p>
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\subsection{গুরুত্বপূর্ণ CSS প্রোপার্টি}

এই পাঠে style অ্যাট্রিবিউট শেখার জন্য নিচের প্রোপার্টিগুলো বেশি গুরুত্বপূর্ণ।

\begin{center}
\footnotesize
\begin{tabular}{|p{0.24\textwidth}|p{0.34\textwidth}|p{0.28\textwidth}|}
\hline
\textbf{প্রোপার্টি} & \textbf{কাজ} & \textbf{উদাহরণ} \\
\hline
\texttt{color} & লেখার রং নির্ধারণ করে & \texttt{color:red;} \\
\hline
\texttt{background-color} & পটভূমির রং নির্ধারণ করে & {\scriptsize\ttfamily background-color:yellow;} \\
\hline
\texttt{font-size} & অক্ষরের আকার নির্ধারণ করে & \texttt{font-size:20px;} \\
\hline
\texttt{font-family} & ফন্টের পরিবার নির্ধারণ করে & {\scriptsize\ttfamily font-family:Arial;} \\
\hline
\texttt{text-align} & লেখা সাজানোর অবস্থান নির্ধারণ করে & {\scriptsize\ttfamily text-align:center;} \\
\hline
\texttt{width} & এলিমেন্টের প্রস্থ নির্ধারণ করে & \texttt{width:200px;} \\
\hline
\texttt{height} & এলিমেন্টের উচ্চতা নির্ধারণ করে & \texttt{height:100px;} \\
\hline
\texttt{margin} & এলিমেন্টের বাইরের ফাঁকা জায়গা & \texttt{margin:10px;} \\
\hline
\texttt{padding} & এলিমেন্টের ভিতরের ফাঁকা জায়গা & \texttt{padding:10px;} \\
\hline

\subsection{সাধারণ ভুল}

\begin{multicols}{2}
\begin{itemize}
\item নির্দিষ্ট শব্দ রীতি প্রয়োগ করতে \texttt{<div>} ব্যবহার করা।
\item পুরো অংশ একসঙ্গে রাখতে \texttt{<span>} ব্যবহার করা।
\item \texttt{style="color=red"} লেখা; সঠিক \texttt{style="color:red;"}।
\item CSS ঘোষণার মাঝে কমা ব্যবহার করা।
\item \texttt{background color} লেখা; সঠিক \texttt{background-color}।
\item একাধিক CSS ঘোষণার মাঝের সেমিকোলন \texttt{;} না দেওয়া। শেষ ঘোষণার পরে এটি ঐচ্ছিক, তবে নিয়মিত দিলে নতুন ঘোষণা যোগ করার সময় ভুল কমে।
\item \texttt{<div>} ও \texttt{<span>} শেষের ট্যাগ না দেওয়া।
\item ইনলাইন CSS বেশি ব্যবহার করে কোড অগোছালো করা।
\end{itemize}
