\section{ওয়েবসাইট প্রকাশ ও হোস্টিং (Website Publishing and Hosting)}

% Composition pass 2026-10-04
নিজের কম্পিউটারে তৈরি সাইটটি অন্য শিক্ষার্থীদের যন্ত্র থেকে দেখতে হবে।

\chFourMotivation{সবার কাছে URL দিয়ে পৌঁছানো $\rightarrow$ হোস্টিং ও প্রকাশনা $\rightarrow$ ফাইল সার্ভারে রাখা, প্রয়োজনমতো DNS ও HTTPS নির্ধারণ $\rightarrow$ দূর থেকে কলেজের সাইট দেখা।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item ডোমেইন, হোস্টিং, URL, DNS ও ওয়েব সার্ভারের সম্পর্ক ব্যাখ্যা করতে।
\item ওয়েবসাইট প্রকাশ কী তা সংজ্ঞাসহ লিখতে।
\item ওয়েবসাইট প্রকাশের ধাপ লিখতে।
\item FTP (File Transfer Protocol), SFTP (SSH File Transfer Protocol), নিয়ন্ত্রণ প্যানেল ও স্ট্যাটিক হোস্টিংয়ের ধারণা বুঝতে।
\item ওয়েবসাইট প্রকাশের পরে সাধারণ সমস্যা শনাক্ত করতে।
\end{itemize}
\end{learningbox}

\subsection{ওয়েবসাইট প্রকাশ (Website Publishing)}

\begin{definitionbox}{}
ওয়েবসাইটের ফাইল ওয়েব সার্ভারে রেখে ইন্টারনেটে URL দিয়ে দেখার উপযোগী করার প্রক্রিয়াকে \textbf{ওয়েবসাইট প্রকাশ} বলে।
\end{definitionbox}

নিজের কম্পিউটারের \texttt{index.html} ফাইল খুললে সেই কম্পিউটার থেকে পেজ দেখা যায়। ইন্টারনেটে প্রকাশ করলে অন্য ব্যবহারকারীও URL দিয়ে পেজ দেখতে পারেন। হোস্টিং সেবার দেওয়া ঠিকানা ব্যবহার করা যায়; নিজের ডোমেইন যুক্ত করতে চাইলে ডোমেইন ও DNS-এর ব্যবস্থা করতে হয়।

\subsection{ডোমেইন ও হোস্টিং}

ওয়েবসাইট তৈরির পরে ইন্টারনেটে দেখানোর জন্য তার ফাইল সার্ভারে রাখতে হয়। ব্যবহারকারী ডোমেইন নামসহ URL দিয়ে সেই সার্ভার থেকে পেজ পান।

\textbf{আগের ধারণা মনে করি:} ডোমেইন নাম, যেমন \texttt{example.com}, প্রকাশিত সাইটের ঠিকানায় ব্যবহার করা যায়। হোস্টিং সেবার দেওয়া ঠিকানাতেও সাইট প্রকাশ করা যায়; নিজের ডোমেইন যুক্ত করা ঐচ্ছিক।

\begin{definitionbox}{ওয়েব হোস্টিং (Web Hosting)}
ইন্টারনেটে ওয়েবসাইটের ফাইল ও প্রয়োজনীয় সম্পদ রাখার এবং ব্যবহারকারীর অনুরোধে সরবরাহ করার সার্ভার-সেবাকে \textbf{ওয়েব হোস্টিং} বলে। এতে HTML, ছবি, CSS, JavaScript এবং প্রয়োজন অনুযায়ী ডেটাবেজ রাখার ব্যবস্থা থাকে।
\end{definitionbox}

\subsection{ডোমেইন, হোস্টিং, URL ও DNS-এর সম্পর্ক}

\begin{keypointbox}{সহজ ব্যাখ্যা}
\begin{itemize}
\item \textbf{ডোমেইন নাম} হলো ওয়েবসাইট চেনার নাম বা ঠিকানার অংশ।
\item \textbf{হোস্টিং} হলো ওয়েবসাইটের ফাইল সার্ভারে রেখে সরবরাহ করার সেবা।
\item \textbf{URL} হলো নির্দিষ্ট ওয়েবপেজ বা উপকরণের ঠিকানা।
\item \textbf{DNS} ডোমেইন নাম থেকে সংশ্লিষ্ট সার্ভারের ঠিকানা খুঁজতে সাহায্য করে।
\end{itemize}
\end{keypointbox}

\begin{examplebox}{উদাহরণ}
\begin{itemize}
\item Domain: \texttt{abccollege.edu.bd}
\item Hosting: যে সার্ভারে \texttt{index.html}, ছবি ও CSS ফাইল রাখা আছে
\item URL: \texttt{https://abccollege.edu.bd/notice.html}
\end{itemize}
\end{examplebox}

\subsection{ওয়েবসাইট দেখা না গেলে সম্ভাব্য কারণ}

ওয়েবসাইট প্রকাশের পরে কখনো URL লিখলেও পেজ দেখা যায় না। সম্ভাব্য কারণগুলো হলো:

\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item নিবন্ধিত ডোমেইন নামের মেয়াদ শেষ হওয়া।
\item হোস্টিং অ্যাকাউন্টের মেয়াদ শেষ হওয়া বা সাময়িকভাবে বন্ধ থাকা।
\item ওয়েব সার্ভার অচল থাকা।
\item DNS-এর সেটিং ভুল হওয়া বা পরিবর্তিত তথ্য এখনো কার্যকর না হওয়া।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item ডেটা আদান-প্রদানের নির্ধারিত সীমা শেষ হয়ে যাওয়া।
\item \texttt{index.html} ফাইল সার্ভারে না পাঠানো বা ভুল ফোল্ডারে রাখা।
\item HTML ফাইল, ছবির পথ বা প্রবেশের অনুমতিতে ভুল থাকা।
\item HTTPS সার্টিফিকেটে সমস্যা হওয়া।
\end{itemize}
\end{minipage}\par


\subsection{ওয়েবসাইটের মৌলিক নিরাপত্তা}

ওয়েবসাইট প্রকাশের পরে মৌলিক নিরাপত্তা বজায় রাখা জরুরি। HTTPS-এ এখন TLS (Transport Layer Security) ব্যবহৃত হয়; হোস্টিং প্যানেলে SSL (Secure Sockets Layer) কথাটি প্রচলিত পুরোনো নামে দেখা যেতে পারে। HTTPS সংযোগের তথ্য সুরক্ষিত করে; ওয়েবসাইটের সব তথ্য বিশ্বাসযোগ্য হওয়ার নিশ্চয়তা দেয় না।

\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item HTTPS-এর জন্য বৈধ সার্টিফিকেট ব্যবহার করা।
\item হোস্টিং নিয়ন্ত্রণ প্যানেলে শক্তিশালী পাসওয়ার্ড রাখা।
\item নিয়মিত ব্যাকআপ রাখা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item অপ্রয়োজনীয় বা ব্যক্তিগত ফাইল সবার জন্য উন্মুক্ত ফোল্ডারে না রাখা।
\item ফর্ম থাকলে ব্যবহারকারীর তথ্যের বৈধতা পরীক্ষা করা; সার্ভারেও এই পরীক্ষা করা।
\item হোস্টিং প্যানেল, CMS (Content Management System) ও ব্যবহৃত স্ক্রিপ্ট হালনাগাদ রাখা।
\end{itemize}
\end{minipage}\par


\subsection{সংক্ষিপ্ত প্রকাশের ধাপ}

নিচে ফাইল-ভিত্তিক হোস্টিংয়ে প্রকাশের একটি উদাহরণ দেওয়া হয়েছে। সেবাভেদে ফোল্ডারের নাম, ফাইল পাঠানোর পদ্ধতি ও DNS রেকর্ড আলাদা হতে পারে। DNS-এ কোনো সেবা সার্ভারের IP ঠিকানা, আবার কোনো সেবা নির্দিষ্ট হোস্টনেম চায়। SFTP-তে SSH (Secure Shell) সংযোগ তথ্য সুরক্ষিত রাখে। সাধারণ FTP নিজে এনক্রিপশন দেয় না।

\begin{examplebox}{FTP/SFTP দিয়ে প্রকাশের নমুনা ধাপ}
\begin{verbatim}
1. Test locally (open index.html in browser)
2. Connect to host via SFTP/FTP client
3. Upload files to public_html or www directory
4. Point domain DNS to hosting server IP
5. Open URL in browser and test
\end{verbatim}
\end{examplebox}


\subsection{অনুসন্ধানে সাইট খুঁজে পাওয়া ও SEO}
সার্চ ইঞ্জিন যেন সাইটের বিষয়বস্তু বুঝতে পারে এবং ব্যবহারকারী প্রাসঙ্গিক অনুসন্ধানে সাইটটি খুঁজে পান, সে জন্য বিষয়বস্তু ও গঠন উন্নত করার কাজকে SEO (Search Engine Optimization) বলে। অর্থপূর্ণ শিরোনাম, পরিষ্কার লেখা, উপযুক্ত লিংক ও সহজ নেভিগেশন এতে সাহায্য করে। সাইট প্রকাশের জন্য সার্চ ইঞ্জিনে আলাদা নিবন্ধন বাধ্যতামূলক নয়। প্রকাশিত পেজ সার্চ ইঞ্জিনের সূচিতে আসতে সময় লাগতে পারে; SEO করলেই প্রথম অবস্থানে আসার নিশ্চয়তা নেই।

\subsection{সমস্যা ও সমাধান}

\begin{practicebox}
\textbf{সমস্যা:} রাফি একটি কলেজের ওয়েবসাইট তৈরি করেছে। নিজের কম্পিউটারে পেজ সঠিকভাবে দেখা যায়। হোস্টিং সার্ভারে পাঠানোর পরে ছবি দেখা যাচ্ছে না এবং হোম পেজ নিজে থেকে খুলছে না। সম্ভাব্য কারণ ও সমাধান লেখ।

\textbf{সমাধান:}
\begin{itemize}
\item হোম পেজ নিজে থেকে না খুললে ফাইলের নাম \texttt{index.html} আছে কি না এবং তা সঠিক প্রকাশযোগ্য ফোল্ডারে রাখা হয়েছে কি না দেখতে হবে।
\item ছবি না দেখালে ছবির ফাইল সার্ভারে পাঠানো হয়েছে কি না দেখতে হবে।
\item ফাইলের পথে বড় ও ছোট অক্ষরের মিল দেখতে হবে; যেমন \texttt{Campus.jpg} ও \texttt{campus.jpg} ভিন্ন হতে পারে।
\item HTML ফাইল ও ছবি একই ফোল্ডারে না থাকলে ছবির ফোল্ডারসহ সঠিক পথ দিতে হবে।
\item ব্রাউজারের ক্যাশ মুছে আবার পরীক্ষা করা যেতে পারে।
\end{itemize}
\end{practicebox}



\begin{mistakebox}{সাধারণ ভুল}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item হোস্টিং সেবা যে প্রাথমিক ফাইলের নাম চায়, তা না রাখা; সাধারণ উদাহরণ \texttt{index.html}।
\item ছবি না পাঠিয়ে শুধু HTML ফাইল সার্ভারে পাঠানো।
\item নিজের কম্পিউটারের সম্পূর্ণ স্থানীয় পথ HTML কোডে ব্যবহার করা।
\item ডোমেইন কিনে হোস্টিং/DNS-এর ব্যবস্থা না করা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item DNS-এর পরিবর্তিত তথ্য কার্যকর হতে সময় লাগতে পারে—এটি বিবেচনা না করা।
\item HTTPS-এর ব্যবস্থা না করা।
\item প্রকাশের আগে লিংক, ছবি, ফর্ম ও মোবাইলে পেজ পরীক্ষা না করা।
\end{itemize}
\end{minipage}\par

\end{mistakebox}

\begin{practicebox}
\begin{enumerate}
\item ডোমেইন ও হোস্টিংয়ের পার্থক্য লেখ।
\item \texttt{index.html} ফাইল কেন গুরুত্বপূর্ণ?
\item ওয়েবসাইট প্রকাশের পাঁচটি ধাপ লেখ।
\item DNS-এর কাজ কী?
\item ওয়েবসাইট প্রকাশের আগে কোন কোন বিষয় পরীক্ষা করা উচিত?
\end{enumerate}
\end{practicebox}
