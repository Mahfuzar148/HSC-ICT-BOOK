\section{ব্যবহারিক অনুশীলন: ছোট ওয়েবসাইট প্রকল্প (Mini Project)}

% Composition pass 2026-10-04
একটি পেজের বদলে কলেজের পরিচিতি, নোটিশ, ফল ও যোগাযোগের আলাদা পেজ দরকার।

\chFourMotivation{সম্পূর্ণ ছোট সাইট তৈরি $\rightarrow$ মিনি প্রকল্প $\rightarrow$ ফোল্ডার, পেজ, লিংক ও সম্পদের পথ মিলিয়ে পরীক্ষা $\rightarrow$ পাঁচ পেজের কলেজ ওয়েবসাইট।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item HTML ব্যবহার করে একটি সহজ স্ট্যাটিক ওয়েবসাইটের পরিকল্পনা করতে।
\item Home, About, Notice, Result ও Contact পেজের কাঠামো বুঝতে।
\item শিরোনাম, অনুচ্ছেদ, ছবি, হাইপারলিংক, তালিকা, সারণি ও ফর্ম একসঙ্গে ব্যবহার করতে।
\item ফাইল ও ফোল্ডার সঠিকভাবে সাজিয়ে ওয়েবসাইটের প্রকল্প তৈরি করতে।
\item উচ্চমাধ্যমিকের ব্যবহারিক প্রশ্নের চাহিদা অনুযায়ী ওয়েবপেজ তৈরি করতে।
\end{itemize}
\end{learningbox}

এই ছোট প্রকল্পে ওয়েবপেজ ডিজাইন, HTML ট্যাগ, হাইপারলিংক, ছবি, সারণি, ফর্ম ও প্রকাশনার কাজ একসঙ্গে প্রয়োগ করা হবে। পরিচিত ট্যাগ দিয়ে পাঁচটি পেজ তৈরি করে তাদের পারস্পরিক সংযোগ পরীক্ষা করব।

\begin{keypointbox}{অধ্যায়ের ধারণা একসঙ্গে প্রয়োগ}
প্রকল্পে প্রতিটি ট্যাগের কাজ এবং পুরো পেজের সঙ্গে তার সম্পর্ক দেখা যায়। ফাইল তৈরি, লিংক যুক্ত করা ও ব্রাউজারে পরীক্ষা করার কাজ একই সঙ্গে অনুশীলন হয়।
\end{keypointbox}

\subsection{প্রকল্পের নাম}

\textbf{``আমার কলেজ'' ওয়েবসাইট তৈরি}। এটি একটি সহজ স্ট্যাটিক ওয়েবসাইটের প্রকল্প। HTML-এর মৌলিক ট্যাগ ব্যবহার করে কয়েকটি পেজ তৈরি করা হবে। উদাহরণের ABC College, শিক্ষার্থী ও যোগাযোগের তথ্য কাল্পনিক।

\subsection{প্রকল্পের উদ্দেশ্য}

এই প্রকল্প শেষে শিক্ষার্থীরা HTML দিয়ে একটি সহজ স্ট্যাটিক ওয়েবসাইট তৈরি করতে পারবে। এতে শিরোনাম, অনুচ্ছেদ, ছবি, হাইপারলিংক, তালিকা, সারণি ও ফর্ম থাকবে।

\subsection{প্রকল্পের ধরন}

\begin{itemize}
\item স্ট্যাটিক ওয়েবসাইট
\item নিজের কম্পিউটারে তৈরি ও পরীক্ষা করা ওয়েবপেজের প্রকল্প
\item উচ্চমাধ্যমিক ICT-এর চতুর্থ অধ্যায়ের ব্যবহারিক অনুশীলন
\end{itemize}

\item লেখা ফরম্যাটিং
\item তালিকা
\item হাইপারলিংক
\end{itemize}
\end{minipage}\hfill

\item HTML ফাইল সংরক্ষণ করা: \texttt{.html}

\subsection{ফোল্ডারের কাঠামো}

প্রকল্পের ফোল্ডার এভাবে সাজাবে:

\begin{verbatim}
college-website/
  index.html
  about.html
  notice.html
  result.html
  contact.html
  images/
    logo.png
    college.jpg
\end{verbatim}

\begin{keypointbox}{ছবির ফাইল ও লিংকের গন্তব্য}
\texttt{college.jpg} ও \texttt{logo.png} ছবির ফাইল \texttt{images} ফোল্ডারের ভেতরে রাখতে হবে। HTML ফাইল থেকে ছবি দেখাতে \texttt{src="images/college.jpg"} ধরনের পথ ব্যবহার করতে হবে। Notice পেজের \texttt{hsc-form-fillup.pdf}, \texttt{class-routine.pdf}, \texttt{ict-practical.pdf} ও \texttt{admission-result.pdf} লিংক চালাতে ঐ নামের PDF ফাইল প্রকল্পের মূল ফোল্ডারে রাখতে হবে।
\end{keypointbox}

\subsubsection{পেজ ৪: Result — \texttt{result.html}}

\texttt{result.html} পেজে শিক্ষার্থীর ফলের সারণি থাকবে।

\begin{center}
\begin{tabular}{|c|c|p{0.30\textwidth}|c|}

এই অংশে \texttt{<table>}, \texttt{<tr>}, \texttt{<th>}, \texttt{<td>}, \texttt{border} ও \texttt{<caption>} ব্যবহারের অনুশীলন হবে। এই তিন সারির প্রাথমিক নমুনাটি পরে পূর্ণ কোডে বিস্তারিত তথ্যসহ চার সারিতে দেখানো হয়েছে। GPA এখানে পৃথক নমুনা তথ্য; পরের কোডের ICT Marks থেকে GPA হিসাব করা হয়নি।

\subsubsection{পেজ ৫: Contact — \texttt{contact.html}}

\texttt{contact.html} পেজে একটি সহজ ফর্ম থাকবে:
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}

\item Email
\item Gender
\item Group
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item Message
\item Submit বোতাম
\item Reset বোতাম
\end{itemize}
\end{minipage}\par


এই অংশে \texttt{<form>}, \texttt{<input>}, \texttt{<textarea>}, \texttt{<select>}, \texttt{<option>} ও \texttt{<button>}-এর কাজ মিলিয়ে অনুশীলন হবে। নিচের পূর্ণ কোডে Submit ও Reset বোতাম \texttt{<input>} দিয়ে তৈরি করা হয়েছে; \texttt{<button>} দিয়েও বোতাম তৈরি করা যায়।

ফর্মটি তথ্য গ্রহণের নিয়ন্ত্রণ উপাদান দেখানোর নমুনা। এখানে \texttt{submit.php}-এর সার্ভার প্রোগ্রাম নেই। তাই শুধু HTML ফাইল খুলে Submit চাপলে তথ্য সংরক্ষিত হবে না বা কলেজের অফিসে পৌঁছাবে না। ওই কাজের জন্য সার্ভারের প্রোগ্রাম বা কার্যকর ফর্ম সেবা প্রয়োজন। Password ঘরটি ইনপুটের ধরন দেখায়; সাধারণ যোগাযোগের ফর্মে পাসওয়ার্ড চাওয়ার প্রয়োজন নেই।

উদাহরণে \texttt{<font>}, \texttt{<center>}, \texttt{bgcolor}, \texttt{align}, \texttt{cellpadding} ও \texttt{cellspacing}-এর মতো পুরোনো উপস্থাপনা রূপ আছে। ব্রাউজারে এগুলো দেখা যেতে পারে, কিন্তু আধুনিক HTML-এ নতুন কাজের জন্য CSS ব্যবহার করা হয়। সারণির তথ্য দেখানো ও সারণি দিয়ে পুরো পেজের বিন্যাস করা আলাদা বিষয়।

\subsection{কাজ করার ধাপ}

\begin{enumerate}
\item একটি ফোল্ডার তৈরি করো: \texttt{college-website}
\item \texttt{index.html} ফাইল তৈরি করো।
\item অন্য পেজ তৈরি করো: \texttt{about.html} ও \texttt{notice.html}
\item এরপর তৈরি করো: \texttt{result.html} ও \texttt{contact.html}
\item \texttt{images} ফোল্ডার তৈরি করে \texttt{logo.png} ও \texttt{college.jpg} রাখো।
\item সব পেজে নেভিগেশন লিংক দাও।
\item ব্রাউজারে \texttt{index.html} খুলে ফল দেখো।
\item লিংক, ছবি, সারণি ও ফর্মের নিয়ন্ত্রণগুলো কাজ করছে কি না পরীক্ষা করো।
\item ভুল থাকলে কোড সংশোধন করে ব্রাউজারে পেজ আবার লোড করো।
\end{enumerate}

\item \texttt{<ul>} কলেজের সুবিধার তালিকা এবং \texttt{<ol>} গুরুত্বপূর্ণ লিংকের তালিকা তৈরি করেছে।

\subsection{সাধারণ ভুল}

\begin{mistakebox}{প্রকল্প তৈরির সময় সাধারণ ভুল}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{index.html} ফাইলের নাম ভুল লেখা।
\item \texttt{images} ফোল্ডার না বানিয়ে ছবির পথ দেওয়া।
\item \texttt{src="images/college.jpg"} পথ ভুল লেখা।
\item নেভিগেশন লিংকে ফাইলের নাম ভুল লেখা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item উদাহরণের সব HTML পেজ একই ফোল্ডারে না রাখা।
\item যেসব HTML ট্যাগের সমাপ্তি ট্যাগ দরকার, সেগুলো সঠিকভাবে শেষ না করা।
\item ব্রাউজারে পেজ পুনরায় লোড না করে ফল যাচাই করা।
\item ছবির ফাইলের এক্সটেনশন \texttt{.jpg}, \texttt{.png} ভুল লেখা।
\end{itemize}
\end{minipage}\par

\end{mistakebox}

\subsection{সংক্ষিপ্ত অনুশীলন}

\begin{boardbox}{ছোট প্রকল্প কেন উপযোগী?}
নিজে ওয়েবপেজ তৈরি করলে HTML-এর ব্যবহার বোঝা যায়। ছোট প্রকল্পে শিক্ষার্থী শিরোনাম, অনুচ্ছেদ, ছবি, হাইপারলিংক, তালিকা, সারণি ও ফর্ম একসঙ্গে ব্যবহার করতে পারে। ফলে এই অধ্যায়ের গুরুত্বপূর্ণ ট্যাগ ও কাঠামো ব্যবহারিকভাবে অনুশীলন হয়।
\end{boardbox}
