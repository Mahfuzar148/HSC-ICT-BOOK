\section{সারণির ঘর একত্র করা: rowspan ও colspan}

% Composition pass 2026-10-04
ফলের সারণিতে একটি শিরোনাম তিন কলামের ওপর এবং একটি বিষয় দুই সারির পাশে দেখাতে হবে।

\chFourMotivation{একাধিক মূল ঘরের স্থানজুড়ে একই তথ্য $\rightarrow$ colspan/rowspan $\rightarrow$ বিস্তৃত ঘরের দখল করা স্থান গোনা $\rightarrow$ ফল ও রুটিনের মিলিত শিরোনাম।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item HTML সারণির ঘর একত্র করার ধারণা ব্যাখ্যা করতে।
\item \texttt{colspan} অ্যাট্রিবিউট ব্যবহার করে ঘরকে একাধিক কলামজুড়ে দেখাতে।
\item \texttt{rowspan} অ্যাট্রিবিউট ব্যবহার করে ঘরকে একাধিক সারিজুড়ে দেখাতে।
\item চিত্র দেখে \texttt{rowspan} ও \texttt{colspan} নির্ধারণ করতে।
\item একত্র করা ঘরের জন্য অতিরিক্ত \texttt{<td>} বা \texttt{<th>} না লেখার নিয়ম বুঝতে।
\end{itemize}
\end{learningbox}



একটি পরীক্ষার ফলের সারণিতে Roll, Name ও GPA তিনটি কলাম আছে। উপরে Result Sheet শিরোনামটি তিন কলামের মাঝ বরাবর দেখাতে আলাদা তিনটি ঘরে একই কথা লেখার প্রয়োজন নেই; একটি ঘরকে তিন কলামজুড়ে বিস্তৃত করলেই হয়। আবার একজনের দুটি ফোন নম্বর থাকলে Phone পরিচিতিটি দুই সারিজুড়ে একবার দেখানো যায়। এই অংশে \texttt{colspan} ও \texttt{rowspan} দিয়ে ঘরের বিস্তার এবং মূল ছকের স্থান গণনা শেখা হবে।

\subsection{ঘর একত্র করা (Cell Merge)}

\begin{definitionbox}{}
HTML সারণিতে একাধিক সারি বা কলামের স্থানজুড়ে একটি ঘর বিস্তৃত করে দেখানোকে \textbf{ঘর একত্র করা} বলে। এর জন্য \texttt{rowspan} ও \texttt{colspan} অ্যাট্রিবিউট ব্যবহার করা হয়।
\end{definitionbox}

ঘর একত্র করলে ছকের বিন্যাস বদলায়। আলাদা ঘরের লেখা স্বয়ংক্রিয়ভাবে একসঙ্গে যুক্ত হয় না। বিস্তৃত ঘরে কী লেখা থাকবে এবং ঘরটি কত সারি বা কলাম দখল করবে, তা কোডে নির্ধারণ করতে হয়।

\begin{examplebox}{সহজ উদাহরণ}
\begin{center}
\begin{tabular}{|c|c|c|}
\hline
\multicolumn{3}{|c|}{\textbf{Result Sheet}} \\
\hline
\textbf{Roll} & \textbf{Name} & \textbf{GPA} \\
\hline
\end{tabular}
\end{center}
এখানে \textbf{Result Sheet} ঘরটি ৩টি কলামজুড়ে আছে। তাই HTML কোডে \texttt{colspan="3"} ব্যবহার করতে হবে।
\end{examplebox}

\subsection{\texttt{colspan} দিয়ে কলামজুড়ে ঘর}

\texttt{colspan} একটি ঘর কতটি কলামজুড়ে থাকবে, তা নির্ধারণ করে। নিচের উদাহরণে Math ও ICT একই Subject শিরোনামের অধীনে আছে।

\texttt{colspan} নামটি Column Span অর্থাৎ কলামজুড়ে বিস্তার বোঝায়। এটি সাধারণত \texttt{<th>} বা \texttt{<td>} ট্যাগের ভেতরে লেখা হয়।

\end{boardbox}

\subsection{\texttt{rowspan} দিয়ে সারিজুড়ে ঘর}

\texttt{rowspan} একটি ঘর কতটি সারিজুড়ে থাকবে, তা নির্ধারণ করে। নিচের উদাহরণে Phone ঘরের পাশে দুই সারিতে দুটি নম্বর আছে।

\texttt{rowspan} নামটি Row Span অর্থাৎ সারিজুড়ে বিস্তার বোঝায়। এটি \texttt{<th>} অথবা \texttt{<td>} ট্যাগে ব্যবহার করা যায়।

\subsection{অ্যাট্রিবিউট দুটি কোথায় লেখা হয়?}

\texttt{rowspan} ও \texttt{colspan} সাধারণত \texttt{<td>} অথবা \texttt{<th>} ট্যাগের ভেতরে লেখা হয়।

\begin{examplebox}{কোড}
\begin{verbatim}
<th colspan="3">Result Sheet</th>
<td rowspan="2">ICT</td>
\end{verbatim}
\end{examplebox}

\begin{keypointbox}{মূল নিয়ম}
\texttt{colspan}-এর জন্য ধনাত্মক পূর্ণসংখ্যা দিতে হয়; যেমন \texttt{colspan="3"} মানে তিনটি কলাম। এখানকার \texttt{rowspan="2"} মানে দুইটি সারি। বিশেষ ক্ষেত্রে \texttt{rowspan="0"} বর্তমান সারি-দলের অবশিষ্ট সব সারি বোঝায়; সাধারণ চিত্রভিত্তিক অনুশীলনে যতটি সারি বিস্তৃত হয়েছে, সেই সংখ্যা লিখবে।
\end{keypointbox}

\subsection{একই সারণিতে rowspan ও colspan}

একটি জটিল সারণিতে \texttt{rowspan} ও \texttt{colspan}--দুটিই ব্যবহার হতে পারে।

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.54\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<table border="1" cellspacing="0" cellpadding="5">
  <tr>
    <th colspan="5">Marks Sheet</th>
  </tr>
  <tr>
    <th rowspan="2">Roll No.</th><th colspan="2">Bangla</th>
    <th colspan="2">English</th>
  </tr>
  <tr>
    <th>1st</th><th>2nd</th><th>1st</th><th>2nd</th>
  </tr>
  <tr>
    <td>001</td><td>85</td><td>67</td><td>58</td><td>70</td>
  </tr>
  <tr>
    <td>002</td><td>65</td><td>75</td><td>60</td><td>55</td>
  </tr>
</table>
\end{verbatim}
\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.42\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}ফলের ধারণা}\par
\begin{center}
\begin{tabular}{|c|c|c|c|c|}
\hline
\multicolumn{5}{|c|}{\textbf{Marks Sheet}} \\
\hline
\raisebox{-.5\normalbaselineskip}[0pt][0pt]{\textbf{Roll No.}} & \multicolumn{2}{c|}{\textbf{Bangla}} & \multicolumn{2}{c|}{\textbf{English}} \\
\cline{2-5}
 & \textbf{1st} & \textbf{2nd} & \textbf{1st} & \textbf{2nd} \\
\hline
001 & 85 & 67 & 58 & 70 \\
\hline
002 & 65 & 75 & 60 & 55 \\
\hline
\end{tabular}
\end{center}
\textbf{ব্যাখ্যা:} Marks Sheet ৫টি কলামজুড়ে আছে, Roll No. দুইটি সারিজুড়ে আছে, Bangla দুইটি কলামজুড়ে আছে এবং English দুইটি কলামজুড়ে আছে।

চিত্রে \texttt{colspan} দিয়ে আড়াআড়ি বিস্তার এবং \texttt{rowspan} দিয়ে লম্বালম্বি বিস্তার দেখানো হয়েছে। চিত্র দেখে কোড লেখার সময় বিস্তৃত ঘর আগে শনাক্ত করতে হয়।
\end{keypointbox}


\begin{keypointbox}{মনে রাখার উপায়}
\texttt{colspan} = কলাম জুড়ে পাশের দিকে বিস্তার।\\
\texttt{rowspan} = সারি জুড়ে নিচের দিকে বিস্তার।
\end{keypointbox}
\end{minipage}\par\smallskip

\subsubsection{মূল ছকের স্থান গণনা}

বিস্তৃত ঘর হলো এমন ঘর, যা একাধিক সারি বা কলামের জায়গা নেয়। টেবিল চিত্র দেখে কোড লেখার সময় বিস্তৃত ঘর ভুল হলে পুরো টেবিলের বিন্যাস নষ্ট হয়ে যায়। তাই \texttt{rowspan} ও \texttt{colspan} ব্যবহার করার আগে মূল কলামসংখ্যা ঠিক করা দরকার।

\begin{boardbox}{\texttt{colspan} কখন ব্যবহার করবে}
যদি কোনো ঘর বাম থেকে ডানে একাধিক কলামের জায়গা নেয়, তখন \texttt{colspan} ব্যবহার করবে। যেমন, একটি শিরোনাম যদি টেবিলের তিনটি কলাম জুড়ে থাকে, তাহলে কোড হবে \texttt{<th colspan="3">Title</th>}।
\end{boardbox}

\begin{boardbox}{\texttt{rowspan} কখন ব্যবহার করবে}
যদি কোনো ঘর উপর থেকে নিচে একাধিক সারির জায়গা নেয়, তখন \texttt{rowspan} ব্যবহার করবে। যেমন, একটি বিভাগের নাম যদি দুইটি শাখার সারির জন্য একই হয়, তাহলে কোড হবে \texttt{<td rowspan="2">Science</td>}।
\end{boardbox}

\begin{keypointbox}{বিস্তৃত ঘরের কারণে ঘর গণনা কমে যায়}
\texttt{colspan} দেওয়া ঘর একই সারিতে একাধিক কলামের স্থান নেয়। \texttt{rowspan} দেওয়া ঘর পরের সারির স্থানও দখল করে; সেই পরের সারিতে ওই স্থানের জন্য নতুন ঘর লিখতে হয় না। তাই প্রতিটি সারির দৃশ্যমান ঘর গুনলেই মূল কলামসংখ্যা পাওয়া যায় না; বিস্তৃত ঘরের দখল করা স্থানও ধরতে হবে।
\end{keypointbox}


\subsection{চিত্র থেকে rowspan ও colspan নির্ধারণ}

সারণির চিত্র দেখে কোড লেখার আগে মূল কলাম ও মূল সারিগুলো শনাক্ত করো। বিস্তৃত ঘরটি মূল ছকের কতটি স্থান দখল করেছে, তা গুনতে হয়। শুধু দৃশ্যমান ঘরের সংখ্যা থেকে মূল সারি বা কলামসংখ্যা পাওয়া যায় না। ধাপগুলো হলো:

\begin{enumerate}
\item মোট কয়টি কলাম আছে গুনে নাও।
\item কোনো ঘর ডানদিকে কয়টি কলামজুড়ে আছে দেখো।
\item ঘর ডানদিকে ২টি কলাম জুড়লে \texttt{colspan="2"}।
\item ঘর ডানদিকে ৩টি কলাম জুড়লে \texttt{colspan="3"}।
\item কোনো ঘর নিচের দিকে কয়টি সারিজুড়ে আছে দেখো।
\item ঘর নিচের দিকে ২টি সারি জুড়লে \texttt{rowspan="2"}।
\item ঘর নিচের দিকে ৩টি সারি জুড়লে \texttt{rowspan="3"}।
\item যে সারি বা কলাম একত্র হয়ে গেছে, সেই জায়গার জন্য আলাদা \texttt{<td>} বা \texttt{<th>} আর লিখবে না।
\end{enumerate}

\begin{center}

\subsection{ফাঁকা ঘর দেখানোর নিয়ম}

খালি \texttt{<td></td>} একটি বৈধ ঘর। সেটি দেখানোর জন্য \texttt{<br>} দেওয়া বাধ্যতামূলক নয়। ঘরের উচ্চতা, ফাঁকা জায়গা ও সীমানা CSS-এর নিয়মের ওপর নির্ভর করে। নিচের পুরোনো উদাহরণে \texttt{<br>} একটি ফাঁকা লাইনের জায়গা রাখে।

\begin{examplebox}{কোড}
\begin{verbatim}
<td><br></td>
\end{verbatim}
\end{examplebox}

নিচের \texttt{\&nbsp;} একটি অচ্ছেদ্য ফাঁকা অক্ষর রাখে; তাই এই ঘরটি আর সম্পূর্ণ খালি থাকে না। এটি ফাঁকা ঘরকে বৈধ করার নিয়ম নয়। নতুন পেজে প্রয়োজনীয় উচ্চতা ও ফাঁকা জায়গা CSS দিয়ে নির্ধারণ করা যায়।

\begin{examplebox}{কোড}
\begin{verbatim}
<td>&nbsp;</td>
\end{verbatim}
\end{examplebox}

\subsection{সাধারণ ভুল}

\begin{mistakebox}{সাধারণ ভুল}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{rowspan} ও \texttt{colspan} গুলিয়ে ফেলা।
\item কলাম একত্র করতে \texttt{rowspan} ব্যবহার করা।
\item সারি একত্র করতে \texttt{colspan} ব্যবহার করা।
\item \texttt{colspan="2"} দেওয়ার পরও একই সারিতে অতিরিক্ত ঘর লিখে ফেলা।
\item \texttt{rowspan="2"} দেওয়ার পরও পরের সারিতে একই ঘর আবার লিখে ফেলা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item উদ্ধৃতি না থাকলেই মান ভুল ভাবা: \texttt{colspan=2} বৈধ HTML। তবে একই লেখার রীতি রাখতে \texttt{colspan="2"} ব্যবহার করা সুবিধাজনক।
\item \texttt{<tr>} ট্যাগে \texttt{colspan} লেখা।
\item \texttt{<table>} ট্যাগে \texttt{rowspan} লেখা।
\item চিত্রের মোট সারি/কলাম না গুনে সরাসরি কোড লেখা।
\end{itemize}
\end{minipage}\par

\end{mistakebox}
