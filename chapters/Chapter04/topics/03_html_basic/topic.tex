\section{HTML-এর মৌলিক বিষয়}

% Composition pass 2026-10-04
শুধু সাধারণ লেখা লিখলে ব্রাউজার বুঝবে না কোনটি শিরোনাম, তালিকা বা লিংক।

\chFourMotivation{বিষয়বস্তুর পরিচয় দেওয়ার প্রয়োজন $\rightarrow$ HTML $\rightarrow$ ট্যাগ দিয়ে গঠন ও অর্থ প্রকাশ $\rightarrow$ কলেজের তথ্য একটি সুসংগঠিত ওয়েবপেজে দেখানো।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item HTML-এর পূর্ণরূপ ও কাজ ব্যাখ্যা করতে।
\item HTML ও World Wide Web-এর ইতিহাসে Tim Berners-Lee, Robert Cailliau, CERN, W3C (World Wide Web Consortium) ও WHATWG (Web Hypertext Application Technology Working Group)-এর অবদান সংক্ষেপে বলতে।
\item HTML ফাইল, ব্রাউজারে ফল প্রদর্শনের প্রক্রিয়া ও ডকুমেন্টের মৌলিক কাঠামো বুঝতে।
\item HTML, CSS (Cascading Style Sheets) ও JavaScript-এর ভূমিকা আলাদা করতে।
\end{itemize}
\end{learningbox}

\subsection{HTML কী}
\begin{definitionbox}{HTML — HyperText Markup Language}
ওয়েবপেজের বিষয়বস্তুর গঠন ও অর্থ প্রকাশের মার্কআপ ভাষাকে HTML বলে। ট্যাগ দিয়ে শিরোনাম, অনুচ্ছেদ, ছবি, লিংক, তালিকা, সারণি ও ফর্মের মতো উপাদান চিহ্নিত করা হয়।
\end{definitionbox}
HTML ফাইল সাধারণত \texttt{.html} বা \texttt{.htm} এক্সটেনশনে সংরক্ষণ করা হয়। ব্রাউজার ফাইলের মার্কআপ পড়ে পেজ দেখায়; আলাদা কম্পাইলার প্রয়োজন হয় না।

\textbf{কোড পড়ার নিয়ম:} প্রথম পূর্ণ নথির পর ছোট উদাহরণে শুধু প্রয়োজনীয় অংশ দেখানো হয়েছে। সাধারণত অংশটি \texttt{body}-তে বসাবে; \texttt{style}/\texttt{head} দেখানো থাকলে সেটি নথির যথাস্থানে রাখবে।

ট্যাগ বিষয়বস্তুর ধরন ও অর্থ জানায়। ব্রাউজার নিজের স্বাভাবিক রীতি এবং প্রয়োগ করা CSS অনুসারে পেজ দেখায়। তাই ট্যাগের অর্থ ও লেখার দৃশ্যমান রূপ আলাদা করে বুঝতে হয়।

\subsubsection{HTML-এর বিকাশ}
Tim Berners-Lee ওয়েব প্রকল্পে HTML-এর সূচনা করেন। ওয়েবের প্রস্তাব, প্রথম ব্রাউজার ও সার্ভার এবং CERN, Robert Cailliau, W3C ও WHATWG-এর অবদান WWW-এর ইতিহাস অংশে রয়েছে। HTML নথির গঠন প্রকাশ করে, HTTP ব্রাউজার–সার্ভারের যোগাযোগের নিয়ম দেয় এবং URL রিসোর্সের ঠিকানা নির্দেশ করে।
\begin{itemize}
\item ১৯৯৩ সালে HTML-এর প্রাথমিক বর্ণনা প্রকাশে ট্যাগ ব্যবহারের নিয়ম আরও স্পষ্ট হয়।
\item ১৯৯৭ সালে HTML 4.0 W3C Recommendation হিসেবে প্রকাশিত হয়।
\item ২০১৪ সালে HTML5 W3C Recommendation প্রকাশিত হয়; মাল্টিমিডিয়া, ওয়েব অ্যাপ্লিকেশন ও অর্থবোধক এলিমেন্টের ব্যবহার গুরুত্ব পায়।
\item HTML আধুনিক ওয়েবের একটি মৌলিক মান; HTML Living Standard-এর মাধ্যমে এর নিয়ম উন্নত করা হয়।
\end{itemize}

\subsection{HTML ব্যবহারের সুবিধা}

\begin{keypointbox}{HTML-এর প্রধান সুবিধা}
\begin{enumerate}
\item \textbf{সহজে শেখা:} HTML ব্যবহার করা ও বোঝা তুলনামূলক সহজ। অল্প কিছু ট্যাগ শিখেই কার্যকর ওয়েবপেজের কাঠামো লেখা যায়।
\item \textbf{সাধারণ এডিটরে কাজ:} HTML টেক্সটভিত্তিক। Notepad, Notepad++, Sublime Text বা VS Code-এর মতো সাধারণ টেক্সট বা কোড এডিটরে লেখা ও সম্পাদনা করা যায়।
\item \textbf{ব্রাউজারের সমর্থন:} প্রায় সব ওয়েব ব্রাউজার HTML সমর্থন করে।
\item \textbf{দ্রুত নির্মাণ:} HTML দিয়ে স্থির ওয়েবপেজ দ্রুত তৈরি করা যায়।
\item \textbf{ছোট ফাইল:} সাধারণ HTML ফাইল তুলনামূলক ছোট হয়, তাই দ্রুত লোড হতে পারে। তবে ছবি, ভিডিও, স্ক্রিপ্ট ও নেটওয়ার্কের গতিও লোড হওয়ার সময়কে প্রভাবিত করে।
\item \textbf{স্পষ্ট গঠন:} ট্যাগ দিয়ে বিষয়বস্তুর কাঠামো পরিষ্কার করা যায়; সার্চ ইঞ্জিন সাধারণত HTML কাঠামো সহজে পড়তে পারে।
\item \textbf{অন্য প্রযুক্তির সঙ্গে ব্যবহার:} CSS ও JavaScript-এর সঙ্গে HTML ব্যবহার করে সুন্দর ও ব্যবহারকারীর কাজে সাড়া দেয় এমন ওয়েবপেজ তৈরি করা যায়।
\end{enumerate}
\end{keypointbox}
\textbf{ফাইলের পরিচয়:} HTML ফাইল \texttt{.html} বা \texttt{.htm} এক্সটেনশনে সংরক্ষণ করা যায়। ট্যাগ ও অ্যাট্রিবিউটের নামের বড়–ছোট অক্ষরের নিয়ম ট্যাগ ও সিনট্যাক্স অংশে রয়েছে।

\subsection{HTML, CSS ও JavaScript}
HTML একটি মার্কআপ ভাষা; এতে প্রোগ্রামের মতো শর্ত বা লুপ লিখে হিসাব করা হয় না। CSS (Cascading Style Sheets) উপাদানের রং, ফন্ট, আকার, ফাঁক ও বিন্যাস নির্ধারণ করে। JavaScript ব্যবহারকারীর কাজ অনুযায়ী পেজে পরিবর্তন, হিসাব বা প্রতিক্রিয়া যোগ করে। নিচের সারণিতে তিনটির প্রয়োগ তুলনা করা হয়েছে।
\begin{center}
\begin{tabular}{|p{0.22\textwidth}|p{0.34\textwidth}|p{0.30\textwidth}|}
\hline
\textbf{প্রযুক্তি} & \textbf{কাজ} & \textbf{উদাহরণ} \\
\hline
HTML & বিষয়বস্তুর কাঠামো & শিরোনাম, অনুচ্ছেদ, সারণি \\
\hline
CSS & উপস্থাপন/রীতি & রং, বিন্যাস, ফাঁক \\
\hline
JavaScript & আচরণ/সক্রিয় প্রতিক্রিয়া & বৈধতা যাচাই, বোতাম ক্লিক, অ্যানিমেশন \\
\hline
\end{tabular}
\end{center}

\subsection{HTML ট্যাগের তালিকা ও উদাহরণ}

নিচের ছোট উদাহরণগুলোতে পরিচিত ট্যাগের কোড ও দৃশ্যমান ফল পাশাপাশি দেওয়া হয়েছে। প্রতিটি ট্যাগের পূর্ণ ব্যাখ্যা সংশ্লিষ্ট পাঠে রয়েছে।

\begin{keypointbox}{প্রধান HTML ট্যাগ (সংক্ষেপ)}
\begin{itemize}
\item \texttt{<html>, <head>, <body>} : ডকুমেন্টের মূল কঙ্কাল
\item \texttt{<h1>.. <h6>} : শিরোনাম
\item \texttt{<p>} : অনুচ্ছেদ
\item \texttt{<a>} : হাইপারলিংক
\item \texttt{<img>} : ছবি বসানো
\item \texttt{<ul>, <ol>, <li>} : তালিকা
\item \texttt{<table>, <tr>, <td>, <th>} : টেবিল
\item \texttt{<form>, <input>, <button>, <label>, <select>, <option>} : ফর্ম উপাদান
\item \texttt{<div>, <span>} : ধারক, ইনলাইন/ব্লক
\item \texttt{<strong>, <em>} : গুরুত্ব ও জোর
\item \texttt{<br>, <hr>} : লাইনে বিরতি, অনুভূমিক রেখা
\item \texttt{<pre>, <code>} : কোড/প্রি-ফরম্যাটেড টেক্সট
\item \texttt{<header>, <nav>, <main>, <section>, <article>, <aside>, <footer>} : পেজের অংশের অর্থ বোঝানো ট্যাগ
\end{itemize}
\end{keypointbox}

\begin{examplebox}{ট্যাগগুলোর নমুনা ও ফল}
\par\noindent\textbf{1. শিরোনাম (headings)}\par
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}
\begin{chFourBanglaCode}
\begin{verbatim}
<h1>বড় শিরোনাম</h1>
<h2>ছোট শিরোনাম</h2>
\end{verbatim}
\end{chFourBanglaCode}
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
ফল: বড় শিরোনাম (বড় ফন্ট), নিচে ছোট শিরোনাম (ছোট ফন্ট)।
\end{minipage}



\vspace{6pt}
\par\noindent\begin{minipage}{\linewidth}
\textbf{2. অনুচ্ছেদ}\\
\begin{chFourBanglaCode}
\begin{verbatim}
<p>এইটা একটি প্যারাগ্রাফ। HTML-এ প্যারাগ্রাফ আলাদা লাইনে দেখায়।</p>
\end{verbatim}
\end{chFourBanglaCode}
ফল: কোডের লেখাটি একটি পৃথক অনুচ্ছেদ হিসেবে দেখা যাবে।
\end{minipage}\par

\vspace{6pt}
\par\noindent\begin{minipage}{\linewidth}
\textbf{3. লিংক}\\
\begin{chFourBanglaCode}
\begin{verbatim}
<a href="https://www.example.com">Example সাইট</a>
\end{verbatim}
\end{chFourBanglaCode}
ফল: ক্লিকযোগ্য "Example সাইট" লিংক, ক্লিক করলে নির্দিষ্ট URL-এ যাবে।
\end{minipage}\par

\vspace{6pt}
\par\noindent\begin{minipage}{\linewidth}
\textbf{4. ছবি}\\
\begin{verbatim}
<img src="images/logo.png" alt="Logo" width="200">
\end{verbatim}
ফল: নির্দিষ্ট পথের ছবিটি পাওয়া গেলে তা দেখা যাবে। ছবি দেখানো না গেলে বিকল্প লেখা দেখা যেতে পারে; স্ক্রিন রিডারও এই বিকল্প লেখা ব্যবহার করে।
\end{minipage}\par

\vspace{6pt}
\par\noindent\textbf{5. তালিকা (বুলেট ও ক্রমযুক্ত)}\par
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}
\begin{chFourBanglaCode}
\begin{verbatim}
<ul>
	<li>প্রথম আইটেম</li>
	<li>দ্বিতীয় আইটেম</li>
</ul>
<ol>
	<li>প্রথম</li>
	<li>দ্বিতীয়</li>
</ol>
\end{verbatim}
\end{chFourBanglaCode}
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
ফল: বুলেট তালিকা ও নম্বর তালিকা আলাদাভাবে প্রদর্শিত হবে।
\end{minipage}
\chFourWhy{\texttt{ol}/\texttt{ul} তালিকার ধরন এবং \texttt{li} প্রতিটি উপাদান নির্ধারণ করে।}


\vspace{6pt}
\par\noindent\textbf{6. টেবিল}\par
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}
\begin{chFourBanglaCode}
\begin{verbatim}
<table border="1">
  <tr>
    <th>নাম</th><th>বয়স</th>
  </tr>
  <tr>
    <td>আজ</td><td>16</td>
  </tr>
</table>
\end{verbatim}
\end{chFourBanglaCode}
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
\textbf{ফল:}\par
\begin{tabular}{|c|c|}
\hline
\textbf{নাম} & \textbf{বয়স} \\
\hline
আজ & 16 \\
\hline
\end{tabular}\par
প্রথম সারির শিরোনাম সাধারণত গাঢ় ও মাঝখানে দেখাবে।
\end{minipage}
\chFourWhy{\texttt{tr} সারি এবং \texttt{th}/\texttt{td} সেই সারির ঘর তৈরি করে।}


\vspace{6pt}
\par\noindent\begin{minipage}{\linewidth}
\textbf{7. ফর্মের ইনপুট}\\
\begin{verbatim}
<form action="/submit">
	<label>Name: <input type="text" name="name"></label>
	<button type="submit">Send</button>
</form>
\end{verbatim}
ফল: একটি টেক্সট ইনপুট ঘর ও সাবমিট বোতাম প্রদর্শিত হবে; সাবমিট করলে \texttt{action}-এ দেওয়া ঠিকানায় তথ্য পাঠানোর অনুরোধ করবে।
\end{minipage}\par

\vspace{6pt}
\par\noindent\begin{minipage}{\linewidth}
\textbf{8. div এবং span}\\
\begin{verbatim}
<div style="background:#eef;padding:10px;">Block level container</div>
<span style="color:red;">Inline text</span>
\end{verbatim}
ফল: স্বাভাবিক প্রদর্শনে \texttt{div} নতুন ব্লকে বিষয়বস্তু রাখে। এখানে \texttt{span}-এর \texttt{style} অ্যাট্রিবিউট লেখাকে লাল করেছে; শুধু \texttt{span} লিখলেই রং বদলায় না।
\end{minipage}\par

\end{verbatim}
\end{chFourBanglaCode}
ফল: ব্রাউজার সাধারণত \texttt{strong}-এর লেখা মোটা অক্ষরে এবং \texttt{em}-এর লেখা বাঁকা অক্ষরে দেখায়।
\end{minipage}\par

\vspace{6pt}
\par\noindent\textbf{10. বিরতি এবং অনুভূমিক বিভাজক}\par
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}
\begin{verbatim}
Line 1<br>Line 2
<hr>
\end{verbatim}
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
ফল: লাইনে বিরতি এবং অনুভূমিক লাইনের মতো বিভাজক দেখা যাবে।
\end{minipage}



\vspace{6pt}
\par\noindent\begin{minipage}{\linewidth}
\textbf{11. pre এবং code}\\
\begin{verbatim}
<pre>
function greet() {
	console.log('Hello');
}
</pre>
<p>Inline code: <code>const x = 5;</code></p>
\end{verbatim}
ফল: \texttt{pre} ব্লক যেখানে স্পেস ও লাইনের ফরম্যাট রাখা হয়; \texttt{code} একই প্রস্থের অক্ষরে ইনলাইন টেক্সট দেখায়।
\end{minipage}\par

\vspace{6pt}
\par\noindent\begin{minipage}{\linewidth}
\textbf{12. পেজের অংশের অর্থ বোঝানো ট্যাগ (সংক্ষিপ্ত)}\par
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}
\begin{verbatim}
<header>Header content</header>
<nav>Navigation links</nav>
<main>Main article</main>
<aside>Sidebar</aside>
<footer>Footer info</footer>
\end{verbatim}
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
ফল: ব্রাউজারে এগুলো সাধারণত আলাদা ব্লক হিসেবে দেখা যায়। পেজের কোন অংশের কী ভূমিকা, তা বোঝাতে এগুলো সাহায্য করে; ফলে স্ক্রিন রিডার ও সার্চ ইঞ্জিনও গঠন বুঝতে পারে। শুধু ট্যাগ ব্যবহারে পেজের সার্চ ফলের অবস্থান নিশ্চিত হয় না।
\end{minipage}

\end{minipage}\par
