\section{ওয়েব ডিজাইনের ধারণা}

% Composition pass 2026-10-04
কলেজের তথ্য অনলাইনে প্রকাশ করলেই তা সহজে খুঁজে পাওয়া যায় না।

\chFourMotivation{তথ্য দ্রুত খুঁজে পাওয়ার প্রয়োজন $\rightarrow$ ওয়েব ডিজাইন $\rightarrow$ শিরোনাম, মেনু ও বিষয়বস্তু পরিকল্পিতভাবে সাজানো $\rightarrow$ শিক্ষার্থী সহজে নোটিশ ও ফল পায়।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item ওয়েব ডিজাইন কী তা সংজ্ঞাসহ ব্যাখ্যা করতে।
\item ওয়েবপেজ, ওয়েবসাইট, ওয়েব সার্ভার ও ওয়েব ব্রাউজারের ধারণা লিখতে।
\item ওয়েব ডিজাইনের উদ্দেশ্য, উপাদান, সুবিধা ও ব্যবহারক্ষেত্র ব্যাখ্যা করতে।
\item HTML (HyperText Markup Language), CSS (Cascading Style Sheets) ও JavaScript-এর ভূমিকা আলাদা করতে।
\item ভালো ওয়েব ডিজাইনের বৈশিষ্ট্য ও পরীক্ষায় গুরুত্বপূর্ণ বিষয় মনে রাখতে।
\end{itemize}
\end{learningbox}

\subsection{ভূমিকা}
ওয়েবসাইটে তথ্য প্রকাশের আগে পেজগুলোর নকশা পরিকল্পনা করতে হয়। কোন তথ্য কোথায় থাকবে, ছবি ও লিংক কীভাবে সাজানো হবে এবং ব্যবহারকারী কোন পথে প্রয়োজনীয় পেজে যাবেন—পরিকল্পনায় এসব ঠিক করা হয়। ব্যক্তি, শিক্ষাপ্রতিষ্ঠান, ব্যবসাপ্রতিষ্ঠান, সংবাদমাধ্যম ও সরকারি সংস্থা এই নকশার সাহায্যে নিজেদের তথ্য ও সেবা উপস্থাপন করে।

এই অধ্যায়ে ওয়েবসাইটের কাঠামো থেকে HTML-এর ব্যবহার পর্যন্ত ধাপে ধাপে আলোচনা করা হয়েছে। লেখার বিন্যাস, তালিকা, হাইপারলিংক, ছবি, সারণি ও ফর্মের উদাহরণ দেখে শিক্ষার্থী ওয়েবপেজ তৈরি, পরীক্ষা ও প্রকাশের কাজ শিখবে।

\subsection{ওয়েব ডিজাইন কী?}

\begin{definitionbox}{ওয়েব ডিজাইন (Web Design)}
ওয়েবসাইট বা ওয়েবপেজের তথ্য, ছবি, লিংক, টেবিল, মেনু ও অন্যান্য উপাদানকে নির্দিষ্ট কাঠামো অনুযায়ী সুন্দর ও ব্যবহারবান্ধবভাবে সাজানোর প্রক্রিয়াকে ওয়েব ডিজাইন বলে।
\end{definitionbox}

ওয়েব ডিজাইনে তথ্য যৌক্তিক ক্রমে সাজানো হয়, যাতে লেখা পড়া এবং প্রয়োজনীয় পেজ খুঁজে পাওয়া সহজ হয়। শুধু বাহ্যিক সৌন্দর্য নয়, মেনু ব্যবহার, তথ্য ডাউনলোড ও সেবা গ্রহণের সুবিধাও নকশার অংশ।

\subsection{ওয়েব ডিজাইনের মূল উদ্দেশ্য}

ওয়েব ডিজাইনের উদ্দেশ্য শুধু সুন্দর পেজ তৈরি করা নয়; বরং তথ্যকে এমনভাবে সাজানো যাতে ব্যবহারকারী দ্রুত, সহজে এবং স্পষ্টভাবে প্রয়োজনীয় তথ্য পায়।

\begin{keypointbox}{প্রধান উদ্দেশ্য}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item তথ্য সুন্দর ও সংগঠিতভাবে উপস্থাপন করা।
\item ব্যবহারকারীর জন্য সহজ নেভিগেশন তৈরি করা।
\item ওয়েবসাইটকে আকর্ষণীয় ও ব্যবহারবান্ধব করা।
\item প্রয়োজনীয় তথ্য দ্রুত খুঁজে পাওয়ার ব্যবস্থা করা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item লেখা, ছবি, লিংক, তালিকা, সারণি ও ফর্মের সঠিক ব্যবহার নিশ্চিত করা।
\item বিভিন্ন ব্রাউজার ও যন্ত্রে ওয়েবপেজ ঠিকভাবে দেখা যায় এমনভাবে তৈরি করা।
\item প্রতিষ্ঠান, পণ্য, সেবা বা ব্যক্তিগত পরিচিতি অনলাইনে প্রকাশ করা।
\item দর্শনার্থী বা ব্যবহারকারী যেন সহজে যোগাযোগ, ডাউনলোড বা সেবা নিতে পারে সে ব্যবস্থা রাখা।
\end{itemize}
\end{minipage}\par

\end{keypointbox}

ওয়েব ডিজাইন বুঝতে ওয়েবপেজ, ওয়েবসাইট ও হোমপেজের সম্পর্ক জানা দরকার। পরের অংশে এগুলোর পরিচয় এবং ক্লায়েন্ট ও সার্ভারের কাজ আলোচনা করা হয়েছে। ওয়েবের ইতিহাস ও ব্রাউজারের ভূমিকা রয়েছে WWW-সংক্রান্ত অংশে।

\subsection{নকশা থেকে পর্দায় ওয়েবপেজ}
নকশা অনুযায়ী HTML-এ পেজের কাঠামো, CSS-এ তার দৃশ্যমান বিন্যাস এবং প্রয়োজনে JavaScript-এ ব্যবহারকারীর কাজের প্রতিক্রিয়া তৈরি করা হয়। প্রকাশের পর ব্রাউজার সার্ভার থেকে এসব ফাইল নিয়ে পেজ দেখায়। অনুরোধ ও সাড়া আদান-প্রদানের বিস্তারিত ধাপ ক্লায়েন্ট–সার্ভার ও ওয়েব ঠিকানা অংশে রয়েছে।

\subsection{ওয়েব ডিজাইনে HTML-এর ভূমিকা}
HTML দিয়ে শিরোনাম, অনুচ্ছেদ, ছবি, তালিকা, সারণি, ফর্ম ও লিংকের অবস্থান এবং পারস্পরিক গঠন লেখা হয়। CSS ও JavaScript কীভাবে এই কাঠামোর সঙ্গে কাজ করে, তা HTML-এর মৌলিক বিষয় অংশে ব্যাখ্যা করা হয়েছে।

\subsection{ওয়েব ডিজাইনের সুবিধা}

\begin{enumerate}
\item \textbf{তথ্য প্রকাশ:} পরিকল্পিত ওয়েবসাইটে ব্যক্তি ও প্রতিষ্ঠানের তথ্য সহজে ও দ্রুত প্রকাশ করা যায়।
\item \textbf{তথ্যে প্রবেশ:} সার্ভার ও সংযোগ সচল থাকলে শিক্ষার্থী, অভিভাবক ও গ্রাহক বিভিন্ন স্থান থেকে ২৪ ঘণ্টা প্রয়োজনীয় তথ্য দেখতে পারেন।
\item প্রতিষ্ঠান বা ব্যবসার পরিচিতি বাড়ে।
\item অনলাইন সেবা দেওয়া যায়।
\item শিক্ষার্থীরা অনলাইন ক্লাস, নোট, ফল, রুটিন ও নোটিশ পেতে পারে।
\item গ্রাহক বা দর্শনার্থী সহজে যোগাযোগ করতে পারে।
\item কম খরচে বিশ্বব্যাপী প্রচার করা যায়।
\item প্রয়োজনীয় ফাইল, PDF, ফর্ম বা নোটিশ ডাউনলোড করা যায়।
\end{enumerate}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\subsection{ওয়েব ডিজাইনের ব্যবহারক্ষেত্র}

\begin{enumerate}
\item \textbf{শিক্ষা প্রতিষ্ঠান:} নোটিশ, ফল, ভর্তি, রুটিন, শিক্ষকদের তথ্য।
\item \textbf{ব্যবসা প্রতিষ্ঠান:} পণ্য, সেবা, অর্ডার, অর্থ পরিশোধের তথ্য।
\item \textbf{সংবাদমাধ্যম:} অনলাইন সংবাদ পোর্টাল।
\item \textbf{সরকারি সেবা:} নাগরিক সেবা, আবেদন, তথ্য প্রকাশ।
\item \textbf{স্বাস্থ্যসেবা:} সাক্ষাতের সময় নেওয়া, চিকিৎসকের তথ্য, স্বাস্থ্যসংক্রান্ত নোটিশ।
\item \textbf{ব্যক্তিগত পোর্টফোলিও:} নিজের কাজ, দক্ষতা, প্রকল্প দেখানো।
\item \textbf{ই-কমার্স:} অনলাইন কেনাকাটা, পণ্যের তালিকা, কেনাকাটার ঝুড়ি।
\item \textbf{ব্যাংকিং:} অনলাইন ব্যাংকিং তথ্য ও সেবা নোটিশ।
\end{enumerate}
\end{minipage}\par\smallskip
\subsection{ভালো ওয়েব ডিজাইনের বৈশিষ্ট্য}

একটি ভালো ওয়েবসাইট শুধু দেখতে সুন্দর হলেই যথেষ্ট নয়; সেটি ব্যবহার করাও সহজ হতে হবে।

\begin{keypointbox}{ভালো ওয়েব ডিজাইনের বৈশিষ্ট্য}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{enumerate}
\item পরিষ্কার ও সহজ বিন্যাস।
\item সহজ নেভিগেশন মেনু।
\item পাঠযোগ্য ফন্ট ও রং।
\item দ্রুত পেজ লোড হওয়া।
\item প্রয়োজনীয় ছবি ও বিষয়বস্তু।
\item মোবাইলে ব্যবহারযোগ্য নকশা।
\end{enumerate}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{enumerate}[start=7]
\item অকার্যকর লিংক না থাকা।
\item ব্যবহারবান্ধব কাঠামো।
\item তথ্য নিয়মিত হালনাগাদ করা।
\item ব্রাউজারের সঙ্গে সামঞ্জস্য থাকা।
\item অপ্রয়োজনীয় অ্যানিমেশন বা অতিরিক্ত নকশা এড়িয়ে চলা।
\end{enumerate}
\end{minipage}\par

\end{keypointbox}

\subsection{ওয়েবসাইট তৈরির ধাপ}

\begin{enumerate}
\item ওয়েবসাইটের উদ্দেশ্য ও লক্ষ্য ব্যবহারকারী নির্ধারণ করা।
\item কোন কোন পেজ থাকবে তা ঠিক করা।
\item বিষয়বস্তু, ছবি, ফাইল ও লিংক সংগ্রহ করা।
\item সাইটম্যাপ বা পেজের কাঠামো তৈরি করা।
\item খসড়া বিন্যাস বা ওয়্যারফ্রেম তৈরি করা।
\item HTML দিয়ে ওয়েবপেজের কাঠামো তৈরি করা।
\item প্রয়োজনে CSS দিয়ে পেজ সুন্দর করা।
\item ব্রাউজারে ফল পরীক্ষা করা।
\item লিংক, ছবি, সারণি ও ফর্ম ঠিক আছে কিনা পরীক্ষা করা।
\item হোস্টিং সার্ভারে ওয়েবসাইট প্রকাশ করা।
\end{enumerate}

ওয়েবসাইটের কাজ অনুযায়ী প্রযুক্তি বেছে নিতে হয়। পরিচিতি প্রকাশ ও রোলভিত্তিক ফল অনুসন্ধানের চাহিদা এক নয়। স্ট্যাটিক, ডাইনামিক, লোকাল ও রিমোট সাইটের পার্থক্য ওয়েবসাইটের প্রকারভেদ অংশে আলোচনা করা হয়েছে।

\subsection{একটি সাধারণ HTML কাঠামো}

নিচের কোডটি একটি সাধারণ ওয়েবপেজের মৌলিক কাঠামো দেখায়।

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.54\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <title>Example College</title>
</head>
<body>
  <h1>Example College</h1>
  <p>Welcome to our college website.</p>
  <a href="notice.html">Notice</a>
  <br>
  <img src="images/college.jpg" alt="College" width="300">
</body>
</html>
\end{verbatim}
\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.42\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}ফল সম্পর্কে ধারণা}\par
ব্রাউজারের ট্যাবে \texttt{Example College} লেখা দেখা যাবে। পেজে থাকবে কলেজের শিরোনাম, স্বাগতবার্তা, Notice লিংক ও কলেজের ছবি। \texttt{images} ফোল্ডারে \texttt{college.jpg} ফাইলটি থাকলে ছবিটি দেখা যাবে।
\par\smallskip
\end{minipage}\par


\begin{keypointbox}{কোডের ব্যাখ্যা}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{<!DOCTYPE html>} আধুনিক HTML-এর ডকুমেন্ট ঘোষণা; এটি ব্রাউজারকে মানসম্মত মোডে পেজ দেখাতে সাহায্য করে।
\item \texttt{<title>} ব্রাউজারের ট্যাবের শিরোনাম নির্ধারণ করে।
\item \texttt{<h1>} সর্বোচ্চ স্তরের শিরোনাম তৈরি করে; সাধারণ বিন্যাসে এটি বড় আকারে দেখা যায়।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{<p>} অনুচ্ছেদ তৈরি করে।
\item \texttt{<a href="notice.html">} অভ্যন্তরীণ হাইপারলিংক তৈরি করে।
\item \texttt{<img>} ওয়েবপেজে ছবি দেখায়।
\end{itemize}
\end{minipage}\par

\end{keypointbox}
