\section{ওয়েবসাইটের প্রকারভেদ}

% Composition pass 2026-10-04
পরিচিতি পেজে সবার জন্য একই লেখা থাকে, কিন্তু ফল খোঁজার পেজে রোলভেদে তথ্য বদলায়।

\chFourMotivation{চাহিদা অনুযায়ী প্রযুক্তি নির্বাচন $\rightarrow$ স্ট্যাটিক/ডাইনামিক সাইট $\rightarrow$ প্রস্তুত ফাইল বা প্রোগ্রামে তৈরি ফল $\rightarrow$ পরিচিতি প্রকাশ ও শিক্ষার্থীভিত্তিক সেবা। লোকাল/রিমোট বোঝায় সাইট কোথায় আছে।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী স্ট্যাটিক ও ডাইনামিক ওয়েবসাইটের মধ্যে পার্থক্য, এবং লোকাল ও রিমোট ওয়েবসাইট কী তা ব্যাখ্যা করতে পারবে।
\end{learningbox}

\subsection{ভূমিকা}
ওয়েবসাইটকে এখানে দুটি ভিত্তিতে ভাগ করা হয়েছে। বিষয়বস্তু পরিবেশনের ধরন অনুযায়ী সাইট স্ট্যাটিক বা ডাইনামিক; সার্ভারের অবস্থান অনুযায়ী লোকাল বা রিমোট। একটি স্ট্যাটিক সাইট যেমন দূরবর্তী সার্ভারে প্রকাশ করা যায়, তেমনি ডাইনামিক সাইটও নিজের কম্পিউটারে পরীক্ষা করা যায়।

\subsection{স্ট্যাটিক ওয়েবসাইট (Static Website)}
\begin{definitionbox}{}
যে ওয়েবসাইট প্রস্তুত পেজ পরিবেশন করে এবং সাধারণভাবে ব্যবহারকারীভেদে বিষয়বস্তু তৈরি করে না, তাকে \textbf{স্ট্যাটিক ওয়েবসাইট} বলে। বিষয়বস্তু বদলাতে সংরক্ষিত পেজ সম্পাদনা করতে হয়।
\end{definitionbox}

\begin{keypointbox}{স্ট্যাটিক ওয়েবসাইটের বৈশিষ্ট্য}
\begin{enumerate}
\item \textbf{প্রস্তুত বিষয়বস্তু:} বিষয়বস্তু স্থির থাকে; সাধারণত HTML/CSS দিয়ে তৈরি।
\item \textbf{পেজ পরিবেশন:} সরল স্ট্যাটিক সাইটে পেজ পাঠানোর সময় ডেটাবেজ থেকে নতুন HTML তৈরি করার প্রয়োজন হয় না।
\end{enumerate}
\end{keypointbox}
\subsubsection{সুবিধা}
\begin{enumerate}
\item \textbf{সহজ নির্মাণ:} সরল পেজ তৈরি ও রক্ষণাবেক্ষণ সহজ।
\item \textbf{লোড হওয়ার গতি:} সরল পেজ সাধারণত দ্রুত লোড হয়; গতি ফাইলের আকার ও সংযোগের ওপরও নির্ভর করে।
\end{enumerate}
\subsubsection{ব্যবহার}
\begin{enumerate}
\item \textbf{পরিচিতি ও বিজ্ঞপ্তি:} পোর্টফোলিও, বিবরণী ও নোটিশ পেজে উপযোগী।
\end{enumerate}

\subsection{ডাইনামিক ওয়েবসাইট (Dynamic Website)}
\begin{definitionbox}{}
যে ওয়েবসাইট ব্যবহারকারীর ইনপুট, সময়, ডেটাবেজ বা প্রোগ্রামের শর্ত অনুযায়ী বিষয়বস্তু তৈরি বা পরিবর্তন করে, তাকে \textbf{ডাইনামিক ওয়েবসাইট} বলে।
\end{definitionbox}

\begin{keypointbox}{ডাইনামিক ওয়েবসাইটের বৈশিষ্ট্য}
\begin{enumerate}
\item \textbf{পরিবর্তনশীল বিষয়বস্তু:} সার্ভার বা ক্লায়েন্টে চলা স্ক্রিপ্ট বিষয়বস্তু পরিবর্তন করতে পারে; প্রয়োজন অনুযায়ী ডেটাবেজ ব্যবহার হয়।
\item \textbf{সেবার ব্যবস্থা:} লগইন, নিবন্ধন, মন্তব্য, অনুসন্ধান ও অনলাইন অর্ডারের ব্যবস্থা থাকতে পারে।
\end{enumerate}
\end{keypointbox}
\subsubsection{ব্যবহার}
\begin{enumerate}
\item \textbf{ব্যক্তিভিত্তিক ও অনলাইন সেবা:} ই-কমার্স, সামাজিক যোগাযোগমাধ্যম, শিক্ষার্থীর লগইন পোর্টাল ও অনলাইন ফল অনুসন্ধানে ব্যবহার হয়।
\end{enumerate}


\subsubsection{HTML ফাইল ও সার্ভারের স্ক্রিপ্ট}
HTML ফাইলের প্রচলিত এক্সটেনশন \texttt{.html} বা \texttt{.htm}। ডাইনামিক পেজ তৈরিতে PHP-এর মতো সার্ভারে চলা স্ক্রিপ্ট ব্যবহার করা যায়। PHP-এর পূর্ণরূপ PHP: Hypertext Preprocessor; এর ফাইলের প্রচলিত এক্সটেনশন \texttt{.php}। সার্ভার স্ক্রিপ্ট চালিয়ে যে HTML তৈরি করে, ব্রাউজার তা পায় ও দেখায়। তাই PHP কোডকে সাধারণ HTML ফাইলের মতো শুধু ব্রাউজারে খুললে সার্ভারের কাজ হয় না।

কোনো URL-এ এক্সটেনশন দেখা বা না দেখা থেকেই সাইট স্ট্যাটিক না ডাইনামিক, তা নিশ্চিত করা যায় না। সার্ভারের নিয়মে এক্সটেনশন আড়াল করা যায়; একই সাইটে প্রস্তুত পেজ ও প্রোগ্রামে তৈরি পেজ দুটোই থাকতে পারে।

\subsection{লোকাল ওয়েবসাইট (Local Website)}
\begin{definitionbox}{}
নিজ কম্পিউটার বা স্থানীয় সার্ভারে রাখা ও ব্যবহার করা ওয়েবসাইটকে \textbf{লোকাল ওয়েবসাইট} বলে। সাধারণত সাইট তৈরি ও পরীক্ষার সময় এই ব্যবস্থা ব্যবহৃত হয়।
\end{definitionbox}

\begin{keypointbox}{লোকাল ওয়েবসাইটের বৈশিষ্ট্য}
\begin{enumerate}
\item \textbf{স্থানীয় পরিবেশ:} অফলাইন বা স্থানীয় পরিবেশে পরীক্ষা করা যায়; XAMPP, WAMP ও Live Server পরিচিত সহায়ক সফটওয়্যার।
\end{enumerate}
\end{keypointbox}
\subsubsection{সুবিধা}
\begin{enumerate}
\item \textbf{হোস্টিং খরচ:} নিজের যন্ত্র ব্যবহার করলে সাধারণত আলাদা হোস্টিং খরচ লাগে না।
\end{enumerate}
\subsubsection{ব্যবহার}
\begin{enumerate}
\item \textbf{নির্মাণ ও পরীক্ষা:} ত্রুটি পরীক্ষা ও দ্রুত নির্মাণের জন্য উপযোগী।
\end{enumerate}

\subsection{রিমোট ওয়েবসাইট (Remote Website)}
\begin{definitionbox}{}
ব্যবহারকারীর নিজের কম্পিউটারের বাইরে দূরবর্তী সার্ভারে রাখা ওয়েবসাইটকে \textbf{রিমোট ওয়েবসাইট} বলে। এটি ইন্টারনেটের মাধ্যমে ব্যবহার করা যায়; প্রবেশাধিকার প্রকাশ্য বা নিয়ন্ত্রিত হতে পারে।
\end{definitionbox}

\begin{keypointbox}{রিমোট ওয়েবসাইটের বৈশিষ্ট্য}
\begin{enumerate}
\item \textbf{অনলাইন প্রবেশ:} বিভিন্ন স্থান থেকে ব্যবহার করা যায়।
\item \textbf{সার্ভার ও ঠিকানা:} সার্ভারে রাখার ব্যবস্থা ও উপযুক্ত ঠিকানা প্রয়োজন। আলাদা ডোমেইন নাম সাধারণত ব্যবহার করা হয়, তবে প্রতিটি ক্ষেত্রে তা কেনা বাধ্যতামূলক নয়।
\end{enumerate}
\end{keypointbox}
\subsubsection{রক্ষণাবেক্ষণের শর্ত}
\begin{enumerate}
\item \textbf{সুরক্ষা ও সংরক্ষণ:} রক্ষণাবেক্ষণ, ব্যাকআপ ও নিরাপত্তা জরুরি।
\end{enumerate}

\hline

\hline
অবস্থান & নিজের কম্পিউটার/লোকাল সার্ভার & হোস্টিং/দূরবর্তী সার্ভার \\
\hline
প্রবেশের সুযোগ & নিজের যন্ত্র বা অনুমোদিত স্থানীয় নেটওয়ার্ক থেকে & ইন্টারনেটের মাধ্যমে; অনুমতির শর্ত থাকতে পারে \\

\hline

\hline
\end{tabular}
\end{center}

\begin{examplebox}{ওয়েবসাইটের ধরন বোঝার ধাপ}
স্ট্যাটিক: ব্যবহারকারী → ব্রাউজার → ওয়েব সার্ভার → স্থির HTML পেজ \\
ডাইনামিক: ব্যবহারকারী → ব্রাউজার → ওয়েব সার্ভার → সার্ভার স্ক্রিপ্ট → ডেটাবেজ → তৈরি করা পেজ

ডাইনামিক উদাহরণে ডেটাবেজ দেখানো হয়েছে; সব ডাইনামিক সাইটে এটি বাধ্যতামূলক নয়। স্ট্যাটিক ও ডাইনামিক বিষয়বস্তু পরিবেশনের ধরন, আর লোকাল ও রিমোট সাইটের অবস্থান বোঝায়। তাই একটি স্ট্যাটিক সাইটও রিমোট হতে পারে। \\
লোকাল ও রিমোট: কম্পিউটার → ব্রাউজার → (লোকাল ফাইল) / (ইন্টারনেট → ওয়েব সার্ভার)
\end{examplebox}

\subsection{স্ট্যাটিক ওয়েবসাইটের উপযোগী ক্ষেত্র}

যেখানে তথ্য ঘন ঘন বদলায় না এবং ব্যবহারকারীর জন্য আলাদা তথ্য তৈরি করতে হয় না, সেখানে সাধারণ স্ট্যাটিক সাইট যথেষ্ট হতে পারে। HTML/CSS দিয়ে এমন পেজ তৈরি করা সহজ। ছবির আকার ও ইন্টারনেট সংযোগ অনুকূল থাকলে পেজ দ্রুত খোলে।

\begin{examplebox}{স্ট্যাটিক ওয়েবসাইটের উদাহরণ}
\begin{itemize}
\item ব্যক্তিগত পোর্টফোলিও
\item কলেজ নোটিশ পেজ
\item পণ্যের তথ্য পেজ
\item অনুষ্ঠানের ঘোষণার পেজ
\item প্রতিষ্ঠানের সংক্ষিপ্ত পরিচিতি
\end{itemize}
\end{examplebox}

Facebook-এর নিউজ ফিডে ব্যবহারকারীভেদে ভিন্ন পোস্ট দেখা যায়। এটি পরিবর্তনশীল বিষয়বস্তুর একটি পরিচিত উদাহরণ। ডেটাবেজ, প্রোগ্রাম ও নিরাপত্তাব্যবস্থা থাকায় ডাইনামিক সাইটের রক্ষণাবেক্ষণ তুলনামূলক জটিল হতে পারে।

\subsection{ডাইনামিক ওয়েবসাইটের উপযোগী ক্ষেত্র}

ব্যবহারকারীর ইনপুট যাচাই করে আলাদা ফল দেখানো বা তথ্য সংরক্ষণের জন্য প্রোগ্রাম দরকার হয়। HTML/CSS-এর সঙ্গে সার্ভারের প্রোগ্রাম ও প্রয়োজনে ডেটাবেজ এই কাজ করে। লগইন ও অর্থ পরিশোধের মতো কাজে নিরাপত্তার ব্যবস্থাও প্রয়োজন।

\subsection{স্ট্যাটিক থেকে ডাইনামিক হওয়ার উদাহরণ}

একটি কলেজ ওয়েবসাইট প্রথমে শুধু নোটিশ, ছবি ও যোগাযোগের তথ্য দেখালে সেটি স্ট্যাটিক হতে পারে। কিন্তু একই ওয়েবসাইটে যদি শিক্ষার্থী লগইন, অনলাইন ফল অনুসন্ধান, ভর্তি ফর্ম জমা দেওয়ার ব্যবস্থা বা ডেটাবেজ থেকে নোটিশ হালনাগাদ করার ব্যবস্থা থাকে, তাহলে সেটি ডাইনামিক ওয়েবসাইট হবে।

\begin{keypointbox}{সহজ সিদ্ধান্ত নেওয়ার নিয়ম}
\begin{itemize}
\item শুধু স্থির তথ্য দেখালে: স্ট্যাটিক ওয়েবসাইট।
\item ব্যবহারকারীর ইনপুট নিয়ে ফল/অনুসন্ধান/লগইন দেখালে: ডাইনামিক ওয়েবসাইট।
\item নিজের কম্পিউটারে পরীক্ষা করলে: লোকাল ওয়েবসাইট।
\item দূরবর্তী সার্ভারে রেখে ইন্টারনেটে ব্যবহার করলে: রিমোট ওয়েবসাইট; এটি প্রকাশ্য হওয়া বাধ্যতামূলক নয়।
\end{itemize}
\end{keypointbox}

\subsection{সমস্যা ও সমাধান}

\begin{practicebox}
\textbf{সমস্যা:} একটি কলেজ ওয়েবসাইটে শুধু কলেজ পরিচিতি, ছবি, বিষয়ের তালিকা এবং যোগাযোগের ঠিকানা আছে। কিন্তু আরেকটি ওয়েবসাইটে শিক্ষার্থীর রোল দিলে ডেটাবেজ থেকে ফল দেখা যায়। কোনটি স্ট্যাটিক এবং কোনটি ডাইনামিক?

\textbf{সমাধান:}
\begin{itemize}
\item প্রথম ওয়েবসাইটে শুধু স্থির তথ্য আছে, তাই এটি স্ট্যাটিক ওয়েবসাইট।
\item দ্বিতীয় ওয়েবসাইট ব্যবহারকারীর ইনপুট নেয় এবং ডেটাবেজ থেকে ফল দেখায়, তাই এটি ডাইনামিক ওয়েবসাইট।
\item যদি প্রথম ওয়েবসাইট নিজের কম্পিউটারে পরীক্ষা করা হয়, তাহলে সেটি লোকাল ওয়েবসাইট।
\item যদি সার্ভারে আপলোড করে ইন্টারনেটে দেখা যায়, তাহলে সেটি রিমোট ওয়েবসাইট।
\end{itemize}
\end{practicebox}



\begin{mistakebox}{সাধারণ ভুল}
\begin{itemize}
\item শুধু অনলাইন আছে বলেই ওয়েবসাইটকে ডাইনামিক বলা।
\item ডেটাবেজ থাকলেই ডাইনামিক এবং না থাকলেই স্ট্যাটিক—এমন সিদ্ধান্ত নেওয়া।
\item লোকাল ওয়েবসাইট মানে ছোট ওয়েবসাইট মনে করা।
\item রিমোট ওয়েবসাইট মানে শুধু বিদেশি ওয়েবসাইট মনে করা।
\item স্ট্যাটিক ওয়েবসাইটে কখনো ছবি/লিংক/সারণি থাকতে পারে না--এমন ভুল ধারণা রাখা।
\end{itemize}
\end{mistakebox}


\begin{practicebox}
\begin{enumerate}
\item একটি স্ট্যাটিক ও একটি ডাইনামিক ওয়েবসাইটের উদাহরণ লেখ।
\item লোকাল ও রিমোট ওয়েবসাইটের মধ্যে ৩টি পার্থক্য লেখ।
\item লগইন, ব্যক্তিগত তথ্য ও অর্থ পরিশোধের কাজযুক্ত ডাইনামিক সাইটে নিরাপত্তা বিশেষভাবে গুরুত্বপূর্ণ কেন—ব্যাখ্যা কর।
\item অনলাইন ফল প্রকাশের ব্যবস্থা স্ট্যাটিক না ডাইনামিক? কারণসহ লেখ।
\item নিজের কম্পিউটারে \texttt{index.html} খুললে সেটি লোকাল না রিমোট? ব্যাখ্যা কর।
\end{enumerate}
\end{practicebox}
