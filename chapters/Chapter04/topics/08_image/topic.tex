\section{HTML-এ ছবি যুক্ত করা}

% Composition pass 2026-10-04
কলেজের পরিচিতি বা দোকানের পণ্য শুধু লেখা দিয়ে দেখালে চেনা কঠিন হতে পারে।

\chFourMotivation{দৃশ্যমান পরিচয় $\rightarrow$ img $\rightarrow$ src থেকে ছবির রিসোর্স লোড ও alt দিয়ে বিকল্প অর্থ $\rightarrow$ ক্যাম্পাসের ছবি, ব্যানার ও পণ্য প্রদর্শন।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item HTML পেজে ছবি যুক্ত করার নিয়ম ব্যাখ্যা করতে।
\item \texttt{<img>} ট্যাগ এম্পটি ট্যাগ কেন তা লিখতে।
\item \texttt{src}, \texttt{alt}, \texttt{width} ও \texttt{height} অ্যাট্রিবিউট ব্যবহার করতে।
\item ছবির অবস্থানের পথ, ছবির ফরম্যাট, ছবির অবস্থান, বর্ডার ও ছবিতে লিংক বুঝতে।
\item ধারকের মাপ অনুযায়ী ছবি ইঙ্গিত যেমন \texttt{srcset}, \texttt{sizes} ও \texttt{loading} চিনতে।
\item সারণির ঘরের ভিতরে ছবি বসানোর কোড লিখতে।
\item HSC ব্যবহারিক অনুশীলন ও বোর্ড পরীক্ষার ছবি সম্পর্কিত প্রশ্নের উত্তর দিতে।
\end{itemize}
\end{learningbox}

হাইপারলিংক শেখার পর ছবি শেখা সহজ, কারণ ওয়েবপেজে ছবি শুধু দেখানোই নয়, ছবির ওপর লিংকও দেওয়া যায়। কোনো কলেজের ওয়েবসাইটে লোগো, ব্যানার, শিক্ষকের ছবি, ক্যাম্পাসের ছবি, বিজ্ঞপ্তি বা ছবির সংগ্রহ দেখাতে HTML-এর \texttt{<img>} ট্যাগ ব্যবহার করা হয়।

\subsection{ছবি ট্যাগ কী?}

\begin{definitionbox}{ছবির ট্যাগ (Image Tag)}
ওয়েবপেজে ছবি যুক্ত করতে \texttt{<img>} ট্যাগ ব্যবহার করা হয়। এটি ছবির ট্যাগ।
\end{definitionbox}

তবে শুধু \texttt{<img>} লিখলে ছবি দেখা যাবে না। ব্রাউজারকে ছবির ফাইল কোথায় আছে তা জানাতে \texttt{src} অ্যাট্রিবিউট ব্যবহার করতে হয়। ছবির বিকল্প লেখা দেওয়ার জন্য \texttt{alt} অ্যাট্রিবিউট ব্যবহার করা হয়। ছবি দেখানো না গেলে এই লেখা দেখা যেতে পারে; স্ক্রিন রিডারও এটি ব্যবহার করে।

\subsection{\texttt{<img>} কেন এম্পটি ট্যাগ?}

\texttt{<img>} ওয়েবপেজে ছবি যুক্ত করে। এটি ভয়েড এলিমেন্টের শুরুর ট্যাগ; পাঠ্যবইয়ে এম্পটি ট্যাগ নামে পরিচিত।

\texttt{<img>} ট্যাগকে এম্পটি ট্যাগ বলা হয়, কারণ:

\begin{itemize}
\item এর শেষের ট্যাগ নেই।
\item এর ভিতরে লেখা বিষয়বস্তু থাকে না।
\item এটি অ্যাট্রিবিউট ব্যবহার করে ছবি প্রদর্শন করে।
\end{itemize}



\subsection{ছবি যুক্ত করার সাধারণ রূপ}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<img src="image-name">
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<img src="image-name.jpg" alt="Image description">
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\subsection{\texttt{src} অ্যাট্রিবিউট}

\begin{definitionbox}{\texttt{src}}
\texttt{src} অ্যাট্রিবিউট ছবির ফাইলের ঠিকানা জানায়। ব্রাউজার ওই ঠিকানা থেকে ছবিটি সংগ্রহ করে।
\end{definitionbox}

\subsubsection{উদাহরণ}

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}

\begin{verbatim}
<img src="rose.jpg" alt="Rose">
\end{verbatim}

\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
\textbf{ফল:} ব্রাউজার একই ফোল্ডার থেকে \texttt{rose.jpg} ছবি খুঁজে দেখাবে।
\end{minipage}
\chFourWhy{HTML ফাইল ও ছবি একই ফোল্ডারে থাকলে শুধু ছবির ফাইলের নাম লিখলেই চলে। অন্য ফোল্ডারে থাকলে পথ দিতে হয়।}
\end{chFourWholeExample}



\subsection{\texttt{alt} অ্যাট্রিবিউট}

\begin{definitionbox}{\texttt{alt}}
\texttt{alt} অ্যাট্রিবিউট ছবির বিকল্প লেখা নির্ধারণ করে। ছবি লোড না হলে বা স্ক্রিন রিডার ব্যবহার করলে এই লেখা কাজে লাগে।
\end{definitionbox}

\subsubsection{উদাহরণ}

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.54\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<img src="college.jpg" alt="ABC College Building">
\end{verbatim}

\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.42\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}ফল ও ব্যাখ্যা}\par
\textbf{ফল:} \texttt{college.jpg} ছবি লোড হলে কলেজ ভবনের ছবি দেখা যাবে। ছবি লোড না হলে \textbf{ABC College Building} লেখা দেখা যেতে পারে।

\textbf{ব্যাখ্যা:} ছবির উদ্দেশ্য অনুযায়ী বিকল্প লেখা দিলে ছবি দেখতে না পাওয়া ব্যবহারকারীও তথ্য বুঝতে পারেন। কেবল সাজানোর জন্য ব্যবহৃত ছবিতে \texttt{alt=""} দিয়ে তা বোঝানো যায়।
\par\smallskip
\end{minipage}\par





\subsection{\texttt{width} ও \texttt{height} অ্যাট্রিবিউট}

\begin{definitionbox}{\texttt{width} ও \texttt{height}}
\texttt{width} ছবির প্রস্থ এবং \texttt{height} ছবির উচ্চতা নির্ধারণ করে।
\end{definitionbox}

\subsubsection{সিনট্যাক্স}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<img src="image.jpg" width="300" height="200" alt="Image">
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\subsubsection{উদাহরণ}

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.54\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<img src="flower.png" width="400" height="200" alt="Flower">
\end{verbatim}

\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.42\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}ফল ও ব্যাখ্যা}\par
\textbf{ফল:} ব্রাউজার \texttt{flower.png} ছবি 400 পিক্সেল প্রস্থ এবং 200 পিক্সেল উচ্চতায় দেখাবে।

\textbf{ব্যাখ্যা:} \texttt{width} ও \texttt{height} অ্যাট্রিবিউট ছবির প্রদর্শিত আকার নিয়ন্ত্রণ করে। শুধু প্রস্থ দিলে ব্রাউজার স্বাভাবিকভাবে অনুপাত ধরে উচ্চতা নির্ধারণ করতে পারে।
\par\smallskip
\end{minipage}\par


\begin{keypointbox}{আধুনিক HTML-এর নিয়ম}
ধারকের মাপ অনুযায়ী বদলানো নকশার জন্য CSS ব্যবহার করা ভালো। \texttt{width} ও \texttt{height} দিয়ে ছবির প্রদর্শিত প্রস্থ ও উচ্চতাও নির্ধারণ করা যায়।
\end{keypointbox}

\subsection{ছবির ফাইলের ফরম্যাট}

ওয়েবপেজে \texttt{.jpg}, \texttt{.jpeg}, \texttt{.png}, \texttt{.gif} ও \texttt{.bmp} ফরম্যাটের ছবি ব্যবহার করা যায়। \texttt{.svg}, \texttt{.webp} ও \texttt{.avif}-ও আধুনিক ওয়েবে ব্যবহৃত হয়। ফরম্যাটভেদে ফাইলের আকার, স্বচ্ছতা ও সংকোচনের ধরন আলাদা হয়।

\begin{center}
\begin{tabular}{|p{0.25\textwidth}|p{0.55\textwidth}|}
\hline
\textbf{ফরম্যাট} & \textbf{ব্যবহার} \\
\hline
\texttt{.jpg} / \texttt{.jpeg} & ফটোগ্রাফ ও ক্যামেরায় তোলা ছবি \\
\hline
\texttt{.png} & স্বচ্ছ পটভূমির ছবি, স্ক্রিনশট ও লোগো \\
\hline
\texttt{.gif} & সহজ অ্যানিমেশন বা ছোট গ্রাফিক \\
\hline
\texttt{.svg} & ভেক্টর ছবি, লোগো ও আইকন \\
\hline
\texttt{.bmp} & বিটম্যাপ ছবি; ওয়েবে কম ব্যবহৃত \\
\hline
\texttt{.webp} & সংকুচিত স্থির বা অ্যানিমেটেড ছবি \\
\hline
\texttt{.avif} & বেশি সংকোচনের সুবিধাযুক্ত ছবি \\
\hline
\end{tabular}
\end{center}

HTML মানে একটি নির্দিষ্ট ছবির ফরম্যাট বাধ্যতামূলক নয়; ব্রাউজারের সমর্থন ভিন্ন হতে পারে। WebP ও AVIF-এ সংকুচিত ছবি রাখা যায়, আর SVG ভেক্টর ছবির আকার পরিবর্তনেও রেখা স্পষ্ট রাখতে পারে। ছবির ধরন, আকার ও প্রয়োজনীয় ব্রাউজারের সমর্থন দেখে ফরম্যাট বেছে নিতে হয়।

\subsection{ছবির অবস্থানের পথ}

ছবির ফাইল কোথায় রাখা আছে তার উপর \texttt{src} মান নির্ভর করে।

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}

\begin{verbatim}
<img src="flower.jpg" alt="Flower">
\end{verbatim}

\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
HTML ফাইল ও \texttt{flower.jpg} একই ফোল্ডারে থাকলে ব্রাউজার ছবি দেখাবে।
\end{minipage}

\end{chFourWholeExample}

\subsubsection{ছবি আলাদা ফোল্ডারে থাকলে}

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}

\begin{verbatim}

\end{verbatim}

\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
এখানে ব্রাউজার \texttt{images} ফোল্ডারের ভিতরে \texttt{flower.jpg} ছবি খুঁজবে।
\end{minipage}

\end{chFourWholeExample}

\begin{mistakebox}{ছবি না দেখালে কী যাচাই করবে?}
\begin{itemize}
\item \texttt{src} পথ ভুল কি না।
\item ছবির ফাইলের নাম ঠিক আছে কি না।
\item এক্সটেনশন \texttt{.jpg}, \texttt{.png}, \texttt{.gif} ঠিক আছে কি না।
\item ছবির ফাইল নির্ধারিত ফোল্ডারে আছে কি না।
\item ইংরেজি বড় বা ছোট হাতের অক্ষর ও বানানে অমিল আছে কি না।
\end{itemize}
\end{mistakebox}

\subsection{ছবি যুক্ত করার পূর্ণাঙ্গ উদাহরণ}

\Needspace{6\baselineskip}

\begin{multicols}{2}
\begin{verbatim}
<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <title>Image Example</title>
</head>
<body>
  <h1>My College</h1>
  <p>Below is an image of our college:</p>
  <img src="college.jpg"
       alt="ABC College Building"
       width="400" height="250">
  <p>Click the banner below to visit our website:</p>
  <a href="https://www.example.com">
    <img src="banner.jpg"
         alt="College Website Banner"
         width="300" height="100">
  </a>
</body>
</html>
\end{verbatim}
\end{multicols}


\textbf{\color{chFourTeal}ফল কী হবে?}\par
ব্রাউজারে দেখা যাবে:
\begin{itemize}
\item \textbf{My College} শিরোনাম।
\item একটি কলেজ ছবি।
\item একটি ব্যানারের ছবি।
\item ব্যানারের ছবিতে ক্লিক করলে নির্দিষ্ট ওয়েবসাইট খুলবে।
\end{itemize}
\par\smallskip



\subsection{সাধারণ ভুল}

\begin{multicols}{2}
\small
\begin{itemize}
\item \texttt{<image>} ট্যাগ লেখা; সঠিক ট্যাগ হলো \texttt{<img>}।
\item \texttt{src} অ্যাট্রিবিউট না দেওয়া।
\item \texttt{src}-এর জায়গায় \texttt{href} লেখা।
\item ছবির ফাইলের নাম ভুল লেখা।
\item \texttt{.jpg} এর বদলে \texttt{.png} বা এক্সটেনশন ভুল দেওয়া।
\item ছবি অন্য ফোল্ডারে থাকলেও শুধু ফাইলের নাম লেখা।
\item \texttt{alt} অ্যাট্রিবিউট না দেওয়া।
\item \texttt{width} ও \texttt{height} উল্টো দেওয়া।
\item ছবিতে লিংক করতে \texttt{<img>} ট্যাগ \texttt{<a>} ট্যাগের বাইরে রাখা।
\end{itemize}
\end{multicols}
