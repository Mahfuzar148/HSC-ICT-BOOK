\section{HTML-এ টেবিল}

% Composition pass 2026-10-04
ফলাফলের নাম, রোল ও GPA অনুচ্ছেদে লিখলে শিক্ষার্থীদের ফল তুলনা করা কঠিন।

\chFourMotivation{একই বৈশিষ্ট্যের তথ্য তুলনা $\rightarrow$ HTML টেবিল $\rightarrow$ সারি, কলাম ও শিরোনামের ঘরে তথ্য সাজানো $\rightarrow$ ফলাফল, রুটিন ও মূল্যতালিকা।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item HTML টেবিল কী এবং ওয়েবপেজে টেবিল কখন ব্যবহার করা হয় তা ব্যাখ্যা করতে।
\item \texttt{<table>}, \texttt{<tr>}, \texttt{<th>}, \texttt{<td>} ও \texttt{<caption>} ট্যাগের কাজ লিখতে।
\item টেবিলের সারি, কলাম ও ঘর দেখে শনাক্ত করতে।
\item \texttt{border}, \texttt{width}, \texttt{align}, \texttt{bgcolor}, \texttt{cellspacing} ও \texttt{cellpadding} অ্যাট্রিবিউট ব্যবহার করতে।
\item \texttt{rowspan} ও \texttt{colspan} ব্যবহার করে ঘরের বিস্তার করতে।
\item টেবিলের ভিতরে লেখা, বিন্যাস, লিংক, ছবি, তালিকা ও ভেতরের টেবিল ব্যবহার করতে।
\item চিত্র দেখে পরীক্ষার ধরনে HTML টেবিল কোড লিখতে।
\item পুরোনো অ্যাট্রিবিউট ও বর্তমান CSS রীতির পার্থক্য বুঝতে।
\end{itemize}
\end{learningbox}



\begin{keypointbox}{চিত্র ও কোড মিলিয়ে পড়বে}
ছকগুলো সারি, কলাম, বিস্তৃত ঘর ও বিষয়বস্তুর অবস্থান বোঝায়। ব্রাউজারে রং, মাপ ও ফাঁকের দৃশ্যমান রূপ ব্যবহৃত শৈলীর ওপর নির্ভর করে। ছবির ঘরে নাম বা চিহ্ন থাকলে তা ছবির স্থান বোঝানোর লেবেল; প্রকৃত ছবি দেখতে কোডের \texttt{src}-এ দেওয়া ফাইলটি থাকতে হবে। প্রতিটি উদাহরণে চিত্রের ঘরের অবস্থান ও কোডের ট্যাগ মিলিয়ে দেখবে।
\end{keypointbox}

\subsection{ভূমিকা}

কলেজের ফলাফল দেখার সময় একই সারিতে শিক্ষার্থীর নাম, রোল ও GPA পেলে তুলনা করা সহজ হয়। ক্লাসের রুটিন, পণ্যের মূল্যতালিকা, বেতনের হিসাব, বিলের সারসংক্ষেপ ও ভর্তি তালিকাতেও তথ্য সারি ও কলামে সাজানো থাকে। ওয়েবপেজে এই ধরনের তথ্য দেখাতে HTML-এর \textbf{টেবিল} ব্যবহার করা হয়। সৃজনশীল ও ব্যবহারিক প্রশ্নে টেবিলের সঙ্গে ছবি, লিংক, তালিকা এবং একাধিক সারি বা কলাম জুড়ে থাকা ঘরও দেওয়া হতে পারে।

\begin{keypointbox}{মূল ধারণা}
HTML টেবিল শেখার মূল বিষয় হলো: \textbf{সারি}, \textbf{কলাম}, \textbf{ঘর}, \textbf{ঘরের বিস্তার} এবং \textbf{ঘরের ভিতরে অন্য HTML এলিমেন্ট ব্যবহার}। এগুলো ঠিকমতো বুঝলে যেকোনো চিত্র দেখে টেবিল কোড লেখা সহজ হয়।
\end{keypointbox}

\begin{boardbox}{কখন টেবিল ব্যবহার করবে}
ফলাফল, রুটিন, মূল্যতালিকা, নম্বরবণ্টন, যোগাযোগের তালিকা, পণ্যের বিবরণ ও ভর্তি সময়সূচির মতো তথ্যের জন্য টেবিল উপযুক্ত। নতুন ওয়েবপেজের সামগ্রিক বিন্যাস তৈরিতে CSS ব্যবহার করবে। পুরোনো পরীক্ষার চিত্রে টেবিল দিয়ে পৃষ্ঠা সাজানো থাকলে সেই চিত্রের কাঠামো বিশ্লেষণ করতে হবে; সেটিকে বর্তমান ওয়েব নকশার অনুসরণীয় রীতি ভাববে না।
\end{boardbox}

\begin{keypointbox}{টেবিল বোঝার সহজ ভাষা}
একটি টেবিলকে খাতার ছকের মতো ভাবা যায়। প্রথমে ছকের সারি গুনবে, তারপর প্রতিটি সারিতে কয়টি ঘর আছে তা দেখবে। যে ঘরটি শিরোনাম হিসেবে কাজ করে সেটি সাধারণত \texttt{<th>} দিয়ে লেখা হয়, আর সাধারণ তথ্যের ঘর \texttt{<td>} দিয়ে লেখা হয়। একটি ঘর একাধিক কলাম জুড়ে থাকলে \texttt{colspan} এবং একাধিক সারি জুড়ে থাকলে \texttt{rowspan} ব্যবহার করা হয়।
\end{keypointbox}

\subsection{HTML টেবিল কী?}

\begin{definitionbox}{HTML টেবিল (HTML Table)}
HTML নথিতে তথ্যকে সারি ও কলামে সাজিয়ে দেখানোর কাঠামোকে HTML টেবিল বলে।
\end{definitionbox}

টেবিল মূলত \textbf{সারি–কলামে সাজানো তথ্য} দেখানোর জন্য ব্যবহার করা হয়। একটি টেবিল বুঝতে তিনটি বিষয় মনে রাখা দরকার: সারি, কলাম এবং ঘর।

\begin{center}
\begin{tabular}{|p{0.22\textwidth}|p{0.58\textwidth}|}
\hline
\textbf{অংশ} & \textbf{অর্থ} \\
\hline
সারি (Row) & আড়াআড়িভাবে সাজানো ঘরগুলোর সমষ্টি \\
\hline
কলাম (Column) & একই অবস্থানে লম্বালম্বিভাবে সাজানো ঘরগুলোর সমষ্টি \\
\hline
ঘর (Cell) & সারি ও কলাম মিলিত হয়ে যে ঘর তৈরি হয় \\
\hline
\end{tabular}
\end{center}

\begin{examplebox}{ধারণা}
\noindent
\begin{minipage}[t]{0.56\textwidth}
\vspace{0pt}
\begin{center}
\begin{tabular}{|c|c|c|}
\hline
\textbf{Roll} & \textbf{Name} & \textbf{GPA} \\
\hline
101 & Rafi & 5.00 \\
\hline
102 & Sakib & 4.75 \\
\hline
\end{tabular}
\end{center}
এখানে Roll, Name ও GPA হলো শিরোনামের ঘর; 101, Rafi ও 5.00 হলো তথ্যের ঘর। আড়াআড়িভাবে সাজানো ঘরগুলো মিলে একটি সারি হয়; একই অবস্থানে লম্বালম্বিভাবে সাজানো ঘরগুলো একটি কলাম। রেখা ঘরগুলোর সীমানা দেখায়; রেখা নিজে সারি নয়।

\textbf{চিত্র ব্যাখ্যা:} চিত্রে HTML টেবিলের সারি, কলাম এবং ঘর আলাদা লেবেল দিয়ে দেখানো হয়েছে। সারি হলো আড়াআড়িভাবে সাজানো ঘরগুলোর সমষ্টি, কলাম হলো লম্বালম্বি অংশ এবং সারি–কলাম মিলিত হয়ে ঘর তৈরি করে।
\end{minipage}

\subsection{\texttt{<table>} ট্যাগ}

\begin{definitionbox}{}
HTML নথিতে টেবিলের ধারক তৈরি করে \texttt{<table>} ট্যাগ।
\end{definitionbox}

টেবিলের সব সারি ও ঘর এই ট্যাগের ভিতরে লেখা হয়।

% Image layout revision: related content in two fixed columns.
\par\noindent
\begin{minipage}[t]{.52\linewidth}\vspace{0pt}\small
\begin{chFourWholeExample}{কোড}
\begin{verbatim}
<!DOCTYPE html>
<html>
<head>
  <title>HTML Table Example</title>
</head>
<body>
  <table border="1" cellspacing="0" cellpadding="6">
    <tr>
      <td>Data</td>
    </tr>
  </table>
</body>
</html>
\end{verbatim}
\end{chFourWholeExample}
\end{minipage}\hfill
\begin{minipage}[t]{.44\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}ব্রাউজারে ফল}\par
\begin{tabular}{|l|}\hline Data \\\hline\end{tabular}\par\smallskip
\texttt{table} পুরো ছক ধরে; \texttt{tr} একটি সারি ও \texttt{td} তথ্যের ঘর তৈরি করে। এখানে একটি সারিতে একটি ঘর আছে।\par
\texttt{title}-এর লেখা ব্রাউজারের ট্যাবে থাকে; ছকটি \texttt{body}-তে থাকায় পেজে দেখা যায়।
\end{minipage}\par\smallskip


\subsection{\texttt{<tr>} ট্যাগ}

\begin{definitionbox}{}
HTML টেবিলের একটি সারি তৈরি করার জন্য \texttt{<tr>} ট্যাগ ব্যবহার করা হয়।
\end{definitionbox}

প্রতিটি \texttt{<tr>} ট্যাগের ভিতরে এক বা একাধিক \texttt{<th>} অথবা \texttt{<td>} লেখা হয়।

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}
\begin{verbatim}
<table border="1" cellspacing="0"
       cellpadding="6">
  <tr>
    <td>Data 1</td>
    <td>Data 2</td>
  </tr>
</table>
\end{verbatim}
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
\textbf{সরল ফল}\par
\begin{tabular}{|l|l|}\hline
Data 1 & Data 2 \\\hline
\end{tabular}
\end{minipage}
\chFourWhy{সারির ভেতরে দুটি td থাকায় পাশাপাশি দুটি তথ্যের ঘর তৈরি হয়েছে।}
\end{chFourWholeExample}

\subsection{\texttt{<th>} ট্যাগ}

\begin{definitionbox}{}
HTML টেবিলের শিরোনামের ঘর তৈরি করে \texttt{<th>} ট্যাগ।
\end{definitionbox}

ব্রাউজার সাধারণত \texttt{<th>} ট্যাগের লেখা গাঢ় করে ও মাঝখানে দেখায়।

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}
\begin{verbatim}
<table border="1" cellspacing="0"
       cellpadding="6">
  <tr>
    <th>Name</th>
    <th>Roll</th>
  </tr>
</table>
\end{verbatim}
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
\textbf{সরল ফল}\par
\begin{tabular}{|l|l|}\hline
\textbf{Name} & \textbf{Roll} \\\hline
\end{tabular}
\end{minipage}
\chFourWhy{th দুটি শিরোনামের ঘর চিহ্নিত করেছে; এগুলো শুধু গাঢ় লেখা নয়।}
\end{chFourWholeExample}

\subsection{\texttt{<td>} ট্যাগ}

\begin{definitionbox}{}
HTML টেবিলের তথ্যের ঘর তৈরি করে \texttt{<td>} ট্যাগ।
\end{definitionbox}

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}
\begin{verbatim}
<table border="1" cellspacing="0"
       cellpadding="6">
  <tr>
    <td>Rafi</td>
    <td>101</td>
  </tr>
</table>
\end{verbatim}
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
\textbf{সরল ফল}\par
\begin{tabular}{|l|l|}\hline
Rafi & 101 \\\hline
\end{tabular}
\end{minipage}
\chFourWhy{td দুটি তথ্যের ঘর তৈরি করেছে; ঘরের ক্রমেই Rafi ও 101 পাশাপাশি আছে।}
\end{chFourWholeExample}

\subsection{টেবিলের কোডে সারি ও ঘরের সম্পর্ক}

টেবিলের কোডে সারি ও ঘরের অবস্থান অনুযায়ী তথ্য সাজাতে হয়। একটি সারি শুরু করতে \texttt{<tr>} লিখবে। সেই সারির ঘরগুলো \texttt{<th>} অথবা \texttt{<td>} দিয়ে একই সারির ভেতরে লিখবে। ঘরের ক্রম বা বিস্তার বদলালে ফলের কাঠামোও বদলে যায়। এটি কোড বোঝার নিয়ম; ব্রাউজারের অভ্যন্তরীণ কাজের পূর্ণ বর্ণনা নয়।

\begin{keypointbox}{মনে রাখার নিয়ম}
টেবিলের ভিতরে সরাসরি লেখা রাখা উচিত নয়। টেবিলের বিষয়বস্তু সাধারণত ঘরের ভিতরে থাকে। তাই সঠিক কাঠামো হলো:
\texttt{<table>} $\rightarrow$ \texttt{<tr>} $\rightarrow$ \texttt{<th>} বা \texttt{<td>} $\rightarrow$ বিষয়বস্তু।
\end{keypointbox}

\hline
২ & প্রতিটি সারির জন্য \texttt{<tr>} লেখা \\
\hline
৩ & শিরোনাম হলে \texttt{<th>}, তথ্যের ঘর হলে \texttt{<td>} লেখা \\
\hline
৪ & প্রয়োজনীয় শেষের ট্যাগ লেখা \\

\subsection{টেবিল তৈরির আগে কী দেখবে?}

চিত্র দেখে বা প্রশ্ন পড়ে টেবিল কোড লেখার আগে সরাসরি কোড শুরু না করে কয়েকটি বিষয় ঠিক করতে হয়। এতে ভুল কম হয় এবং \texttt{rowspan}/\texttt{colspan} সহজে বোঝা যায়।

\begin{enumerate}
\item টেবিলের শিরোনাম বা ক্যাপশন আছে কি না দেখবে।
\item মোট কয়টি দৃশ্যমান কলাম আছে তা গুনবে।
\item মোট কয়টি সারি আছে তা গুনবে।
\item কোন ঘর শিরোনামের এবং কোনটি সাধারণ তথ্যের, তা আলাদা করবে।
\item কোনো ঘর একাধিক কলাম জুড়ে আছে কি না দেখবে।
\item কোনো ঘর একাধিক সারি জুড়ে আছে কি না দেখবে।
\item ঘরের ভিতরে লেখা, ছবি, লিংক, তালিকা বা ভেতরের টেবিল আছে কি না দেখবে।
\end{enumerate}

\begin{keypointbox}{পরীক্ষার কৌশল}
কোনো চিত্র জটিল মনে হলে প্রথমে পেনসিল দিয়ে সারির সীমানা ও কলামের সীমানা আলাদা করে ভাববে। এরপর বিস্তৃত ঘর বাদ দিয়ে মূল ছক ধরবে। মূল ছক বুঝলে \texttt{rowspan} ও \texttt{colspan} বসানো সহজ হয়।
\end{keypointbox}

\subsection{টেবিলের অ্যাট্রিবিউট}

পুরোনো HTML উদাহরণে টেবিল ও ঘরের বিন্যাসের জন্য কয়েকটি অ্যাট্রিবিউট বারবার ব্যবহৃত হয়। \texttt{border}, \texttt{width}, \texttt{align}, \texttt{cellspacing} ও \texttt{cellpadding} টেবিলের বিন্যাসে প্রভাব ফেলে। ঘরের ক্ষেত্রে \texttt{align}, \texttt{valign}, \texttt{bgcolor}, \texttt{width} ও \texttt{height} দৃশ্যমান বিন্যাস নিয়ন্ত্রণ করে। এগুলো পুরোনো দৃশ্যমান বিন্যাসের অ্যাট্রিবিউট; বর্তমান HTML মান অনুসারে নতুন পৃষ্ঠায় CSS ব্যবহার করবে। \texttt{rowspan} ও \texttt{colspan} ঘরের বিস্তার নির্ধারণ করে এবং বর্তমান HTML-এও বৈধ।

\begin{center}
\begin{tabular}{|p{0.20\textwidth}|p{0.42\textwidth}|p{0.25\textwidth}|}
\hline
\textbf{অ্যাট্রিবিউট} & \textbf{কাজ} & \textbf{উদাহরণ} \\
\hline
\texttt{border} & টেবিল ও ঘরের চারপাশে রেখা দেখায় & \texttt{border="1"} \\
\hline
\texttt{width} & টেবিল বা ঘরের প্রস্থ পিক্সেল বা শতাংশে নির্ধারণ করে & \texttt{width="400"} \\
\hline
\texttt{align} & টেবিলের অবস্থান বা ঘরের লেখার বাম/ডান/মধ্য বিন্যাস নির্ধারণ করে & \texttt{align="center"} \\
\hline
\texttt{cellspacing} & পাশাপাশি ঘরের সীমানার মধ্যবর্তী ফাঁক নির্ধারণ করে & \texttt{cellspacing="0"} \\
\hline
\texttt{cellpadding} & ঘরের বিষয়বস্তু ও রেখার মাঝের ভিতরের ফাঁক নির্ধারণ করে & \texttt{cellpadding="5"} \\
\hline
\texttt{bgcolor} & টেবিল/ঘরের পটভূমির রং দেয় & \texttt{bgcolor="yellow"} \\
\hline
\end{tabular}
\end{center}

\begin{keypointbox}{মনে রাখবে}
CSS বা অন্য শৈলী না থাকলে শুধু টেবিল তৈরির ট্যাগ লিখে রেখা দেখা যায় না। পুরোনো উদাহরণে \texttt{border="1"} ও \texttt{cellspacing="0"} দিয়ে রেখা এবং ঘরের মধ্যবর্তী ফাঁক দেখানো হয়েছে। পুরোনো \texttt{width="75\%"} টেবিলের ধারক এলাকার সাপেক্ষে প্রস্থ বোঝায়; \texttt{width="400"} পিক্সেলে প্রস্থের নির্দেশ দেয়। CSS-এ একই মাপ লিখতে \texttt{width: 400px;} ব্যবহার করবে। টেবিলের তথ্য বেশি চওড়া হলে ঘোষিত প্রস্থের চেয়েও বেশি জায়গা লাগতে পারে।
\end{keypointbox}

\subsection{\texttt{width} অ্যাট্রিবিউট}

পিক্সেলে দেওয়া প্রস্থ নির্দিষ্ট মাপ বোঝায়, তাই \texttt{width="400"} দিলে টেবিল প্রায় 400 পিক্সেল জায়গা নেবে। শতাংশে দেওয়া প্রস্থ ধারক এলাকার সঙ্গে বদলায়; \texttt{width="75\%"} দিলে ধারক এলাকার 75\% জায়গা ব্যবহার করবে। ঘরের \texttt{width} দিলে নির্দিষ্ট কলাম তুলনামূলক বড় বা ছোট করা যায়।

\subsection{\texttt{valign} অ্যাট্রিবিউট}

কোনো ঘরের উচ্চতা বেশি হলে বিষয়বস্তু উপরে, মাঝখানে বা নিচে রাখা দরকার হতে পারে। লাইনবিরতি দিয়ে বিষয়বস্তু নিচে সরানো অবস্থান নির্ধারণের নির্ভরযোগ্য পদ্ধতি নয়। পুরোনো HTML-এ এই অবস্থান নির্ধারণে \texttt{valign="top"}, \texttt{valign="middle"} বা \texttt{valign="bottom"} ব্যবহার করা ভালো।

\subsection{\texttt{cellspacing} অ্যাট্রিবিউট}

\begin{definitionbox}{}
পাশাপাশি ঘরের সীমানার মধ্যবর্তী ফাঁক নির্ধারণ করে \texttt{cellspacing} অ্যাট্রিবিউট।
\end{definitionbox}

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.54\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<table border="1" cellspacing="0">
  <tr>
    <th>Name</th>
    <th>Class</th>
  </tr>
  <tr>
    <td>Rafi</td>
    <td>XI</td>
  </tr>
</table>
\end{verbatim}
\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.42\linewidth}\vspace{0pt}\small
\begin{keypointbox}{সহজ কথা}
ব্রাউজার অনেক সময় প্রাথমিকভাবে ঘরের মাঝে সামান্য ফাঁক রাখে। \texttt{cellspacing="0"} দিলে ঘরের মাঝখানের ফাঁক কমে যায় বা রেখাগুলো একসঙ্গে লেগে থাকে। আবার \texttt{cellspacing="10"} দিলে দুই ঘরের রেখার মাঝে 10 পিক্সেল দূরত্ব তৈরি হয়।
\end{keypointbox}
\end{minipage}\par


\subsection{\texttt{cellpadding} অ্যাট্রিবিউট}

\begin{definitionbox}{}
ঘরের বিষয়বস্তু ও সীমানার মধ্যকার ভেতরের ফাঁক নির্ধারণ করে \texttt{cellpadding} অ্যাট্রিবিউট।
\end{definitionbox}

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.54\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<table border="1" cellpadding="10">
  <tr>
    <th>Subject</th>
    <th>GPA</th>
  </tr>
  <tr>
    <td>Bangla</td>
    <td>5.00</td>
  </tr>
</table>
\end{verbatim}
\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.42\linewidth}\vspace{0pt}\small
\begin{boardbox}{বহুনির্বাচনির গুরুত্বপূর্ণ তথ্য}
\texttt{cellpadding="5"} অর্থ ঘরের বিষয়বস্তু এবং ঘরের রেখার মাঝখানে 5 পিক্সেল ভিতরের ফাঁকা জায়গা রাখা। অনেক ব্রাউজারে প্রাথমিক ঘরের ফাঁক খুব কম থাকে; তাই লেখা রেখার সাথে লেগে গেলে \texttt{cellpadding} বাড়ালে টেবিল পড়তে সহজ হয়।
\end{boardbox}
\end{minipage}\par

\subsection{\texttt{<td>} ও \texttt{<th>} ট্যাগের অ্যাট্রিবিউট}

ঘরের উপর আলাদা নিয়ন্ত্রণ দেওয়ার জন্য \texttt{<td>} বা \texttt{<th>} ট্যাগের ভিতরে অ্যাট্রিবিউট ব্যবহার করা যায়।

এই অংশের টেবিলে \texttt{<th>} ও \texttt{<td>} ঘরের জন্য আলাদা অ্যাট্রিবিউট দেখানো হয়েছে। একই টেবিলের সব ঘর একরকম নাও হতে পারে; কোনো ঘর মাঝখানে সাজানো, কোনো ঘরের পটভূমি সবুজ, কোনো ঘর বেশি চওড়া, আবার কোনো ঘর একাধিক সারি বা কলাম জুড়ে থাকতে পারে।

\begin{center}
\begin{tabular}{|p{0.18\textwidth}|p{0.46\textwidth}|p{0.25\textwidth}|}
\hline
\textbf{অ্যাট্রিবিউট} & \textbf{কাজ} & \textbf{উদাহরণ} \\
\hline
\texttt{align} & ঘরের লেখা বাম/মাঝ/ডানে রাখে & \texttt{align="center"} \\
\hline
\texttt{valign} & ঘরের লেখা উপরে, মাঝখানে বা নিচে রাখে & \texttt{valign="top"} \\
\hline
\texttt{bgcolor} & ঘরের পটভূমির রং দেয় & \texttt{bgcolor="green"} \\
\hline
\texttt{width} & ঘরের প্রস্থ নির্ধারণ করে & \texttt{width="150"} \\
\hline
\texttt{height} & ঘরের উচ্চতা নির্ধারণ করে & \texttt{height="60"} \\
\hline
\texttt{colspan} & ঘরকে একাধিক কলাম জুড়ে দেয় & \texttt{colspan="2"} \\
\hline
\texttt{rowspan} & ঘরকে একাধিক সারি জুড়ে দেয় & \texttt{rowspan="3"} \\

\begin{keypointbox}{রঙের মান}
পুরোনো \texttt{bgcolor}-এ \texttt{yellow}, \texttt{green}, \texttt{silver} বা \texttt{\#FFFF00}-এর মতো রঙের মান দেখা যায়। CSS-এর \texttt{rgb(...)} রূপকে \texttt{bgcolor}-এর মান ধরে লিখবে না; সেটি \texttt{background-color} বৈশিষ্ট্যে ব্যবহার করবে।
\end{keypointbox}

\subsection{খালি ঘর ও \texttt{\&nbsp;}}

খালি ঘর লেখার জন্য \texttt{<td></td>} বৈধ। ঘরের রেখা, মাপ ও ফাঁক নির্ভর করে টেবিলের শৈলীর ওপর। পুরোনো উদাহরণে ঘরের ভেতরে একটি অবিচ্ছেদ্য ফাঁক রাখতে \texttt{\&nbsp;} ব্যবহার করা হয়।

\texttt{\&nbsp;} কোনো টেবিলের মাপ ঠিক করার নিশ্চয়তা নয়; এটি একটি ফাঁকের অক্ষর যোগ করে। \texttt{<br>} একটি লাইন বিরতি তৈরি করে। নতুন পৃষ্ঠায় ঘরের মাপ, padding, রেখা এবং প্রয়োজনে \texttt{empty-cells} CSS দিয়ে নির্ধারণ করবে। খালি ঘর, ফাঁকের অক্ষর থাকা ঘর ও অনুপস্থিত ঘর এক অবস্থা নয়।

\subsection{\texttt{<caption>} ট্যাগ}

\begin{definitionbox}{}
HTML টেবিলের শিরোনাম প্রদর্শনের জন্য \texttt{<caption>} ট্যাগ ব্যবহার করা হয়।
\end{definitionbox}

\texttt{<caption>} ট্যাগ সবসময় \texttt{<table>} ট্যাগের ভিতরে এবং প্রথম সারির আগে লেখা উচিত। এতে ব্রাউজার সাধারণত টেবিলের উপরে শিরোনাম দেখায়। ফলাফল, রুটিন, ভর্তি তালিকা বা পণ্যের তথ্যের টেবিলে ক্যাপশন দিলে টেবিলের উদ্দেশ্য দ্রুত বোঝা যায়। পুরোনো HTML কোডে ক্যাপশনের অবস্থান বোঝাতে \texttt{align="bottom"} ধরনের পুরোনো অ্যাট্রিবিউট দেখা যেতে পারে, তবে বর্তমান HTML-এ ক্যাপশন অবস্থান CSS দিয়ে নিয়ন্ত্রণ করা ভালো।

\subsection{\texttt{colgroup} ট্যাগ}

\begin{definitionbox}{\texttt{<colgroup>} ট্যাগ}
টেবিলের এক বা একাধিক কলামের জন্য একসঙ্গে শৈলী বা বৈশিষ্ট্য নির্ধারণ করতে \texttt{<colgroup>} ট্যাগ ব্যবহার করা হয়।
\end{definitionbox}

\texttt{colgroup} ব্যবহার করলে কোনো নির্দিষ্ট কলাম বা কলামের দলকে আলাদা রং বা শৈলী দেওয়া যায়। এই এলিমেন্ট সারি-দলের আগে লেখা হয়। ফলে কলামের শৈলী ও সারির তথ্য আলাদাভাবে নির্ধারণ করা যায়।

একাধিক কলামে একই শৈলী প্রয়োগ করতে \texttt{<col span="2">} ব্যবহার করা যায়। যেমন প্রথম দুই কলাম এক রং এবং পরের দুই কলাম আরেক রং করতে চাইলে \texttt{span} অ্যাট্রিবিউট সহায়ক। প্রতিটি কলামের জন্য আলাদা শৈলী চাইলে আলাদা আলাদা \texttt{<col>} ট্যাগ লেখা হয়।

  </colgroup>

    <th>Title</th>
    <th>Author</th>
  </tr>
  <tr>
    <td>ICT HSC</td>
    <td>M. Hakim</td>

\textbf{\color{chFourTeal}চিত্র ব্যাখ্যা:} চিত্রে \texttt{<colgroup>} ব্যবহার করে নির্দিষ্ট কলামের দলে শৈলী দেওয়ার ধারণা দেখানো হয়েছে। এতে একাধিক কলাম একসঙ্গে রং বা শৈলী করা যায়।

\subsection{CSS দিয়ে টেবিলের বিন্যাস}

\begin{keypointbox}{বর্তমান রীতি}
এইচএসসি তথ্য ও যোগাযোগ প্রযুক্তির মৌলিক HTML অংশে \texttt{border}, \texttt{align}, \texttt{bgcolor}, \texttt{cellpadding}, \texttt{cellspacing}, \texttt{width} অ্যাট্রিবিউট শেখা দরকার। তবে পেশাগত ওয়েব নকশায় এগুলোর অনেকগুলো কাজ CSS দিয়ে করা হয়। যেমন: \texttt{border}, \texttt{border-collapse}, \texttt{border-spacing}, \texttt{padding}, \texttt{text-align}, \texttt{background-color}, \texttt{width}।
\end{keypointbox}

বামে \texttt{style}-এ ছকের মাপ, রেখা, ঘরের ভেতরের ফাঁক ও শিরোনামের রং নির্ধারণ করা হয়েছে। ডানে সেই ছকের বিষয়বস্তু আছে। দুই অংশ একই HTML ফাইলে পরপর লিখবে।
\end{minipage}\par\smallskip


\subsection{টেবিলের অর্থ ও পাঠযোগ্যতা}

এইচএসসির পাঠ্যবইয়ের পুরোনো রীতির কোডে \texttt{border}, \texttt{cellpadding}, \texttt{cellspacing}, \texttt{align}, \texttt{bgcolor} অ্যাট্রিবিউট দেখা যায়। এগুলো ব্যবহারিক পরীক্ষার জন্য জানা দরকার। তবে বর্তমান HTML-এ টেবিলের দৃশ্যমান শৈলী সাধারণত CSS দিয়ে করা হয়। টেবিলের তথ্যের অর্থ পরিষ্কার রাখতে শিরোনামের ঘরে \texttt{<th>} ব্যবহার করা উচিত এবং টেবিলের শিরোনাম থাকলে \texttt{<caption>} ব্যবহার করা উচিত।

\begin{boardbox}{ভালো টেবিল কোডের বৈশিষ্ট্য}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item টেবিল তথ্যের সারি–কলাম আকারে যৌক্তিক হবে।
\item শিরোনাম ঘরে \texttt{<th>} ব্যবহার করা হবে।
\item ক্যাপশন থাকলে \texttt{<caption>} টেবিলের শুরুতে লেখা হবে।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item ছবি থাকলে \texttt{alt} অ্যাট্রিবিউট ব্যবহার করা হবে।
\item লিংকের লেখায় তার গন্তব্য বা কাজ বোঝা যাবে।
\item ট্যাগ ইনডেন্টেশন পরিষ্কার হলে কোড পড়া সহজ হবে।
\end{itemize}
\end{minipage}\par

\end{boardbox}

\subsection{চিত্র দেখে টেবিল কোড লেখার ধাপ}

পরীক্ষায় চিত্র দেখে টেবিল কোড লিখতে হলে আগে টেবিলটি বিশ্লেষণ করতে হবে।

\begin{enumerate}
\item মোট কয়টি সারি আছে দেখো।
\item সর্বোচ্চ কয়টি কলাম আছে দেখো।
\item শিরোনামের ঘর আছে কি না দেখো।
\item কোনো ঘরের বিস্তার আছে কি না দেখো।
\item ঘরের বিস্তার হলে সেটি \texttt{rowspan} নাকি \texttt{colspan} তা নির্ধারণ করো।
\item রেখা, প্রস্থ, align, cellspacing, cellpadding দরকার কি না দেখো।
\item ছবি, লিংক, তালিকা বা ভেতরের টেবিল আছে কি না দেখো।
\item তারপর \texttt{<table>} কাঠামো লিখে সারি অনুযায়ী ঘর বসাও।
\end{enumerate}

\subsection{চিত্র থেকে টেবিলের কোড যাচাই}

ব্যবহারিক পরীক্ষা বা সৃজনশীল প্রশ্নে টেবিলের চিত্র দেওয়া থাকলে নিচের যাচাইয়ের ধাপগুলো অনুসরণ করবে। সাধারণ টেবিল, বিস্তৃত ঘর এবং ছবি, লিংক, তালিকা বা ভেতরের টেবিল থাকা ছকের ক্ষেত্রে এগুলো কাজে লাগে।

\begin{enumerate}
\item টেবিলের নাম বা ক্যাপশন আছে কি না দেখবে।
\item শিরোনামের সারি আছে কি না দেখবে।
\item দৃশ্যমান কলাম সংখ্যা গুনবে।
\item দৃশ্যমান সারি সংখ্যা গুনবে।
\item বিস্তৃত ঘর থাকলে আগে \texttt{rowspan} না \texttt{colspan} তা ঠিক করবে।
\item যে ঘরে ছবি আছে সেখানে \texttt{<img>} এবং \texttt{alt} লিখবে।
\item যে ঘরে লিংক আছে সেখানে \texttt{<a href="">} ব্যবহার করবে।
\item যে ঘরে তালিকা আছে সেখানে \texttt{<ol>} বা \texttt{<ul>} ব্যবহার করবে।
\item ঘরের ভেতর আরেকটি টেবিল থাকলে তার কোড আলাদা ইনডেন্টেশন দিয়ে লিখবে।
\item শেষে সব \texttt{<table>}, \texttt{<tr>}, \texttt{<td>} ট্যাগ শেষ হয়েছে কি না মিলিয়ে দেখবে।
\end{enumerate}

\begin{keypointbox}{শেষ মুহূর্তের পরীক্ষা}
কোড লেখা শেষ হলে প্রতিটি সারি আলাদা করে দেখবে: সারির মোট কলামসংখ্যা চিত্রের সাথে মিলছে কি না। \texttt{colspan="2"} মানে ওই ঘর দুই কলামের জায়গা নিয়েছে, তাই সেই সারিতে একটি ঘর কম থাকবে। \texttt{rowspan="2"} মানে পরের সারিতে ওই কলামের জন্য নতুন ঘর লিখতে হবে না।
\end{keypointbox}

\textbf{প্রশ্ন}\\
ক. HTML টেবিল কী?\\

\end{practicebox}

\subsection{সাধারণ ভুল}

\begin{mistakebox}{সাধারণ ভুল}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{<table>} ট্যাগ শেষ না করা।
\item \texttt{<tr>} ছাড়া সরাসরি \texttt{<td>} লেখা।
\item শিরোনামের ঘরে \texttt{<td>} ব্যবহার করা, যেখানে \texttt{<th>} ভালো।
\item \texttt{<td>}-এর বানান ভুল করে \texttt{<tb>} লেখা।
\item \texttt{cellpadding} ও \texttt{cellspacing} গুলিয়ে ফেলা।
\item \texttt{rowspan} ও \texttt{colspan} উল্টো ব্যবহার করা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item CSS দিয়ে রেখা দেওয়া থাকলেও \texttt{border} অ্যাট্রিবিউট না থাকায় রেখা দেখা যাবে না ভাবা।
\item টেবিল সারি ও কলাম সংখ্যা চিত্রের সাথে না মেলানো।
\item \texttt{<caption>} ভুল জায়গায় লেখা। ক্যাপশন সাধারণত \texttt{<table>} ট্যাগের ভিতরে, প্রথম সারির আগে লেখা হয়।
\item টেবিলের ভিতরে ছবি, লিংক বা তালিকা রাখার সময় ঘরের শুরুর ও শেষের ট্যাগের মধ্যে সংশ্লিষ্ট কোড না লেখা।
\item ছবির \texttt{src} পথের সঙ্গে ফাইলের অবস্থান না মেলানো; একই ফোল্ডারে রাখা বাধ্যতামূলক নয়।
\item ভেতরের টেবিলের শেষের ট্যাগ ঠিকমতো না দেওয়া।
\end{itemize}
\end{minipage}\par

\end{mistakebox}
