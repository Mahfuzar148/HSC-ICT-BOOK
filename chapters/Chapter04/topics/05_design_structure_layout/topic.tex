\section{ওয়েবপেজের কাঠামো ও বিন্যাস (Structure and Layout)}

% Composition pass 2026-10-04
একই কলেজের ওয়েবপেজ কম্পিউটারে প্রশস্ত, কিন্তু মোবাইলে সরু পর্দায় দেখা হবে।

\chFourMotivation{উভয় পর্দায় পাঠযোগ্যতা $\rightarrow$ বিন্যাস ও রেসপনসিভ ডিজাইন $\rightarrow$ নমনীয় মাপ ও উপযুক্ত CSS-এ অংশ সাজানো $\rightarrow$ মেনু, ছবি ও তথ্য সহজে ব্যবহার করা।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item ওয়েবপেজের বিন্যাস, ওয়্যারফ্রেম ও রেসপনসিভ ডিজাইন ব্যাখ্যা করতে।
\item ফিক্সড, ফ্লুইড ও রেসপনসিভ বিন্যাস আলাদা করতে।
\item হেডার, মেনু, মূল বিষয়বস্তু, সাইডবার ও ফুটারের কাজ বলতে।
\item একটি সহজ ওয়েবপেজের বিন্যাস পরিকল্পনা বা ওয়্যারফ্রেম আঁকতে।
\item ব্যবহারিক কাজের ওয়েবপেজ তৈরির আগে বিন্যাস পরিকল্পনা করতে।
\end{itemize}
\end{learningbox}

\subsection{ওয়েবপেজের বিন্যাস (Webpage Layout)}

\begin{definitionbox}{}
ওয়েবপেজের হেডার, মেনু, মূল বিষয়বস্তু, ছবি, সারণি, সাইডবার ও ফুটারের অবস্থান নির্ধারণের বিন্যাসকে \textbf{ওয়েবপেজ লেআউট} বলে।
\end{definitionbox}

সঠিক বিন্যাস পেজকে পরিষ্কার, পড়ার উপযোগী ও ব্যবহারবান্ধব করে। ব্যবহারিক কাজে পূর্ণাঙ্গ ওয়েবপেজ তৈরির আগে বিন্যাস বুঝে নিলে কোড লেখা সহজ হয়।

\begin{center}

\hline

\hline
Sidebar & বিজ্ঞপ্তি, গুরুত্বপূর্ণ লিংক, বিজ্ঞাপন বা অতিরিক্ত তথ্য রাখে \\

Footer & কপিরাইট, যোগাযোগের তথ্য, ঠিকানা, ইমেইল ও সামাজিক যোগাযোগের লিংক রাখে \\
\hline

\subsection{ওয়্যারফ্রেম (Wireframe)}

\begin{definitionbox}{}
ওয়েবপেজের অংশগুলোর অবস্থান ঘর বা খসড়া চিত্রে দেখানোর প্রাথমিক নকশাকে \textbf{ওয়্যারফ্রেম} বলে। কোড লেখার আগে এই নকশা করা হয়।
\end{definitionbox}

\begin{center}
\fbox{\parbox{0.86\textwidth}{
\centering
\textbf{Simple Webpage Wireframe}\\[5pt]
\fbox{\parbox{0.74\textwidth}{\centering Header}}\\[4pt]
\fbox{\parbox{0.74\textwidth}{\centering Navigation/Menu}}\\[4pt]
\fbox{\parbox{0.46\textwidth}{\centering Main Content\\Text, Image, Table}}
\hspace{0.03\textwidth}
\fbox{\parbox{0.22\textwidth}{\centering Sidebar\\Notice/Links}}\\[4pt]
\fbox{\parbox{0.74\textwidth}{\centering Footer}}
}}
\end{center}

\begin{keypointbox}{ওয়্যারফ্রেম কেন দরকার?}
\begin{itemize}
\item কোড লেখার আগে পেজের পরিকল্পনা পরিষ্কার হয়।
\item কোন বিষয়বস্তু কোথায় থাকবে তা বোঝা যায়।
\item হেডার, মেনু, মূল বিষয়বস্তু ও ফুটার আলাদা করা সহজ হয়।
\item চিত্র দেখে HTML কোড লেখার সময় ভুল কম হয়।
\item বড় প্রকল্পে ডিজাইনার ও ডেভেলপার একই পরিকল্পনা অনুসরণ করতে পারেন।
\end{itemize}
\end{keypointbox}

\subsection{রেসপনসিভ ডিজাইন (Responsive Design)}

\begin{definitionbox}{}
পর্দার আকার অনুযায়ী বিন্যাস বদলে ওয়েবপেজকে পাঠযোগ্য ও ব্যবহারযোগ্য রাখার নকশাপদ্ধতিকে \textbf{রেসপনসিভ ডিজাইন} বলে। এতে একই পেজ ডেস্কটপ, ট্যাবলেট ও মোবাইলে ব্যবহার করা যায়।
\end{definitionbox}

রেসপনসিভ ডিজাইনে নমনীয় প্রস্থ, পাঠযোগ্য অক্ষর, উপযুক্ত মাপের ছবি ও মোবাইলে ব্যবহারযোগ্য মেনু রাখা হয়। viewport মেটা ট্যাগ ব্রাউজারকে পর্দা অনুযায়ী প্রদর্শনের নির্দেশনা দেয়। শুধু এই ট্যাগ যথেষ্ট নয়; CSS দিয়ে অংশগুলোর মাপ ও অবস্থানও নির্ধারণ করতে হয়।

\begin{examplebox}{কোড}
\begin{verbatim}
<meta name="viewport" content="width=device-width, initial-scale=1.0">
\end{verbatim}
\end{examplebox}

\subsection{ওয়েবপেজের বিন্যাসের নীতি}

\begin{keypointbox}{নকশার নীতি}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \textbf{পার্থক্য ফুটিয়ে তোলা (Contrast):} গুরুত্বপূর্ণ অংশ আলাদা করে দেখানো।
\item \textbf{সারিবিন্যাস (Alignment):} লেখা ও উপাদান সুশৃঙ্খল রাখা।
\item \textbf{একই রীতির পুনর্ব্যবহার (Repetition):} একই শৈলী ব্যবহার করে সামঞ্জস্য রাখা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \textbf{নৈকট্য (Proximity):} সম্পর্কিত বিষয়বস্তু কাছাকাছি রাখা।
\item \textbf{ফাঁকা জায়গা (Whitespace):} বিষয়বস্তুর মাঝে পর্যাপ্ত ফাঁকা জায়গা রাখা।
\item \textbf{পাঠযোগ্যতা (Readability):} লেখা যেন সহজে পড়া যায়।
\end{itemize}
\end{minipage}\par

\end{keypointbox}

\subsection{সমস্যা ও সমাধান}

\begin{practicebox}
\textbf{সমস্যা:} একটি কলেজের ওয়েবপেজের বিন্যাস পরিকল্পনা করো। এতে হেডার, মেনু, বিজ্ঞপ্তি, বিষয়ের তালিকা, ফলের সারণি ও ফুটার থাকবে।

\textbf{সমাধানের পরিকল্পনা:}
\begin{enumerate}
\item হেডার অংশে \texttt{<div>}-এর ভেতরে \texttt{<h1>} দিয়ে কলেজের নাম লিখবে।
\item মেনু অংশে \texttt{<a>} দিয়ে Home, Notice, Result ও Contact লিংক লিখবে।
\item মূল বিষয়বস্তুর অংশে বিজ্ঞপ্তির অনুচ্ছেদ লিখবে।
\item বিষয়ের তালিকার জন্য \texttt{<ul>} ও \texttt{<li>} ব্যবহার করবে।
\item ফলের সারণির জন্য \texttt{<table>}, \texttt{<tr>}, \texttt{<th>}, \texttt{<td>} ব্যবহার করবে।
\item ফুটার অংশে কপিরাইট বা ইমেইল লিখবে।
\end{enumerate}
\end{practicebox}

\begin{boardbox}{পরীক্ষার জন্য মনে রাখবে}
ওয়েবপেজ তৈরির ব্যবহারিক কাজে কোড লেখার আগে বিন্যাসের অংশগুলো শনাক্ত করো। হেডার, মেনু, মূল বিষয়বস্তু, সারণি/তালিকা/ছবি ও ফুটার আলাদা করলে কোড সুশৃঙ্খল হয়। এতে চাহিদার চিত্রের সঙ্গে ফল মেলানো সহজ হয়।
\end{boardbox}

\begin{mistakebox}{সাধারণ ভুল}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item বিন্যাস পরিকল্পনা না করে সরাসরি কোড লেখা।
\item হেডার, মেনু, মূল বিষয়বস্তু ও ফুটার আলাদা না করা।
\item সব বিষয়বস্তু এক \texttt{<div>} ট্যাগের ভেতরে অগোছালোভাবে লেখা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item মোবাইল পর্দায় পেজ কেমন হবে তা বিবেচনা না করা।
\item ছবির প্রস্থ বেশি দিয়ে পেজের সীমানা ছাড়িয়ে ফেলা।
\item মেনুর লিংকে \texttt{href} না দেওয়া।
\end{itemize}
\end{minipage}\par

\end{mistakebox}
