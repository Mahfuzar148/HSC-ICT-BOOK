\section{HTML-এ তালিকা (List)}

% Composition pass 2026-10-04
বইয়ের বিষয়গুলো একটানা অনুচ্ছেদে লিখলে আলাদা উপাদান ও তাদের ক্রম বোঝা কঠিন।

\chFourMotivation{তথ্যকে পৃথক উপাদানে সাজানো $\rightarrow$ তালিকা $\rightarrow$ ক্রম, বুলেট বা পরিভাষা–বর্ণনার জোড়া $\rightarrow$ লাইব্রেরির বিভাগ ও বইয়ের বিষয় দ্রুত পড়া যায়।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item HTML তালিকা কী তা ব্যাখ্যা করতে।
\item ক্রমযুক্ত তালিকা, বুলেট তালিকা ও বর্ণনামূলক তালিকার পার্থক্য লিখতে।
\item \texttt{<ol>}, \texttt{<ul>}, \texttt{<li>}, \texttt{<dl>}, \texttt{<dt>} ও \texttt{<dd>} ট্যাগ ব্যবহার করতে।
\item \texttt{type} ও \texttt{start} অ্যাট্রিবিউটের কাজ বুঝতে।
\item নেস্টেড তালিকা তৈরি করতে।
\item HSC ব্যবহারিক অনুশীলন ও বোর্ড পরীক্ষার তালিকা-সম্পর্কিত প্রশ্নের উত্তর দিতে।
\end{itemize}
\end{learningbox}



\subsection{HTML তালিকা কী?}

\begin{definitionbox}{HTML তালিকা (List)}
ওয়েবপেজে একাধিক উপাদানকে ক্রম, বুলেট কিংবা পরিভাষা ও বর্ণনার জোড়ায় সাজানোর পদ্ধতিকে HTML তালিকা বলে।
\end{definitionbox}

পঠিত বিষয়, খাবারের মেনু, বই, রুটিন, পণ্যের বৈশিষ্ট্য, নির্দেশনার ধাপ ও ওয়েবসাইটের মেনু তালিকায় সাজানো যায়। পরিভাষা ও তার অর্থও তালিকায় দেখানো যায়। এতে প্রতিটি উপাদান আলাদা করে পড়া ও তুলনা করা সহজ হয়।




\subsection{HTML তালিকার প্রকারভেদ}

\begin{center}
\begin{tabular}{|p{0.25\textwidth}|p{0.18\textwidth}|p{0.43\textwidth}|}
\hline
\textbf{তালিকার ধরন} & \textbf{ট্যাগ} & \textbf{ব্যবহার} \\
\hline
ক্রমযুক্ত তালিকা & \texttt{<ol>} & সংখ্যা, অক্ষর বা রোমান সংখ্যা চিহ্নসহ ক্রমানুসার তালিকা \\
\hline
বুলেট তালিকা & \texttt{<ul>} & বুলেট/চিহ্নসহ তালিকা; উপাদানের ক্রম গুরুত্বপূর্ণ নয় \\
\hline
বর্ণনামূলক তালিকা & \texttt{<dl>} & পরিভাষা ও বর্ণনা আকারে তালিকা \\
\hline
\end{tabular}
\end{center}

\begin{keypointbox}{মনে রাখো}
\texttt{<ol>} ও \texttt{<ul>} তালিকার প্রতিটি উপাদান লেখার জন্য \texttt{<li>} ট্যাগ ব্যবহার করা হয়। কিন্তু \texttt{<dl>} তালিকায় \texttt{<li>} নয়; পরিভাষার জন্য \texttt{<dt>} এবং বর্ণনার জন্য \texttt{<dd>} ব্যবহার করা হয়।
\end{keypointbox}

\subsection{ক্রমযুক্ত তালিকা: \texttt{<ol>}}

\begin{definitionbox}{ক্রমযুক্ত তালিকা (Ordered List)}
যে তালিকায় উপাদানগুলো নির্দিষ্ট ক্রমে সংখ্যা, অক্ষর বা রোমান সংখ্যার চিহ্নসহ দেখানো হয়, তাকে ক্রমযুক্ত তালিকা বলে।
\end{definitionbox}

উপাদানের ক্রম গুরুত্বপূর্ণ হলে ক্রমযুক্ত তালিকা ব্যবহার করা হয়। নির্দেশনার ধাপ, পরীক্ষার রুটিন, স্থানক্রম, অধ্যায়ের তালিকা, ভর্তি প্রক্রিয়া ও অ্যালগরিদমের ধাপ এর উদাহরণ।

\subsubsection{সিনট্যাক্স}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<ol>
  <li>First item</li>
  <li>Second item</li>
  <li>Third item</li>
</ol>
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\subsection{ক্রমযুক্ত তালিকার অ্যাট্রিবিউট}

\subsubsection{\texttt{type} অ্যাট্রিবিউট}

\texttt{type} অ্যাট্রিবিউট দিয়ে ক্রমযুক্ত তালিকার চিহ্ন কীভাবে দেখাবে তা নির্ধারণ করা যায়।

\begin{center}
\begin{tabular}{|p{0.25\textwidth}|p{0.35\textwidth}|}
\hline
\textbf{ধরন} & \textbf{ফলের চিহ্ন} \\
\hline
\texttt{type="1"} & 1, 2, 3 \\
\hline
\texttt{type="A"} & A, B, C \\
\hline
\texttt{type="a"} & a, b, c \\
\hline
\texttt{type="I"} & I, II, III \\
\hline
\texttt{type="i"} & i, ii, iii \\
\hline
\end{tabular}
\end{center}

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}

\begin{verbatim}
<ol type="A">
  <li>HTML</li>
  <li>CSS</li>
  <li>JavaScript</li>
</ol>
\end{verbatim}

\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
\textbf{ফল:}\\
A. HTML\\
B. CSS\\
C. JavaScript\\
\end{minipage}
\chFourWhy{\texttt{type="A"} দেওয়ায় চিহ্ন সংখ্যা না হয়ে বড় হাতের অক্ষর হিসেবে দেখায়।}
\end{chFourWholeExample}

\subsubsection{\texttt{start} অ্যাট্রিবিউট}

\texttt{start} অ্যাট্রিবিউট দিয়ে ক্রমযুক্ত তালিকা কোন সংখ্যা থেকে শুরু হবে তা নির্ধারণ করা যায়।

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}

\begin{verbatim}
<ol start="5">
  <li>HTML</li>
  <li>CSS</li>
  <li>JavaScript</li>
</ol>
\end{verbatim}

\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
\textbf{ফল:}
\begin{enumerate}[start=5]
\item HTML
\item CSS
\item JavaScript
\end{enumerate}
\end{minipage}
\chFourWhy{\texttt{start="5"} দেওয়ায় তালিকার চিহ্ন ১ থেকে না শুরু হয়ে ৫ থেকে শুরু হয়েছে।}
\end{chFourWholeExample}

\begin{keypointbox}{মান অনুযায়ী বাড়তি তথ্য}
\texttt{reversed} অ্যাট্রিবিউট দিলে ক্রমযুক্ত তালিকা উল্টো ক্রমে গণনা হয়।
নির্দিষ্ট উপাদান থেকে গণনার মান বদলাতে সেই \texttt{<li>} এলিমেন্টে \texttt{value} অ্যাট্রিবিউট ব্যবহার করা যায়।
\end{keypointbox}



\subsection{বুলেট তালিকা: \texttt{<ul>}}

\begin{definitionbox}{বুলেট তালিকা (Unordered List)}
যে তালিকায় উপাদানের নির্দিষ্ট ক্রম গুরুত্বপূর্ণ নয় এবং প্রতিটি উপাদান বুলেট বা চিহ্নসহ দেখানো হয়, তাকে বুলেট তালিকা বলে।
\end{definitionbox}

যখন উপাদানের ক্রম গুরুত্বপূর্ণ নয়, তখন বুলেট তালিকা ব্যবহার করা হয়। যেমন: মেনু, বৈশিষ্ট্য, পঠিত বিষয়, কেনাকাটা, বিজ্ঞপ্তি, গুরুত্বপূর্ণ পাঠ বা ওয়েবসাইটে চলাচলের তালিকা।

\subsubsection{সিনট্যাক্স}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<ul>
  <li>Item one</li>
  <li>Item two</li>
  <li>Item three</li>
</ul>
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\subsubsection{উদাহরণ}

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}

\begin{verbatim}
<h2>Web Design Tools</h2>

<ul>
  <li>HTML</li>
  <li>CSS</li>
  <li>JavaScript</li>
  <li>Browser</li>
</ul>
\end{verbatim}

\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
\textbf{ফল:}
\begin{itemize}
\item HTML
\item CSS
\item JavaScript
\item Browser
\end{itemize}
\end{minipage}
\chFourWhy{\texttt{<ul>} বুলেট তালিকা তৈরি করে। ব্রাউজার স্বাভাবিক অবস্থায় বুলেট চিহ্ন দেখায়।}
\end{chFourWholeExample}

\subsection{বুলেট তালিকার \texttt{type} অ্যাট্রিবিউট}

পরীক্ষায় পরিচিত পুরোনো HTML কোডে বুলেট তালিকার চিহ্ন পরিবর্তনের জন্য \texttt{type} অ্যাট্রিবিউট দেখা যায়।

\begin{center}
\begin{tabular}{|p{0.30\textwidth}|p{0.34\textwidth}|}
\hline
\textbf{ধরন} & \textbf{ফল কী হবে?} \\
\hline
\texttt{type="disc"} & ভরাট বুলেট \\
\hline
\texttt{type="circle"} & ফাঁপা গোলাকার বুলেট \\
\hline
\texttt{type="square"} & বর্গাকার বুলেট \\
\hline
\end{tabular}
\end{center}

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}

\begin{verbatim}
<ul type="square">
  <li>Monitor</li>
  <li>Keyboard</li>
  <li>Mouse</li>
</ul>
\end{verbatim}

\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
\textbf{ফল:}\\
\(\blacksquare\) Monitor\\
\(\blacksquare\) Keyboard\\
\(\blacksquare\) Mouse\\
\end{minipage}
\chFourWhy{\texttt{type="square"} দেওয়ায় তালিকার চিহ্ন বর্গাকার বুলেট আকারে দেখা যাবে।}
\end{chFourWholeExample}

\begin{mistakebox}{আধুনিক HTML-এর নিয়ম}
বর্তমান HTML-এ \texttt{<ul>}-এর \texttt{type} পুরোনো অ্যাট্রিবিউট; বুলেটের ধরন বদলাতে CSS-এর \texttt{list-style-type} ব্যবহার করা হয়। \texttt{<ol>}-এর \texttt{type} কিন্তু বর্তমান মানে বৈধ। পরীক্ষার পুরোনো উদাহরণ বোঝার জন্য \texttt{type="disc"}, \texttt{type="circle"} ও \texttt{type="square"} জানা দরকার।
\end{mistakebox}



\subsection{তালিকার উপাদান: \texttt{<li>}}

\begin{definitionbox}{তালিকার উপাদান (List Item)}
\texttt{<li>} ট্যাগ তালিকার প্রতিটি উপাদান লিখতে ব্যবহৃত হয়।
\end{definitionbox}

\texttt{<ol>} এবং \texttt{<ul>} - দুই ধরনের তালিকার উপাদান লিখতেই \texttt{<li>} ট্যাগ ব্যবহার করা হয়।

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}

\begin{verbatim}
<ul>
  <li>Bangla</li>
  <li>English</li>
  <li>ICT</li>
</ul>
\end{verbatim}

\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
\textbf{ফল:}
\begin{itemize}
\item Bangla
\item English
\item ICT
\end{itemize}
\end{minipage}
\chFourWhy{প্রতিটি \texttt{<li>} ট্যাগ একটি আলাদা তালিকার উপাদান তৈরি করেছে।}
\end{chFourWholeExample}

\begin{keypointbox}{পরীক্ষায় মনে রাখবে}
HTML-এর নির্দিষ্ট নিয়মে \texttt{</li>} বাদ দেওয়া যায়; যেমন পরেই আরেকটি \texttt{<li>} শুরু হলে। এটি শুধু পুরোনো HTML-এর নিয়ম নয়। শেখার সময় \texttt{<li>Item</li>} আকারে শেষের ট্যাগ লিখলে তালিকার গঠন সহজে বোঝা যায়।
\end{keypointbox}

\subsection{বর্ণনামূলক তালিকা: \texttt{<dl>}, \texttt{<dt>}, \texttt{<dd>}}

\begin{definitionbox}{বর্ণনামূলক তালিকা (Description List)}
পরিভাষা বা বিষয়ের নামের সঙ্গে তার বর্ণনা সাজিয়ে দেখানোর তালিকাকে বর্ণনামূলক তালিকা বলে।
\end{definitionbox}

পরিভাষা, সংজ্ঞা, বহুল জিজ্ঞাসিত প্রশ্ন (Frequently Asked Questions বা FAQ), পণ্যের বৈশিষ্ট্য, যোগাযোগের বিস্তারিত তথ্য বা শব্দের অর্থ দেখাতে বর্ণনামূলক তালিকা ব্যবহার করা যায়।

\subsection{নেস্টেড তালিকা}

\begin{definitionbox}{নেস্টেড তালিকা (Nested List)}
একটি তালিকার উপাদানের ভিতরে আরেকটি তালিকা ব্যবহার করলে তাকে নেস্টেড তালিকা বলা যায়।
\end{definitionbox}

\par\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\par\noindent
\begin{verbatim}
<ul>
  <li>Science
    <ol>
      <li>Physics</li>
      <li>Chemistry</li>
      <li>Biology</li>
    </ol>
  </li>

  <li>Business Studies
    <ol>
      <li>Accounting</li>
      <li>Finance</li>
      <li>Management</li>
    </ol>
  </li>
</ul>
\end{verbatim}
\par

  \begin{enumerate}
  \item Physics
  \item Chemistry
  \item Biology
  \end{enumerate}

  \begin{enumerate}
  \item Accounting
  \item Finance
  \item Management
  \end{enumerate}
\end{itemize}

\par\smallskip
\end{minipage}\par


\begin{keypointbox}{নেস্টেড তালিকা কোথায় ব্যবহার হয়}
পঠিত বিষয়ের দল, ওয়েবসাইটের মেনু, অধ্যায় ও উপপাঠ, শ্রেণি ও উপশ্রেণি, কোর্সের অংশ বা পণ্যের বৈশিষ্ট্যের দল দেখাতে নেস্টেড তালিকা ব্যবহার করা যায়।
\end{keypointbox}

\begin{mistakebox}{সাধারণ ভুল}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{<ol>} ও \texttt{<ul>} গুলিয়ে ফেলা।
\item \texttt{<li>} ট্যাগ না দেওয়া।
\item \texttt{<dl>} ট্যাগের ভিতরে \texttt{<li>} ব্যবহার করা।
\item \texttt{<dt>} ও \texttt{<dd>} ভুলভাবে ব্যবহার করা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item ক্রমযুক্ত তালিকায় \texttt{type} মান ভুল লেখা।
\item \texttt{<ul>} ট্যাগে সংখ্যা চিহ্ন আশা করা।
\item তালিকার শেষের ট্যাগ না দেওয়া।
\item নেস্টেড তালিকা ভুলভাবে বন্ধ করা।
\end{itemize}
\end{minipage}\par

\end{mistakebox}
