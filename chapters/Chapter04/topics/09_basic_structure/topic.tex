\section{HTML ডকুমেন্টের মৌলিক কাঠামো}

% Composition pass 2026-10-04
পেজের নাম ব্রাউজারের ট্যাবে এবং কলেজের পরিচিতি পেজের ভেতরে দেখাতে হবে।

\chFourMotivation{পেজের তথ্য ও দৃশ্যমান অংশ আলাদা রাখা $\rightarrow$ head ও body $\rightarrow$ নির্দিষ্ট অংশে উপযুক্ত এলিমেন্ট লেখা $\rightarrow$ সঠিক শিরোনাম, এনকোডিং ও বিষয়বস্তু।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item HTML ডকুমেন্টের কাঠামো কী তা ব্যাখ্যা করতে।
\item \texttt{<!DOCTYPE html>} ঘোষণা এবং \texttt{<html>}, \texttt{<head>}, \texttt{<title>} ও \texttt{<body>} ট্যাগের কাজ লিখতে।
\item \texttt{<head>} ও \texttt{<body>} অংশের পার্থক্য বুঝতে।
\item একটি মৌলিক HTML ফাইল লিখে \texttt{.html} এক্সটেনশন দিয়ে সংরক্ষণ করতে।
\item HSC-এর ব্যবহারিক অনুশীলন ও বোর্ড পরীক্ষার জন্য মানসম্মত HTML কাঠামো মনে রাখতে।
\end{itemize}
\end{learningbox}

\subsection{ভূমিকা}
সম্পূর্ণ HTML নথিতে পেজ-সম্পর্কিত তথ্য ও দৃশ্যমান বিষয়বস্তু আলাদা অংশে লেখা হয়। \texttt{head}-এ থাকে শিরোনাম ও মেটাডেটা; \texttt{body}-এ থাকে ব্যবহারকারীর জন্য পেজের বিষয়বস্তু। শুরুতে \texttt{<!DOCTYPE html>} ঘোষণা এবং \texttt{head}-এ এনকোডিং লিখলে ব্রাউজার নথি পড়তে ও দেখাতে সঠিক নির্দেশনা পায়।

\begin{definitionbox}{HTML ডকুমেন্টের কাঠামো (Document Structure)}
HTML নথির মূল এলিমেন্ট এবং তার অধীনে পেজের তথ্য ও বিষয়বস্তু সাজানোর বিন্যাসকে HTML ডকুমেন্টের কাঠামো বলে।
\end{definitionbox}

\subsection{HTML-এর মৌলিক কাঠামো}

\par\noindent{\color{chFourTeal}\bfseries\small কোড ও ব্রাউজারে ফল: কোডের বাম কলাম শেষে ডান কলাম পড়বে}\par
\begin{multicols}{2}
\begin{verbatim}
<!DOCTYPE html>
<html>
<head>
  <meta charset="UTF-8">
  <title>My First Web Page</title>
</head>
<body>
  <h1>Welcome to HSC ICT</h1>
  <p>This is my first HTML web page.</p>
</body>
</html>
\end{verbatim}
\end{multicols}
ব্রাউজার ট্যাবে \texttt{My First Web Page} দেখা যাবে। ওয়েবপেজের দৃশ্যমান অংশে \texttt{Welcome to HSC ICT} শিরোনাম এবং নিচে অনুচ্ছেদ লেখা দেখা যাবে।



\begin{keypointbox}{আধুনিক ব্যবহারের বাড়তি তথ্য}
\begin{itemize}
\item \texttt{<html>} ট্যাগে \texttt{lang} অ্যাট্রিবিউট দিলে পেজের ভাষা বোঝা সহজ হয়।
\item \texttt{<meta charset="UTF-8">} বাংলা ও অন্যান্য Unicode লেখা ঠিকভাবে দেখাতে সাহায্য করে।
\item \texttt{<meta name="viewport" content="width=device-width, initial-scale=1.0">} মোবাইলের পর্দার মাপ অনুযায়ী পেজ দেখাতে কাজে লাগে।
\item \texttt{<head>} অংশে শিরোনাম, মেটাডেটা, CSS লিংক ও স্ক্রিপ্ট লিংক রাখা হয়।
\end{itemize}
\end{keypointbox}

\subsection{কাঠামোর প্রধান অংশ}

\subsubsection{\texttt{<!DOCTYPE html>}}

\begin{definitionbox}{ডকুমেন্টের ধরন ঘোষণা (Document Type Declaration)}
\texttt{<!DOCTYPE html>} হলো HTML ডকুমেন্টের শুরুতে লেখা ঘোষণা। এটি ব্রাউজারকে পুরোনো প্রদর্শন-পদ্ধতি এড়িয়ে বর্তমান মান অনুযায়ী পেজ দেখাতে সাহায্য করে; ঘোষণাটি কোডের সব ভুল পরীক্ষা করে না।
\end{definitionbox}

\begin{itemize}
\item এটি HTML ট্যাগ নয়, ঘোষণা।
\item সাধারণত HTML ডকুমেন্টের একদম শুরুতে লেখা হয়।
\item ব্রাউজারকে মান অনুযায়ী প্রদর্শনের পদ্ধতিতে পেজ প্রদর্শন করতে সাহায্য করে।
\item HSC মৌলিক কোডে অনেক সময় না দেখালেও আধুনিক HTML কোডে এটি লেখা ভালো।
\end{itemize}

\subsubsection{\texttt{<html>} ট্যাগ}

\begin{definitionbox}{মূল এলিমেন্ট (Root Element)}
\texttt{<html>} ট্যাগ হলো HTML ডকুমেন্টের মূল এলিমেন্ট; এর ভিতরে সম্পূর্ণ ওয়েবপেজের কোড লেখা হয়।
\end{definitionbox}

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<html>
  ...
</html>
\end{verbatim}

\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}\small
\begin{itemize}
\item \texttt{<html>} ট্যাগ ডকুমেন্টের শুরু নির্দেশ করে।
\item \texttt{</html>} ট্যাগ ডকুমেন্টের শেষ নির্দেশ করে।
\item এর ভিতরে সাধারণত দুইটি প্রধান অংশ থাকে: \texttt{<head>} এবং \texttt{<body>}।
\end{itemize}
\end{minipage}\par


\subsubsection{\texttt{<head>} ট্যাগ}

\begin{definitionbox}{পেজের তথ্যের অংশ (Head Section)}
\texttt{<head>} ট্যাগ হলো HTML ডকুমেন্টের এমন অংশ যেখানে পেজ সম্পর্কিত তথ্য, শিরোনাম, রীতি, স্ক্রিপ্ট, লিংক এবং মেটাডেটা রাখা হয়।
\end{definitionbox}

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<head>
  <title>Page Title</title>
</head>
\end{verbatim}

\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}\small
\begin{itemize}
\item এই অংশের বিষয়বস্তু সাধারণত ব্রাউজার উইন্ডোর প্রধান পেজ এলাকায় সরাসরি দেখা যায় না।
\item পেজ শিরোনাম, অক্ষর এনকোডিং, CSS লিংক, স্ক্রিপ্ট লিংক ইত্যাদি এখানে লেখা যায়।
\item মেটাডেটা পেজের পরিচয় ও বৈশিষ্ট্য জানায়; এটি মূল বিষয়বস্তু নয়।
\end{itemize}
\end{minipage}\par


\begin{keypointbox}{head অংশে থাকতে পারে}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{<title>}
\item \texttt{<meta>}
\item \texttt{<link>}
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{<style>}
\item \texttt{<script>}
\item \texttt{<base>}
\end{itemize}
\end{minipage}\par

\end{keypointbox}

\subsubsection{\texttt{<meta charset="UTF-8">}}

\begin{definitionbox}{অক্ষর এনকোডিং (Character Encoding)}
অক্ষরকে বাইটে প্রকাশ এবং সেই বাইট পড়ে অক্ষর শনাক্ত করার নিয়মকে \textbf{অক্ষর এনকোডিং} বলে। \texttt{<meta charset="UTF-8">} জানায় যে নথিটি UTF-8 এনকোডিংয়ে লেখা।
\end{definitionbox}

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<meta charset="UTF-8">
\end{verbatim}

\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}\small
\begin{itemize}
\item বাংলা, ইংরেজি এবং অধিকাংশ ভাষার অক্ষর ঠিকভাবে দেখাতে সাহায্য করে।
\item এটি \texttt{<head>} ট্যাগের ভিতরে লেখা হয়।
\item বাংলা লেখা যুক্ত ওয়েবপেজে এই ঘোষণা দেওয়ার পাশাপাশি ফাইলটিও UTF-8 এনকোডিংয়ে সংরক্ষণ করতে হবে। শুধু ঘোষণা বদলালে অন্য এনকোডিংয়ের ফাইল স্বয়ংক্রিয়ভাবে রূপান্তরিত হয় না।
\end{itemize}
\end{minipage}\par

\subsubsection{\texttt{<title>} ট্যাগ}

\begin{definitionbox}{পেজের শিরোনাম (Page Title)}
\texttt{<title>} ট্যাগ ব্রাউজার ট্যাব বা শিরোনাম বারে ওয়েবপেজের শিরোনাম প্রদর্শন করে।
\end{definitionbox}

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<title>My First Web Page</title>
\end{verbatim}

\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}\small
\begin{itemize}
\item এটি \texttt{<head>} ট্যাগের ভিতরে লেখা হয়।
\item এটি পেজের দৃশ্যমান অংশে দেখা যায় না।
\item বুকমার্ক বা অনুসন্ধান ফলে পেজ শিরোনাম বোঝাতে সাহায্য করে।
\end{itemize}
\end{minipage}\par

\hline
\texttt{<title>} & ব্রাউজার ট্যাব/শিরোনাম বারে \\
\hline
\texttt{<h1>} & ওয়েবপেজের দৃশ্যমান অংশে \\

\subsubsection{\texttt{<body>} ট্যাগ}

\begin{definitionbox}{দৃশ্যমান বিষয়বস্তুর অংশ (Body Section)}
\texttt{<body>} ট্যাগ হলো HTML ডকুমেন্টের প্রধান অংশ, যেখানে ওয়েবপেজের দৃশ্যমান বিষয়বস্তু লেখা হয়।
\end{definitionbox}

% Split revision: code and its explanation share one row.
\par\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<body>
  <h1>Welcome</h1>
  <p>This is a paragraph.</p>
</body>
\end{verbatim}

\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}\small
\begin{itemize}
\item ওয়েবপেজে যে লেখা, ছবি, লিংক, সারণি, তালিকা বা ফর্ম দেখা যায়, তা \texttt{<body>} অংশে লেখা হয়।
\item \texttt{<body>} ট্যাগের বাইরে দৃশ্যমান শিরোনাম বা অনুচ্ছেদ লেখা উচিত নয়।
\item ব্যবহারিক উদাহরণের শিরোনাম, তালিকা ও সারণি এই অংশেই যোগ করা হয়।
\end{itemize}
\end{minipage}\par


\begin{keypointbox}{body অংশে থাকতে পারে}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item শিরোনাম: \texttt{<h1>} থেকে \texttt{<h6>}
\item অনুচ্ছেদ: \texttt{<p>}
\item ছবি: \texttt{<img>}
\item হাইপারলিংক: \texttt{<a>}
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item তালিকা: \texttt{<ul>}, \texttt{<ol>}, \texttt{<li>}
\item সারণি: \texttt{<table>}
\item ফর্ম: \texttt{<form>}
\end{itemize}
\end{minipage}\par

\end{keypointbox}

\textbf{চিত্র থেকে কোড বোঝার নিয়ম}\par
গাছের মতো কাঠামোর চিত্রে বাইরের ট্যাগের ভিতরে ভেতরের ট্যাগ থাকে। যেমন \texttt{<html>} ট্যাগের ভিতরে \texttt{<head>} ও \texttt{<body>} থাকে। আবার \texttt{<body>} ট্যাগের ভিতরে দৃশ্যমান বিষয়বস্তু যেমন শিরোনাম, অনুচ্ছেদ, ছবি, লিংক, তালিকা ও সারণি লেখা হয়।
\par \texttt{style}-ও \texttt{head}-এ থাকতে পারে। নথির শুরুতে \texttt{<!DOCTYPE html>} ঘোষণা লেখা হয়; এটি এলিমেন্টের ট্যাগ নয়।
\end{minipage}

\par\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<!DOCTYPE html>
<html>
  <head>
    <title>Page Title</title>
  </head>

  <body>
    Visible content
  </body>
</html>
\end{verbatim}

\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কাঠামো পড়ার নিয়ম}\par
\texttt{<!DOCTYPE html>} নথির ধরন ঘোষণা করে।\par
\texttt{html}-এর ভিতরে \texttt{head} ও \texttt{body} আছে।\par
\texttt{Page Title} ব্রাউজারের ট্যাবে এবং \texttt{Visible content} পেজের দৃশ্যমান অংশে দেখা যাবে।
\end{minipage}\par

\subsection{HTML ফাইল সংরক্ষণের নিয়ম}

HTML কোড লেখার পর ফাইল সংরক্ষণ করতে হবে \texttt{.html} বা \texttt{.htm} এক্সটেনশন দিয়ে।

\begin{examplebox}{ফাইলের নামের উদাহরণ}
\texttt{index.html}, \texttt{home.html}, \texttt{about.html}
\end{examplebox}

\begin{enumerate}
\item Notepad বা VS Code খোলো।
\item HTML কোড লিখো।
\item ফাইলটি UTF-8 এনকোডিংয়ে \texttt{index.html} নামে সংরক্ষণ করো। Notepad-এ প্রয়োজন হলে \texttt{Save as type: All Files} বেছে নেবে, যাতে নামের শেষে \texttt{.txt} যুক্ত না হয়।
\item ফাইল ব্রাউজার দিয়ে খোলো।
\item ফল ব্রাউজারে দেখো।
\end{enumerate}

\begin{mistakebox}{সাধারণ ভুল}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{<!DOCTYPE html>} না লেখা।
\item \texttt{<head>} ট্যাগের ভিতরে দৃশ্যমান বিষয়বস্তু লেখা।
\item \texttt{<body>} ট্যাগের বাইরে শিরোনাম বা অনুচ্ছেদ লেখা।
\item \texttt{<title>} ট্যাগকে পেজে দৃশ্যমান শিরোনামের ট্যাগ মনে করা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item শেষের ট্যাগ না দেওয়া।
\item ফাইল \texttt{.txt} হিসেবে সংরক্ষণ করা।
\item বাংলা লেখা ঠিকভাবে দেখানোর জন্য \texttt{UTF-8} না দেওয়া।
\item \texttt{<head>} ও \texttt{<body>} ট্যাগের ক্রম ভুল করা।
\end{itemize}
\end{minipage}\par

\end{mistakebox}

\begin{boardbox}{সঠিক ক্রম মনে রাখো}
\texttt{DOCTYPE} $\rightarrow$ \texttt{html} $\rightarrow$ \texttt{head} $\rightarrow$ \texttt{title/meta} $\rightarrow$ \texttt{body} $\rightarrow$ দৃশ্যমান বিষয়বস্তু
\end{boardbox}

\subsection{সংক্ষিপ্ত অনুশীলন}

HTML ডকুমেন্ট একটি নির্দিষ্ট কাঠামো অনুসরণ করে লেখা হয়। আধুনিক HTML ডকুমেন্টের শুরুতে \texttt{<!DOCTYPE html>} লেখা হয়। এরপর \texttt{<html>} ট্যাগের মধ্যে সম্পূর্ণ ডকুমেন্ট থাকে। \texttt{<head>} অংশে ওয়েবপেজ সম্পর্কিত তথ্য, শিরোনাম, মেটাডেটা, রীতি ও স্ক্রিপ্ট রাখা হয়। \texttt{<title>} ট্যাগ ব্রাউজার ট্যাবে পেজ শিরোনাম প্রদর্শন করে। \texttt{<body>} অংশে ওয়েবপেজের দৃশ্যমান বিষয়বস্তু যেমন শিরোনাম, অনুচ্ছেদ, ছবি, হাইপারলিংক, তালিকা, সারণি ও ফর্ম লেখা হয়।
