\begin{enumerate}

\item HTML এর পূর্ণরূপ কী? \textbf{[ঢাকা-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item HyperText Markup Language \checkmark
\item High Transfer Markup Language
\item HyperText Making Language

\item ডোমেইন নামের সাথে যুক্ত .com, .org এগুলোকে কী বলে? \textbf{[সকল বোর্ড-২০১৮]}

\item HTML ফাইলের এক্সটেনশন নিচের কোনটি? \textbf{[চট্টগ্রাম-২০২৩]}

\item .txt
\item .doc
\item .pdf

\item Body
\item Footer
\item Aside
\end{enumerate}

\item HTML এ সবচেয়ে বড় হেডিং ট্যাগ কোনটি? \textbf{[যশোর-২০২৩]}

\item <h1> \checkmark
\item <h6>
\item <head>
\item <header>
\end{enumerate}

\end{enumerate}
\end{multicols}
