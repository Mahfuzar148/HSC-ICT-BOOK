\section{HTML-এ হাইপারলিংক (Hyperlink)}

% Composition pass 2026-10-04
Home, Notice ও Result পেজ তৈরি আছে, কিন্তু এক পেজ থেকে অন্যটিতে যাওয়ার পথ দরকার।

\chFourMotivation{পেজে চলাচল $\rightarrow$ হাইপারলিংক $\rightarrow$ href-এ গন্তব্য ও ক্লিকযোগ্য লেখা/ছবি $\rightarrow$ কলেজের মেনু থেকে প্রয়োজনীয় পেজ খোলা।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item হাইপারলিংক কী এবং ওয়েবপেজ চলাচলে এর প্রয়োজনীয়তা ব্যাখ্যা করতে।
\item \texttt{<a>} ট্যাগ ও \texttt{href} অ্যাট্রিবিউটের কাজ লিখতে।
\item অভ্যন্তরীণ, বাহ্যিক, বুকমার্ক, ইমেইল, ছবি, ফাইল ও নতুন ট্যাবের লিংক তৈরি করতে।
\item লিংকের লেখা, URL, \texttt{mailto:}, \texttt{target} ও \texttt{download} অ্যাট্রিবিউট ব্যবহার করতে।
\item পূর্ণ URL, আপেক্ষিক URL এবং নতুন ট্যাবে লিংক খোলার নিয়ম ব্যাখ্যা করতে।
\item HSC ব্যবহারিক অনুশীলন ও বোর্ড পরীক্ষার হাইপারলিংক-সম্পর্কিত প্রশ্নের উত্তর দিতে।
\end{itemize}
\end{learningbox}



HTML তালিকা শেখার পর হাইপারলিংক শেখা গুরুত্বপূর্ণ, কারণ একটি ওয়েবপেজের লেখা, তালিকা, ছবি বা বোতামের মাধ্যমে অন্য পেজ, ওয়েবসাইট, ফাইল বা একই পেজের নির্দিষ্ট অংশে যাওয়ার ব্যবস্থা হাইপারলিংক দিয়ে করা হয়। ওয়েবসাইটের চলাচল মেনু, ``Read more'', ``Download'', ``Contact us'', ছবি বোতাম - এসব হাইপারলিংকের বাস্তব উদাহরণ।

\subsection{হাইপারলিংক কী?}

\begin{definitionbox}{হাইপারলিংক (Hyperlink)}
ওয়েবপেজের যে লেখা, ছবি বা উপাদান নির্বাচন করলে অন্য পেজ, ওয়েবসাইট, ফাইল, ইমেইল ঠিকানা কিংবা একই পেজের নির্দিষ্ট অংশে যাওয়া যায়, তাকে হাইপারলিংক বলে।
\end{definitionbox}

হাইপারলিংক পেজ ও তথ্যের মধ্যে সংযোগ তৈরি করে। ব্যবহারকারী এই সংযোগ ধরে প্রয়োজনীয় তথ্যে পৌঁছান। তাই ওয়েবসাইটে সহজে চলাচলের জন্য অর্থবহ লিংক থাকা প্রয়োজন।



\subsection{অ্যাঙ্কর ট্যাগ: \texttt{<a>}}

\begin{definitionbox}{অ্যাঙ্কর ট্যাগ (Anchor Tag)}
HTML-এ হাইপারলিংক তৈরি করার জন্য \texttt{<a>} ট্যাগ ব্যবহার করা হয়। \texttt{<a>} ট্যাগকে অ্যাঙ্কর ট্যাগ বলা হয়।
\end{definitionbox}

অ্যাঙ্কর ট্যাগের সবচেয়ে গুরুত্বপূর্ণ অ্যাট্রিবিউট হলো \texttt{href}। \texttt{href} ছাড়া \texttt{<a>} ট্যাগ লিখলে সাধারণত লিংক কার্যকর হয় না, কারণ ব্রাউজার জানে না ক্লিক করলে কোথায় যেতে হবে।

\subsubsection{লেখার সাধারণ রূপ}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<a href="url">Link Text</a>
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\begin{center}
\begin{tabular}{|p{0.22\textwidth}|p{0.62\textwidth}|}
\hline
\textbf{অংশ} & \textbf{ব্যাখ্যা} \\
\hline
\texttt{<a>} & অ্যাঙ্কর ট্যাগ শুরু করে \\
\hline
\texttt{href} & লিংকের গন্তব্য বা ঠিকানা নির্ধারণ করে \\
\hline
\texttt{"url"} & যে পেজ, ওয়েবসাইট, ফাইল বা ঠিকানায় লিংক যাবে \\
\hline
\texttt{Link Text} & ব্যবহারকারী যে লেখায় ক্লিক করবে \\
\hline
\texttt{</a>} & অ্যাঙ্কর ট্যাগ শেষ করে \\
\hline
\end{tabular}
\end{center}

\end{minipage}\hfill

\end{minipage}
\chFourWhy{এখানে \texttt{href} অ্যাট্রিবিউট লিংকের গন্তব্য হিসেবে Google ওয়েবসাইটের ঠিকানা দিয়েছে এবং \texttt{Google} হলো ক্লিকযোগ্য লিংকের লেখা।}
\end{chFourWholeExample}

\subsection{\texttt{href} অ্যাট্রিবিউট}

\begin{definitionbox}{\texttt{href}}
\texttt{href} অ্যাট্রিবিউট লিংকের গন্তব্য ঠিকানা নির্ধারণ করে। Hyperlink Reference কথাটি দিয়ে এর কাজ বোঝানো হয়; কোডে অ্যাট্রিবিউটের নাম \texttt{href}।
\end{definitionbox}

\begin{keypointbox}{মনে রাখো}
\texttt{href} ছাড়া \texttt{<a>} এলিমেন্ট হাইপারলিংক তৈরি করে না; এটি লিংকের জন্য স্থানধারক হিসেবে থাকতে পারে।
\end{keypointbox}


\begin{keypointbox}{লিংকের লেখা বাছাইয়ের নিয়ম}
\texttt{Click here} এর চেয়ে \texttt{Download Routine}, \texttt{Visit Google} বা \texttt{Go to Contact} - এ ধরনের অর্থবহ লিংকের লেখা ব্যবহার করা ভালো।
\end{keypointbox}

\subsection{হাইপারলিংকের প্রকারভেদ}

গন্তব্য, ক্লিকযোগ্য অংশ ও খোলার পদ্ধতি অনুযায়ী লিংকের নিচের রূপগুলো পরিচিত। এগুলো পরস্পর বিচ্ছিন্ন শ্রেণি নয়; একই বাহ্যিক লিংক ছবিতে দেওয়া যেতে পারে এবং নতুন ট্যাবেও খুলতে পারে:

\begin{enumerate}
\item বাহ্যিক লিংক
\item অভ্যন্তরীণ লিংক
\item একই পেজ বা বুকমার্ক লিংক
\item ইমেইল লিংক
\item ছবিতে লিংক
\item ডাউনলোড বা ফাইলের লিংক
\item নতুন ট্যাবের লিংক
\end{enumerate}

\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{definitionbox}{বাহ্যিক লিংক (External Link)}
বাহ্যিক লিংক হলো এমন হাইপারলিংক, যা বর্তমান ওয়েবসাইট বা বর্তমান পেজ থেকে অন্য কোনো ওয়েবসাইটে নিয়ে যায়।
\end{definitionbox}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{definitionbox}{অভ্যন্তরীণ লিংক (Internal Link)}
অভ্যন্তরীণ লিংক হলো এমন হাইপারলিংক, যা একই ওয়েবসাইটের একটি পেজ থেকে অন্য পেজে নিয়ে যায়।
\end{definitionbox}
\end{minipage}\par

\subsection{পূর্ণ URL ও আপেক্ষিক URL}

\begin{definitionbox}{পূর্ণ ঠিকানা (Absolute URL)}
পূর্ণ URL হলো সম্পূর্ণ ওয়েব ঠিকানা, যেখানে প্রোটোকল, ডোমেইন এবং প্রয়োজনীয় পথ একসঙ্গে লেখা থাকে।
\end{definitionbox}

\begin{definitionbox}{আপেক্ষিক ঠিকানা (Relative URL)}
যে ঠিকানা একটি ভিত্তি ঠিকানার সঙ্গে মিলিয়ে সম্পূর্ণ করা হয়, তাকে \textbf{আপেক্ষিক URL} বলে। একই ওয়েবসাইটের পেজ বা ফাইলের ক্ষেত্রে প্রায়ই শুধু আপেক্ষিক পথ লেখা হয়; যেমন \texttt{about.html}।
\end{definitionbox}

\begin{center}
\begin{tabular}{|p{0.28\textwidth}|p{0.30\textwidth}|p{0.28\textwidth}|}
\hline
\textbf{বিষয়} & \textbf{পূর্ণ URL} & \textbf{আপেক্ষিক URL} \\
\hline
ঠিকানা & সম্পূর্ণ ঠিকানা & ভিত্তি ঠিকানার সঙ্গে মিলিয়ে সম্পূর্ণ করতে হয় \\
\hline
প্রোটোকল & থাকে & সাধারণত থাকে না \\
\hline
ব্যবহার & একই বা অন্য সাইটের পূর্ণ ঠিকানা & ভিত্তি ঠিকানা অনুযায়ী গন্তব্য \\
\hline
উদাহরণ & \texttt{https://google.com} & \texttt{about.html} \\
\hline
\end{tabular}
\end{center}

\subsubsection{লিংক থেকে গন্তব্য বোঝার ধাপ}
একটি আপেক্ষিক ঠিকানা বুঝতে বর্তমান পেজের ভিত্তি ঠিকানাও লাগে। নিচের উদাহরণে ভিত্তি ঠিকানা \texttt{https://college.example/} ধরা হয়েছে।

\subsection{একই পেজের লিংক বা বুকমার্ক লিংক}

\begin{definitionbox}{একই পেজের লিংক (Same Page / Bookmark Link)}
একই পেজ লিংক হলো এমন হাইপারলিংক, যার মাধ্যমে একই ওয়েবপেজের নির্দিষ্ট অংশে যাওয়া যায়।
\end{definitionbox}

বড় ওয়েবপেজ, নিবন্ধ, সূচিপত্র বা বহুল জিজ্ঞাসিত প্রশ্নের অংশে একই পেজের লিংক ব্যবহার করলে ব্যবহারকারী দ্রুত প্রয়োজনীয় অংশে যেতে পারে।

\subsubsection{পরিচিত \texttt{target} মান}

\begin{center}
\begin{tabular}{|p{0.20\textwidth}|p{0.60\textwidth}|}
\hline
\textbf{মান} & \textbf{কাজ} \\
\hline
\texttt{\_self} & একই ট্যাব/উইন্ডোতে খোলে; স্বাভাবিক আচরণ \\
\hline
\texttt{\_blank} & নতুন ট্যাব/উইন্ডোতে খোলে \\
\hline
\texttt{\_parent} & বাইরের ফ্রেমে খোলে \\
\hline
\texttt{\_top} & পুরো ব্রাউজার উইন্ডোতে খোলে \\
\hline
\end{tabular}
\end{center}

\begin{keypointbox}{আধুনিক ব্যবহারের নিয়ম}
\texttt{rel="noopener"} নতুন পেজকে মূল পেজের \texttt{window.opener} সংযোগ পাওয়া থেকে আটকায়। \texttt{noreferrer} গন্তব্যে উৎস পেজের Referer তথ্য পাঠানো বন্ধ করে। বর্তমান HTML-এ \texttt{target="\_blank"} সাধারণত নিজেই opener সংযোগ বিচ্ছিন্ন করে; নিচের উদাহরণে দুটি উদ্দেশ্য স্পষ্ট করে লেখা হয়েছে।
\end{keypointbox}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<a href="https://www.google.com" target="_blank" rel="noopener noreferrer">
  Open Google
</a>
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\subsection{সাধারণ ভুল}

\begin{mistakebox}{সাধারণ ভুল}
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item \texttt{href} অ্যাট্রিবিউট না দেওয়া।
\item \texttt{href}-এর জায়গায় \texttt{src} লেখা।
\item বাহ্যিক লিংকে \texttt{https://} না দেওয়া।
\item শেষের \texttt{</a>} ট্যাগ না দেওয়া।
\item \texttt{<a>} ট্যাগের ভিতরে ক্লিক করার লেখা বা ছবি না রাখা।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item ছবিতে লিংক করতে \texttt{<img>} ট্যাগ \texttt{<a>} ট্যাগের বাইরে রাখা।
\item ইমেইল লিংকে \texttt{mailto:} না লেখা।
\item একই পেজের লিংকে \texttt{\#} না দেওয়া।
\item \texttt{id} এবং \texttt{href="\#id"} মান মিল না রাখা।
\end{itemize}
\end{minipage}\par

\end{mistakebox}

\subsection{বোর্ডভিত্তিক সৃজনশীল প্রশ্ন}

\begin{boardbox}{সৃজনশীল প্রশ্ন ১}
একটি কলেজ ওয়েবসাইটে \texttt{Home}, \texttt{Notice}, \texttt{Result}, \texttt{Email Us} এবং \texttt{Open Google} লিংক আছে। \texttt{Notice} ও \texttt{Result} একই ওয়েবসাইটের পেজে নিয়ে যায়, \texttt{Email Us} ইমেইল অ্যাপ্লিকেশন খোলার অনুরোধ করে, আর \texttt{Open Google} নতুন ট্যাবে যায়।
\end{boardbox}



\begin{boardbox}{সৃজনশীল প্রশ্ন ২}
একটি ওয়েবপেজে \texttt{Go to Admission Section} লিংক আছে, যা একই পেজের ভর্তি অংশে নিয়ে যায়। আরেকটি কোর্স পরিচিতিপত্র ছবিতে ক্লিক করলে সংশ্লিষ্ট কোর্স পেজ খোলে।
\end{boardbox}
