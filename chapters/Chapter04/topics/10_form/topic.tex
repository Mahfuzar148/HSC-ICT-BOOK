\section{HTML ফর্ম (Form)}

% Composition pass 2026-10-04
ওয়েবপেজে ভর্তির তথ্য পড়া যায়, কিন্তু শিক্ষার্থীকে আবেদনও জমা দিতে হবে।

\chFourMotivation{ব্যবহারকারীর তথ্য গ্রহণ $\rightarrow$ HTML ফর্ম $\rightarrow$ নিয়ন্ত্রণ উপাদানে তথ্য নিয়ে সার্ভারে অনুরোধ পাঠানো $\rightarrow$ আবেদন ও যোগাযোগ। সংরক্ষণের কাজ সার্ভারের প্রোগ্রাম করে।}


\begin{learningbox}
এই অংশ শেষে শিক্ষার্থী পারবে:
\begin{itemize}
\item HTML ফর্ম কী এবং কেন ব্যবহার করা হয় তা ব্যাখ্যা করতে।
\item \texttt{<form>}, \texttt{<input>}, \texttt{<textarea>}, \texttt{<select>}, \texttt{<option>}, \texttt{<button>} ও \texttt{<label>} ট্যাগ ব্যবহার করতে।
\item এক লাইনের টেক্সট ঘর, পাসওয়ার্ড ঘর, রেডিও বোতাম, চেকবক্স, টেক্সটএরিয়া, ড্রপডাউন, সাবমিট ও রিসেট বোতাম তৈরি করতে।
\item \texttt{action}, \texttt{method}, \texttt{name}, \texttt{value}, \texttt{placeholder}, \texttt{checked} ও \texttt{required} অ্যাট্রিবিউটের কাজ বুঝতে।
\item HSC ব্যবহারিক অনুশীলন ও বোর্ড পরীক্ষার ফর্ম-সম্পর্কিত প্রশ্নের উত্তর দিতে।
\end{itemize}
\end{learningbox}

সারণি দিয়ে ওয়েবপেজে ডেটা সাজিয়ে দেখানো যায়। কিন্তু ব্যবহারকারীর কাছ থেকে ডেটা নিতে হলে HTML ফর্ম ব্যবহার করতে হয়। অনলাইন নিবন্ধন, লগইন, যোগাযোগ বার্তা, অনুসন্ধান ঘর, মতামত, ভর্তির আবেদন, অর্ডারের ফর্ম বা জরিপের ফর্ম - এসব ক্ষেত্রে ফর্ম খুব গুরুত্বপূর্ণ।

\subsection{HTML ফর্ম কী?}

\begin{definitionbox}{HTML ফর্ম (Form)}
ব্যবহারকারীর তথ্য লেখা বা নির্বাচন করার জন্য ওয়েবপেজে সাজানো নিয়ন্ত্রণ উপাদানগুলোর ব্যবস্থাকে HTML ফর্ম বলে। ফর্ম জমা দিয়ে সেই তথ্য নির্দিষ্ট গন্তব্যে পাঠানো যায়।
\end{definitionbox}

ফর্মে নাম, ঠিকানা, ইমেইল, পাসওয়ার্ড ও মন্তব্য লেখা যায়। লিঙ্গ, পঠিত বিষয় ও অন্যান্য পছন্দ বিকল্প থেকে নির্বাচন করা যায়। ভর্তি ও লগইনের ফর্মে এভাবে প্রয়োজনীয় তথ্য নেওয়া হয়। ফর্ম জমা দিলে ডেটা গ্রহণের জন্য নির্ধারিত সার্ভারের প্রোগ্রামে অনুরোধ যায়।



\subsection{ফর্ম কেন ব্যবহার করা হয়?}

ফর্ম ব্যবহার করা হয়:
\par\noindent
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item লগইন করার জন্য।
\item নিবন্ধন করার জন্য।
\item যোগাযোগ বার্তা পাঠানোর জন্য।
\item অনুসন্ধান করার জন্য।
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[t]{.48\linewidth}\vspace{0pt}
\begin{itemize}
\item অনলাইন ভর্তি ফর্ম পূরণ করার জন্য।
\item মতামত নেওয়ার জন্য।
\item অর্ডার বা অর্থ পরিশোধের তথ্য নেওয়ার জন্য।
\item জরিপের উত্তর সংগ্রহের জন্য।
\end{itemize}
\end{minipage}\par


\subsection{\texttt{<form>} ট্যাগ}

\begin{definitionbox}{}
\texttt{<form>} ট্যাগ HTML ডকুমেন্টে একটি ফর্ম এলাকা তৈরি করার জন্য ব্যবহৃত হয়।
\end{definitionbox}

ফর্মে এক লাইনের লেখার ঘর, পাসওয়ার্ড ঘর, রেডিও বোতাম, চেকবক্স, ড্রপডাউন তালিকা ও টেক্সটএরিয়া রাখা যায়। সাবমিট বোতাম তথ্য জমা দেওয়ার কাজ শুরু করে।

\subsubsection{\texttt{action} অ্যাট্রিবিউট}

\texttt{action} অ্যাট্রিবিউট ফর্মের ডেটা কোন সার্ভার ফাইল বা URL-এ পাঠানো হবে তা নির্ধারণ করে।

\par\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}কোড}\par
\begin{verbatim}
<form action="form_action.php">
\end{verbatim}

\par\smallskip
\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}\small
\textbf{\color{chFourTeal}ব্যাখ্যা}\par
ব্যবহারকারী সাবমিট বোতাম ক্লিক করলে ফর্মের বিষয়বস্তু \texttt{action} অ্যাট্রিবিউটে দেওয়া ফাইল বা স্ক্রিপ্টের কাছে যায়।
\end{minipage}\par
\subsubsection{\texttt{method} অ্যাট্রিবিউট}

\texttt{method} অ্যাট্রিবিউট ফর্মের ডেটা কীভাবে সার্ভারে পাঠানো হবে তা নির্ধারণ করে।

\begin{center}
\begin{tabular}{|p{0.18\textwidth}|p{0.62\textwidth}|}
\hline
\textbf{পদ্ধতি} & \textbf{ব্যবহার} \\
\hline
\texttt{get} & ফর্মের ডেটা URL-এর query অংশে যুক্ত করে; অনুসন্ধানের মতো তথ্য উদ্ধারের কাজে প্রচলিত। \\
\hline
\texttt{post} & ডেটা HTTP অনুরোধের মূল অংশে পাঠায়; URL-এর query অংশে সরাসরি যুক্ত করে না। নিবন্ধন বা তথ্য জমা দেওয়ার কাজে প্রচলিত। \\
\hline
\end{tabular}
\end{center}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<form action="submit.php" method="post">
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\begin{keypointbox}{পরীক্ষায় মনে রাখবে}
অনুসন্ধানের ফর্মে \texttt{get} এবং লগইন বা নিবন্ধনের তথ্য জমা দেওয়ার ফর্মে \texttt{post} প্রচলিত। \texttt{post} নিজে ডেটা এনক্রিপ্ট করে না; পাসওয়ার্ড বা ব্যক্তিগত তথ্য পরিবহনের সুরক্ষায় HTTPS দরকার।
\end{keypointbox}

\subsection{\texttt{<input>} ট্যাগ}

\begin{definitionbox}{}
\texttt{<input>} ট্যাগ বিভিন্ন ধরনের তথ্য দেওয়ার ঘর, নির্বাচন উপাদান বা বোতাম তৈরি করে। \texttt{type} অ্যাট্রিবিউট তার ধরন নির্ধারণ করে।
\end{definitionbox}

\texttt{<input>} একটি এম্পটি ট্যাগ। এর শেষের ট্যাগ নেই। \texttt{type} অ্যাট্রিবিউটের মান অনুযায়ী তথ্য লেখার ঘরের ধরন পরিবর্তিত হয়।

\subsubsection{লেখার সাধারণ রূপ}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}
<input type="text" name="username">
\end{verbatim}
\end{minipage}\par
\end{examplebox}

\hline
\textbf{অ্যাট্রিবিউট} & \textbf{কাজ} \\
\hline
\texttt{type} & তথ্য লেখার ঘরের ধরন নির্ধারণ করে \\
\hline
\texttt{name} & সার্ভারে ডেটা পাঠানোর ঘরের নাম নির্ধারণ করে \\
\hline
\texttt{value} & প্রাথমিক মান বা বোতামের লেখা নির্ধারণ করে \\
\hline
\texttt{placeholder} & তথ্য লেখার ঘরে ইঙ্গিতমূলক লেখা দেখায় \\
\hline
\texttt{checked} & রেডিও বা চেকবক্স আগে থেকেই নির্বাচিত রাখে \\
\hline
\texttt{maxlength} & অক্ষরসংখ্যার সর্বোচ্চ সীমা নির্ধারণ করে \\
\hline
\texttt{size} & ঘরের দৃশ্যমান প্রস্থ নির্ধারণ করে \\
\hline
\texttt{required} & ঘর পূরণ বাধ্যতামূলক করে \\
\hline
\end{tabular}
\end{center}

\end{definitionbox}

\subsection{পাসওয়ার্ড ঘর}

\begin{definitionbox}{\texttt{type="password"}}
পাসওয়ার্ড ঘরে লেখা অক্ষরের বদলে সাধারণত বিন্দু বা তারকা চিহ্ন দেখা যায়। এতে পর্দায় মূল লেখাটি আড়াল থাকে।
\end{definitionbox}

\subsubsection{উদাহরণ}

\begin{chFourWholeExample}{কোড ও ব্রাউজারে ফল}
\noindent
\begin{minipage}[t]{.50\linewidth}\vspace{0pt}

\end{verbatim}

\end{minipage}\hfill
\begin{minipage}[t]{.46\linewidth}\vspace{0pt}
ব্যবহারকারী পাসওয়ার্ড লিখলে ব্রাউজার অক্ষরগুলো সরাসরি না দেখিয়ে ডট বা বুলেট দেখাতে পারে। এই আড়াল করা প্রদর্শন পাসওয়ার্ড এনক্রিপ্ট করে না।
\end{minipage}

\end{chFourWholeExample}

\subsection{সাধারণ ভুল}

\begin{mistakebox}{সাধারণ ভুল}
\begin{itemize}
\item সার্ভারে ফর্ম হিসেবে পাঠাতে চাওয়া ঘরকে কোনো ফর্মের সঙ্গে যুক্ত না করা। ফর্মের বাইরে ইনপুট লেখা নিজে সিনট্যাক্সের ভুল নয়; শেখার উদাহরণে সংশ্লিষ্ট ঘরগুলো \texttt{<form>}-এর মধ্যে রাখা সহজ।
\item \texttt{name} অ্যাট্রিবিউট না দেওয়া।
\item রেডিও বোতামের \texttt{name} আলাদা আলাদা দেওয়া।
\item সাবমিট বোতাম না দেওয়া।
\item \texttt{action} অ্যাট্রিবিউট ভুল দেওয়া।
\item পাসওয়ার্ড ঘরে \texttt{type="text"} লেখা।
\item চেকবক্স ও রেডিও বোতাম গুলিয়ে ফেলা।
\item \texttt{<select>}-এর ভিতরে \texttt{<option>} না দেওয়া।
\item \texttt{<textarea>} শেষের ট্যাগ না দেওয়া।
\item \texttt{label for} এবং ইনপুট \texttt{id} মিল না রাখা।
\end{itemize}
\end{mistakebox}

\subsubsection{ভুল রেডিও উদাহরণ}

\begin{examplebox}{কোড}
\par\noindent\begin{minipage}{\linewidth}
\begin{verbatim}

<input type="radio" name="female" value="Female"> Female
\end{verbatim}
\end{minipage}\par
\end{examplebox}
