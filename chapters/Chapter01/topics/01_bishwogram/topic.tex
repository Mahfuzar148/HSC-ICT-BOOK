\begin{topicbody}
\section{বিশ্বগ্রাম}
\label{sec:ch1-global-village}

একজন শিক্ষার্থী ঘরে বসে বিদেশের পাঠে অংশ নেয়। গ্রামের রোগী দূরের চিকিৎসকের পরামর্শ পান, আর স্থানীয় উদ্যোক্তা অনলাইনে পণ্য বিক্রি করেন। তথ্য ও যোগাযোগ প্রযুক্তি এসব কাজে সময় ও দূরত্বের বাধা কমায়। এই বৈশ্বিক সংযোগ বুঝতে প্রথমে ডেটা, তথ্য ও প্রযুক্তির সম্পর্ক জানা প্রয়োজন।

\subsection{ডেটা ও তথ্য}
\label{sec:ch1-data-information}
\definition{ডেটা (Data)}{কোনো ঘটনা, বস্তু, ব্যক্তি বা বিষয় সম্পর্কে সংগ্রহ করা কাঁচা উপাত্তকে ডেটা বলে।}
সংখ্যা, অক্ষর, চিহ্ন, পর্যবেক্ষণ ও রেকর্ড ডেটা হতে পারে। ছবি, শব্দ, ভিডিও, সেন্সরের পাঠ, লেনদেনের বিবরণ, পরীক্ষার নম্বর এবং চিকিৎসা পরীক্ষার ফলও ডেটার উদাহরণ। ডেটার অর্থ বুঝতে তার প্রসঙ্গ জানা দরকার। যেমন, ৮০, ৭৫ ও ৯০ তিনটি নম্বর; কার বা কোন বিষয়ের নম্বর, তা না জানলে পূর্ণ অর্থ বোঝা যায় না। কম্পিউটারে ডেটা ইনপুট হিসেবে ব্যবহৃত হয়।

\definition{তথ্য (Information)}{ডেটা প্রক্রিয়াকরণ, বিশ্লেষণ ও উপস্থাপনের ফলে পাওয়া অর্থপূর্ণ এবং ব্যবহারযোগ্য উপাত্তকে তথ্য বলে।}
তথ্য কোনো বিষয় বুঝতে ও সিদ্ধান্ত নিতে সাহায্য করে। একই ডেটা ভিন্ন উদ্দেশ্যে প্রক্রিয়াকরণ করলে ভিন্ন তথ্য পাওয়া যায়। বাংলা ৮০, ইংরেজি ৭৫ ও ICT ৯০ নম্বর থেকে মোট, গড়, গ্রেড, মেধাক্রম এবং উত্তীর্ণ হওয়ার অবস্থা নির্ণয় করা যায়। গ্রেড নির্ণয়ের ক্ষেত্রে সংশ্লিষ্ট মূল্যায়নের নিয়মও জানা প্রয়োজন।

\subtopichead{ভালো তথ্যের বৈশিষ্ট্য}
\begin{compactpoints}
\item \term{সঠিকতা}: ভুল তথ্য থেকে ভুল সিদ্ধান্ত হতে পারে।
\item \term{সময়োপযোগিতা}: প্রয়োজনের সময় তথ্য পাওয়া যায়।

\item \term{প্রাসঙ্গিকতা}: তথ্য ব্যবহারকারীর প্রয়োজনের সঙ্গে সম্পর্কিত হয়।

\end{compactpoints}

\subsection{ডেটা থেকে তথ্য তৈরির প্রক্রিয়া}
\definition{ডেটা প্রক্রিয়াকরণ (Data Processing)}{ডেটা যাচাই, সাজানো, গণনা, তুলনা ও বিশ্লেষণ করে ব্যবহারযোগ্য তথ্য তৈরির কাজকে ডেটা প্রক্রিয়াকরণ বলে।}
সব ডেটা সংগ্রহের পরই সরাসরি ব্যবহার করা যায় না। ভুল বা অসম্পূর্ণ উপাত্ত যাচাই করে প্রয়োজন অনুযায়ী সাজাতে হয়। প্রক্রিয়াকরণের ফল প্রতিবেদন, ছক, চার্ট বা সারাংশে প্রকাশ করা যায়।
\chOneFigureExplain{\chaponeimg/data_to_information_model.png}{ডেটা থেকে তথ্য তৈরির ধাপ}{fig:ch1-pair-1}{কাঁচা ডেটা ইনপুট হিসেবে নেওয়া হয়। যাচাই, সাজানো, গণনা ও বিশ্লেষণের পর অর্থপূর্ণ তথ্য পাওয়া যায়। সেই তথ্য সিদ্ধান্ত গ্রহণে ব্যবহৃত হয়।}
\begin{compactenum}
\item বিভিন্ন উৎস থেকে ডেটা সংগ্রহ করা।
\item ভুল, অসম্পূর্ণ ও পুনরাবৃত্ত ডেটা শনাক্ত করা।
\item নির্দিষ্ট ক্রম বা শ্রেণি অনুযায়ী ডেটা সাজানো।
\item গণনা, তুলনা, বিশ্লেষণ বা প্রয়োজনীয় রূপান্তর করা।
\item প্রাপ্ত তথ্য প্রতিবেদন, ছক, চার্ট বা সারাংশে উপস্থাপন করা।
\item তথ্যের ভিত্তিতে সিদ্ধান্ত নেওয়া।
\end{compactenum}

\subsection{তথ্যপ্রযুক্তি ও যোগাযোগ প্রযুক্তি}
\label{sec:ch1-ict-definition}
\definition{তথ্যপ্রযুক্তি (Information Technology বা IT)}{তথ্য সংগ্রহ, সংরক্ষণ, প্রক্রিয়াকরণ, বিশ্লেষণ, উপস্থাপন ও ব্যবহারের জন্য কম্পিউটার এবং সংশ্লিষ্ট ডিজিটাল প্রযুক্তির প্রয়োগকে তথ্যপ্রযুক্তি বলে।}
কম্পিউটার, সার্ভার, সংরক্ষণযন্ত্র, ডেটাবেজ, অপারেটিং সিস্টেম ও অ্যাপ্লিকেশন তথ্যপ্রযুক্তির কাজে ব্যবহৃত হয়। বিদ্যালয়ের ফলাফল প্রস্তুত, ব্যাংকের গ্রাহকতথ্য সংরক্ষণ, হাসপাতালের ব্যবস্থাপনা এবং গ্রন্থাগারের বইয়ের হিসাব এর পরিচিত উদাহরণ। তথ্য খোঁজা, সম্পাদনা, সাজানো, হালনাগাদ ও বিশ্লেষণ করে পরিকল্পনা এবং ব্যবস্থাপনায় ব্যবহার করা যায়। ব্যাকআপ, নিরাপত্তা ও প্রবেশাধিকার নিয়ন্ত্রণও এ ব্যবস্থার প্রয়োজনীয় কাজ।

\definition{যোগাযোগ প্রযুক্তি (Communication Technology বা CT)}{ডেটা, তথ্য, বার্তা, শব্দ, ভিডিও বা নথি এক স্থান থেকে অন্য স্থানে পাঠানো ও গ্রহণের জন্য ব্যবহৃত প্রযুক্তিকে যোগাযোগ প্রযুক্তি বলে।}
মানুষ, কম্পিউটার, মোবাইল যন্ত্র, সার্ভার ও নেটওয়ার্কে যুক্ত ব্যবস্থা এই প্রযুক্তির সাহায্যে তথ্য আদান-প্রদান করে। টেলিকমিউনিকেশন, মোবাইল যোগাযোগ, ইন্টারনেট, সম্প্রচার, স্যাটেলাইট যোগাযোগ এবং নেটওয়ার্কিং এর অন্তর্ভুক্ত।

\definition{তথ্য ও যোগাযোগ প্রযুক্তি (ICT)}{তথ্য সংগ্রহ, সংরক্ষণ, প্রক্রিয়াকরণ, বিশ্লেষণ ও আদান-প্রদানের জন্য কম্পিউটার, যোগাযোগ প্রযুক্তি, ডিজিটাল যন্ত্র, সফটওয়্যার ও নেটওয়ার্কভিত্তিক সেবার সমন্বিত ব্যবহারকে তথ্য ও যোগাযোগ প্রযুক্তি বলে। এর পূর্ণরূপ Information and Communication Technology।}
IT তথ্য নিয়ে কাজ করে; CT সেই তথ্য দূরে পৌঁছে দেয়। দুই ধরনের কাজের সমন্বয়ে মানুষ তথ্য তৈরি, ব্যবহার ও ভাগাভাগি করতে পারে। অফলাইনে ফলাফলের ডেটাবেজ তৈরি করা IT-এর কাজ। ফলাফল অনলাইন পোর্টালে প্রকাশ করলে তার সঙ্গে যোগাযোগ প্রযুক্তিও যুক্ত হয়।

\subsection{বিশ্বায়ন ও বিশ্বগ্রাম}
\definition{বিশ্বায়ন (Globalization)}{বিশ্বের বিভিন্ন দেশ, সমাজ, অর্থনীতি, সংস্কৃতি ও মানুষের মধ্যে সম্পর্ক, পারস্পরিক নির্ভরতা এবং আদান-প্রদান বৃদ্ধির প্রক্রিয়াকে বিশ্বায়ন বলে।}
আন্তর্জাতিক বাণিজ্য, বিনিয়োগ, বহুজাতিক প্রতিষ্ঠান, পরিবহন, গণমাধ্যম ও প্রযুক্তি এই প্রক্রিয়াকে এগিয়ে নেয়। পণ্য ও সেবার বাজার দেশের সীমানা ছাড়ায়; উৎপাদন, কর্মসংস্থান ও জ্ঞান বিনিময়ে বিভিন্ন দেশের মানুষ যুক্ত হয়। ডিজিটাল প্ল্যাটফর্ম অনলাইন ব্যবসা, দূরবর্তী সহযোগিতা এবং জ্ঞান ভাগাভাগি সহজ করে।

\definition{বিশ্বগ্রাম (Global Village)}{তথ্য ও যোগাযোগ প্রযুক্তির সাহায্যে পৃথিবীর বিভিন্ন স্থানের মানুষ দ্রুত যোগাযোগ ও তথ্য বিনিময়ের মাধ্যমে একটি গ্রামের মানুষের মতো ঘনিষ্ঠভাবে যুক্ত হওয়ার ধারণাকে বিশ্বগ্রাম বলে।}
ভৌগোলিক দূরত্ব থেকেই যায়, কিন্তু যোগাযোগে তার বাধা কমে। বার্তা, সংবাদ, ছবি, ভিডিও ও নথি দ্রুত দূরে পাঠানো যায়। শিক্ষা, চিকিৎসা, ব্যবসা, গবেষণা ও সহযোগিতায় বিভিন্ন দেশের মানুষ একই কার্যক্রমে অংশ নিতে পারে।

\subsection{বিশ্বগ্রাম ধারণার ইতিহাস}
কানাডীয় যোগাযোগ চিন্তাবিদ Marshall McLuhan বিশ্বগ্রাম ধারণার সঙ্গে বিশেষভাবে পরিচিত। ইলেকট্রনিক গণমাধ্যমের প্রভাবে দূরের মানুষ কীভাবে দ্রুত যুক্ত হচ্ছে, তা তিনি ব্যাখ্যা করেন। তাঁর \textit{The Gutenberg Galaxy: The Making of Typographic Man} প্রকাশিত হয় 1962 সালে; \textit{Understanding Media: The Extensions of Man} প্রকাশিত হয় 1964 সালে। প্রথম বইটি বিশ্বগ্রাম ধারণার সঙ্গে এবং দ্বিতীয়টি গণমাধ্যমের প্রভাবের বিস্তৃত আলোচনার সঙ্গে সম্পর্কিত। ইন্টারনেট জনপ্রিয় হওয়ার আগেই তিনি এ ধারণা তুলে ধরেন।

টেলিগ্রাফ, টেলিফোন, রেডিও, টেলিভিশন, কম্পিউটার, ইন্টারনেট, মোবাইল ফোন, স্যাটেলাইট যোগাযোগ ও World Wide Web-এর ধারাবাহিক উন্নয়ন বৈশ্বিক সংযোগ বাড়িয়েছে। সামাজিক যোগাযোগমাধ্যম, তাৎক্ষণিক বার্তা, অনলাইন সেবা ও সরাসরি ভিডিও যোগাযোগ সেই সংযোগকে দৈনন্দিন জীবনে নিয়ে এসেছে। বিশ্বগ্রাম কোনো এক দিনে তৈরি হওয়া ব্যবস্থা নয়; প্রযুক্তি, অর্থনৈতিক পরিবর্তন ও আন্তর্জাতিক সহযোগিতা এতে ভূমিকা রেখেছে।

\subsection{বিশ্বগ্রামের বৈশিষ্ট্য ও প্রয়োজনীয় উপাদান}
বিশ্বগ্রামে তথ্য দ্রুত আদান-প্রদান হয় এবং শিক্ষা, চিকিৎসা, ব্যবসা ও প্রশাসনের সেবা দূর থেকে নেওয়া যায়। বিভিন্ন দেশের জ্ঞান, সংস্কৃতি ও প্রযুক্তি ছড়িয়ে পড়ে। বৈশ্বিক বাজার, ফ্রিল্যান্সিং ও দূর থেকে কাজের সুযোগ বাড়ে। মানুষ ও প্রতিষ্ঠানের পারস্পরিক নির্ভরতাও বাড়ে।

এই ব্যবস্থা কার্যকর করতে হার্ডওয়্যার, সফটওয়্যার, সংযোগ, ডেটা বা তথ্য এবং মানুষের সক্ষমতা দরকার। পাঁচ উপাদানের কাজ ও বাস্তব প্রয়োগ অংশ \ref{sec:ch1-components}-এ রয়েছে। যন্ত্র থাকলেই সেবা পাওয়া যায় না; উপযুক্ত সফটওয়্যার, সচল সংযোগ, প্রয়োজনীয় তথ্য এবং ব্যবহারকারীর দক্ষতা একসঙ্গে কাজে লাগে।
\chOneFigureExplain{\chaponeimg/global_village_communication_tools.png}{বিশ্বগ্রামে যোগাযোগের মাধ্যম}{fig:ch1-pair-4}{ই-মেইল, তাৎক্ষণিক বার্তা, ভিডিও সভা, সামাজিক যোগাযোগমাধ্যম ও অনলাইনে নথি ভাগাভাগি করে মানুষ দূরে থেকেও আলোচনা ও সিদ্ধান্তে অংশ নেয়। এসব সেবার কাজ ও পার্থক্য পরের অংশে ব্যাখ্যা করা হয়েছে।}

\subsection{তথ্যভিত্তিক যোগাযোগ ও যাতায়াত}
বার্তা, ছবি বা পরামর্শ দূরে পাঠানো তথ্যভিত্তিক যোগাযোগ। মানুষ বা পণ্য এক স্থান থেকে অন্য স্থানে নেওয়া যাতায়াতভিত্তিক যোগাযোগ। অনলাইনে পণ্য অর্ডারের পর বাস্তবে পণ্য পৌঁছে দেওয়া দ্বিতীয় ধরনের কাজ। ভিডিও কল যোগাযোগের বাধা কমায়; পণ্যের পরিবহনের প্রয়োজন দূর করে না।

\subsection{বিশ্বগ্রামের সুযোগ ও ঝুঁকি}
দূরশিক্ষা, চিকিৎসা পরামর্শ, অনলাইন কেনাকাটা, ডিজিটাল লেনদেন, নাগরিক সেবা ও গবেষণায় বিশ্বগ্রামের সুফল পাওয়া যায়। সময়, শ্রম ও খরচ সাশ্রয় হতে পারে। তবে ব্যক্তিগত তথ্য ফাঁস, পরিচয় চুরি, হ্যাকিং, ফিশিং, ম্যালওয়্যার ও আর্থিক প্রতারণার ঝুঁকিও থাকে। ভুল সংবাদ ছড়াতে পারে; অতিরিক্ত ইন্টারনেট, গেম বা সামাজিক যোগাযোগমাধ্যম ব্যবহার পড়াশোনা ও স্বাস্থ্যের ক্ষতি করতে পারে। বাইরের সংস্কৃতির প্রভাবে স্থানীয় ভাষা ও মূল্যবোধ চাপের মুখে পড়তে পারে।

\subsection{ডিজিটাল বৈষম্য}
\label{sec:ch1-digital-divide}
\definition{ডিজিটাল বৈষম্য (Digital Divide)}{ব্যক্তি, পরিবার, প্রতিষ্ঠান, অঞ্চল বা দেশের মধ্যে ICT যন্ত্র, ইন্টারনেট, ডিজিটাল দক্ষতা, ভাষাগত সক্ষমতা ও অনলাইন সেবা ব্যবহারের সুযোগের পার্থক্যকে ডিজিটাল বৈষম্য বলে।}
বিশ্বগ্রাম সবার জন্য সমান সুযোগ নিশ্চিত করে না। স্মার্টফোন, কম্পিউটার, বিদ্যুৎ, নির্ভরযোগ্য সংযোগ ও নিরাপদ ব্যবহারের জ্ঞান না থাকলে মানুষ অনলাইন শিক্ষা, স্বাস্থ্যসেবা, ব্যবসা বা সরকারি সেবা থেকে পিছিয়ে পড়ে।
\begin{compactpoints}
\item যন্ত্র ও সংযোগের অভাব এবং বেশি খরচ সুযোগ সীমিত করে।
\item গ্রামে দুর্বল নেটওয়ার্ক, বিদ্যুৎ ও অবকাঠামোর ঘাটতি বাধা তৈরি করে।
\item ডিজিটাল শিক্ষা, প্রশিক্ষণ এবং স্থানীয় ভাষার পাঠের অভাব ব্যবহার কঠিন করে।
\item আয়, লিঙ্গ, প্রতিবন্ধিতা ও সামাজিক বাধার কারণে প্রবেশাধিকার কমতে পারে।
\end{compactpoints}
বৈষম্য কমাতে সাশ্রয়ী যন্ত্র ও সংযোগ, গ্রামীণ ব্রডব্যান্ড, বিদ্যুৎ এবং বিদ্যালয়ে হাতে-কলমে ICT শিক্ষা প্রয়োজন। বাংলা ও স্থানীয় ভাষায় পাঠ তৈরি, সাইবার নিরাপত্তার প্রশিক্ষণ এবং পিছিয়ে পড়া জনগোষ্ঠীর প্রবেশাধিকার বাড়ানোও জরুরি।

\exampleline{দুই শিক্ষার্থীর ফোন আছে। একজন অ্যাসাইনমেন্ট আপলোড করতে পারে না; অন্যজনের বাড়িতে স্থিতিশীল সংযোগ নেই। প্রথমজনের ব্যবহারিক দক্ষতা এবং দ্বিতীয়জনের নির্ভরযোগ্য সংযোগ ও ডাউনলোডযোগ্য পাঠ দরকার। দুজনকে শুধু নতুন ফোন দিলে সমস্যার পূর্ণ সমাধান হয় না।}

\subsection{বাংলাদেশে বিশ্বগ্রামের প্রয়োগ}
বাংলাদেশে অনলাইন ভর্তি, ফলাফল প্রকাশ, চাকরির আবেদন ও নাগরিক সেবায় বিশ্বগ্রামের প্রয়োগ দেখা যায়। মোবাইল ব্যাংকিং ও ডিজিটাল লেনদেন অর্থ আদান-প্রদান সহজ করে। স্থানীয় ও নারী উদ্যোক্তা সামাজিক যোগাযোগমাধ্যম বা অনলাইন বাজারে পণ্য বিক্রি করতে পারেন; দক্ষ কর্মী দেশি-বিদেশি গ্রাহকের কাজ করতে পারেন। কৃষি তথ্যসেবায় আবহাওয়া, বাজারদর ও চাষের পরামর্শ পাওয়া যায়। এসব সেবার কাজ, প্রয়োজনীয় দক্ষতা, সুবিধা ও সীমাবদ্ধতা পরের অংশে ক্ষেত্র অনুযায়ী ব্যাখ্যা করা হয়েছে।
