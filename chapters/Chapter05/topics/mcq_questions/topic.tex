\section{বহুনির্বাচনি প্রশ্ন (MCQ)}

\subsection*{\textbf{\underline{প্রোগ্রামিং ভাষার স্তর ও অনুবাদক}}}

\begin{multicols}{2}
\begin{enumerate}

\item যান্ত্রিক ভাষা (Machine Language) কোন স্তরের ভাষা? \textbf{[ঢাকা-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item নিম্ন স্তর \checkmark
\item উচ্চ স্তর
\item মধ্যম স্তর
\item চতুর্থ প্রজন্ম
\end{enumerate}

\item নিচের কোনটি চতুর্থ প্রজন্মের ভাষা (4GL)? \textbf{[সকল বোর্ড-২০১৮]}
\begin{enumerate}[label=\alph*)]
\item SQL \checkmark
\item C
\item BASIC
\item Assembly
\end{enumerate}

\item ইন্টারপ্রেটার (Interpreter) এর কাজ কী? \textbf{[চট্টগ্রাম-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item এক লাইন করে অনুবাদ করা \checkmark
\item পুরো প্রোগ্রাম একসাথে অনুবাদ করা
\item ডেটা ডিকোড করা
\item কোনোটিই নয়
\end{enumerate}

\item কম্পাইলার (Compiler) এবং ইন্টারপ্রেটারের মধ্যে প্রধান পার্থক্য কী? \textbf{[যশোর-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item অনুবাদের গতি \checkmark
\item ব্যবহারের ক্ষেত্র
\item আউটপুট মান
\item কোনোটিই নয়
\end{enumerate}

\item অ্যাসেম্বলি ভাষায় ব্যবহৃত সংকেতকে কী বলে? \textbf{[রাজশাহী-২০২২]}
\begin{enumerate}[label=\alph*)]
\item নেমোনিক (Mnemonic) \checkmark
\item বাইনারি
\item ডেসিমেল
\item হেক্সা
\end{enumerate}

\end{enumerate}
\end{multicols}

\subsection*{\textbf{\underline{অ্যালগরিদম, ফ্লোচার্ট ও সি-বেসিক}}}

\begin{multicols}{2}
\begin{enumerate}[resume]

\item ফ্লোচার্টে প্রক্রিয়াকরণ (Processing) এর জন্য কোন চিহ্ন ব্যবহৃত হয়? \textbf{[ঢাকা-২০২২]}
\begin{enumerate}[label=\alph*)]
\item আয়তক্ষেত্র \checkmark
\item বৃত্ত
\item সামান্তরিক
\item রম্বস
\end{enumerate}

\item ফ্লোচার্টে সিদ্ধান্ত (Decision) গ্রহণের জন্য কোন চিহ্ন ব্যবহৃত হয়? \textbf{[রাজশাহী-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item হীরা বা রম্বস \checkmark
\item আয়তক্ষেত্র
\item সামান্তরিক
\item উপবৃত্ত
\end{enumerate}

\item সি (C) প্রোগ্রামের আবশ্যিক ফাংশন কোনটি? \textbf{[চট্টগ্রাম-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item main() \checkmark
\item printf()
\item scanf()
\item return()
\end{enumerate}

\item printf() ফাংশনটি কোন হেডার ফাইলে থাকে? \textbf{[বরিশাল-২০২২]}
\begin{enumerate}[label=\alph*)]
\item stdio.h \checkmark
\item conio.h
\item math.h
\item string.h
\end{enumerate}

\item নিচের কোনটি সি ভাষার কীওয়ার্ড (Keyword)? \textbf{[সিলেট-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item int \checkmark
\item Integer
\item var
\item main
\end{enumerate}

\end{enumerate}
\end{multicols}

\subsection*{\textbf{\underline{অপারেটর, লুপ ও কন্ট্রোল স্টেটমেন্ট}}}

\begin{multicols}{2}
\begin{enumerate}[resume]

\item নিচের কোনটি রিলেশনাল অপারেটর? \textbf{[ঢাকা-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item >= \checkmark
\item +
\item \&\&
\item =
\end{enumerate}

\item মডুলাস (Modulus) অপারেটর (\%) কী কাজ করে? \textbf{[কুমিল্লা-২০২২]}
\begin{enumerate}[label=\alph*)]
\item ভাগশেষ নির্ণয় করে \checkmark
\item ভাগফল নির্ণয় করে
\item গুণফল নির্ণয় করে
\item যোগফল নির্ণয় করে
\end{enumerate}

\item লুপ থেকে বের হয়ে আসার জন্য কোন স্টেটমেন্ট ব্যবহৃত হয়? \textbf{[যশোর-২০২৩]}
\begin{enumerate}[label=\alph*)]
\item break \checkmark
\item continue
\item switch
\item default
\end{enumerate}

\item for(i=1; i<=10; i++) লুপটি কতবার চলবে? \textbf{[দিনাজপুর-২০২২]}
\begin{enumerate}[label=\alph*)]
\item ১০ বার \checkmark
\item ৯ বার
\item ১১ বার
\item অসংখ্যবার
\end{enumerate}

\item নিচের কোনটি কন্ডিশনাল কন্ট্রোল স্টেটমেন্ট? \textbf{[সিলেট-২০২২]}
\begin{enumerate}[label=\alph*)]
\item switch \checkmark
\item for
\item while
\item do-while
\end{enumerate}

\end{enumerate}
\end{multicols}

\subsection*{\textbf{\underline{বহুপদী সমাপ্তিসূচক ও অভিন্ন তথ্যভিত্তিক প্রশ্ন}}}

\begin{multicols}{2}
\begin{enumerate}[resume]

\item \textbf{নিচের উদ্দীপকটি পড়ো এবং প্রশ্নের উত্তর দাও:}

\textit{সি প্রোগ্রামে ১০টি পূর্ণ সংখ্যা সংরক্ষণের জন্য একটি ভেরিয়েবল প্রয়োজন।}

\item উদ্দীপকের কাজের জন্য নিচের কোনটি সঠিক?
\begin{enumerate}[label=\alph*)]
\item int n[10]; \checkmark
\item float n(10);
\item integer n{10};
\item char n[10];
\end{enumerate}

\item উচ্চস্তরের ভাষার সুবিধা হলো—
\begin{enumerate}[label=\Roman*.]
\item মানুষের জন্য বোঝা সহজ
\item হার্ডওয়্যারের ওপর নির্ভর করে না
\item প্রোগ্রাম দ্রুত রান করে
\end{enumerate}

নিচের কোনটি সঠিক?
\begin{enumerate}[label=\alph*)]
\item i ও ii \checkmark
\item i ও iii
\item ii ও iii
\item i, ii ও iii
\end{enumerate}

\item অ্যালগরিদম লেখার প্রধান উদ্দেশ্য হলো—
\begin{enumerate}[label=\Roman*.]
\item সমস্যার যৌক্তিক সমাধান বের করা
\item প্রোগ্রামিং কোড সহজ করা
\item হার্ডওয়্যার ঠিক করা
\end{enumerate}

নিচের কোনটি সঠিক?
\begin{enumerate}[label=\alph*)]
\item i ও ii \checkmark
\item i ও iii
\item ii ও iii
\item i, ii ও iii
\end{enumerate}

\end{enumerate}
\end{multicols}
