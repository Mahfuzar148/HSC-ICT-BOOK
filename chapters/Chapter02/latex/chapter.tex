\chapter{কমিউনিকেশন সিস্টেমস ও নেটওয়ার্কিং}

একটি কলেজে অনলাইন ভর্তি চলছে। শিক্ষার্থীর ফোন থেকে ভর্তি ফরমের ডেটা কলেজের সার্ভারে পৌঁছায়। শিক্ষক Wi-Fi দিয়ে ক্লাস নেন। ল্যাবের কম্পিউটারগুলো একই প্রিন্টার ব্যবহার করে এবং নোট ক্লাউডে সংরক্ষণ করা হয়। এসব কাজের জন্য ডেটা কমিউনিকেশন, উপযুক্ত মাধ্যম, নেটওয়ার্ক ডিভাইস ও সঠিক সংযোগবিন্যাস প্রয়োজন। এই অধ্যায়ে যোগাযোগব্যবস্থার কাজ এবং প্রয়োজন অনুযায়ী উপযুক্ত ব্যবস্থা নির্বাচন শেখানো হয়েছে।

\begin{learningbox}
এই অধ্যায় শেষে শিক্ষার্থী পারবে:
\begin{multicols}{2}
\begin{itemize}
\item ডেটা, তথ্য ও সংকেতের পার্থক্য করতে; প্রেরক, প্রাপক, বার্তা, মাধ্যম ও প্রোটোকলের ভূমিকা এবং অ্যানালগ ও ডিজিটাল সংকেত ব্যাখ্যা করতে।
\item ব্যান্ডউইথ, ডেটা হার, বিলম্ব ও প্রচলিত ব্যান্ডউইথের শ্রেণি আলাদা করতে; বিট, বাইট, ডেটা হার ও প্রেরণের দক্ষতার হিসাব করতে।
\item Serial ও Parallel এবং Asynchronous, Synchronous ও Isochronous পদ্ধতি তুলনা করতে।
\item Simplex, Half-duplex, Full-duplex এবং Unicast, Broadcast ও Multicast চিনতে।
\item তারযুক্ত ও তারবিহীন মাধ্যমের গঠন, প্রতিফলন, কাজ, সুবিধা ও সীমাবদ্ধতা ব্যাখ্যা করতে; দূরত্ব, নয়েজ, খরচ ও নিরাপত্তা অনুযায়ী মাধ্যম নির্বাচন করতে।
\item Bluetooth, Wi-Fi, WiMAX, সেল, বেস স্টেশন, Handoff, Roaming এবং 1G–5G-এর পরিবর্তন ও ব্যবহার ব্যাখ্যা করতে।
\item রিসোর্স ও ডেটা ভাগাভাগির প্রয়োজন ব্যাখ্যা করতে; PAN, LAN, CAN, MAN, WAN, Client–server, Peer-to-peer, Internet, Intranet ও Extranet আলাদা করতে।
\item NIC, মডেম, হাব, সুইচ, রাউটার, রিপিটার, ব্রিজ, গেটওয়ে, AP ও ফায়ারওয়ালের কাজ তুলনা করতে।
\item ভৌত ও যৌক্তিক টপোলজি শনাক্ত করতে; Bus, Star, Ring, Mesh, Tree ও Hybrid-এর চিত্র, সংযোগ নষ্ট হওয়ার ফল ও সম্প্রসারণ বিশ্লেষণ করতে।
\item ক্লাউডের কাজ, স্টোরেজ, অ্যাপ্লিকেশন, সেবা ও স্থাপনা মডেল, ব্যাকআপ, খরচ ও নিরাপত্তা ব্যাখ্যা করতে; উদ্দীপক থেকে MCQ/CQ সমাধান করতে।
\end{itemize}
\end{multicols}
\end{learningbox}

\begin{keypointbox}{অধ্যায়ের ধারণা মানচিত্র}
যোগাযোগের উপাদান ও সংকেত $\rightarrow$ ডেটা হার ও প্রেরণ $\rightarrow$ তারযুক্ত ও তারবিহীন মাধ্যম $\rightarrow$ মোবাইল নেটওয়ার্ক $\rightarrow$ কম্পিউটার নেটওয়ার্ক ও ডিভাইস $\rightarrow$ টপোলজি $\rightarrow$ ক্লাউড সেবা।
\end{keypointbox}
